    10→function hashPassword(pw: string): string {
    11→  const salt = crypto.randomBytes(16).toString("hex");
    12→  const hash = crypto.scryptSync(pw, salt, 32).toString("hex");
    13→  return `${salt}:${hash}`;
    14→}
