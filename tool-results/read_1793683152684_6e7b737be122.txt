   101→   101→      return "bg-emerald-500/10 text-emerald-700 dark:text-emerald-400";

   104→   104→    case "DONOR":

   106→   106→    case "BENEFICIARY":
   107→   107→      return "bg-fuchsia-500/10 text-fuchsia-700 dark:text-fuchsia-400";
   108→   108→    default:
   109→   109→      return "bg-muted text-muted-foreground";
   110→   110→  }
   111→   111→}

   113→   113→function statusTone(status: string): string {
   114→   114→  switch (status) {
   115→   115→    case "ACTIVE":
   116→   116→      return "bg-emerald-500/10 text-emerald-700 dark:text-emerald-400";
   117→   117→    case "PENDING":

   121→   121→    default:

   123→   123→  }

   141→   141→
   142→   142→export default function UsersView() {
   143→   143→  const { user } = useAuth();
   144→   144→  const [rows, setRows] = useState<UserRow[]>([]);
   145→   145→  const [loading, setLoading] = useState(true);
   146→   146→  const [saving, setSaving] = useState(false);
   147→   147→  const [q, setQ] = useState("");
   148→   148→  const [search, setSearch] = useState("");
   149→   149→  const [role, setRole] = useState("ALL");
   150→   150→  const [status, setStatus] = useState("ALL");
   151→   151→
   152→   152→  const [createOpen, setCreateOpen] = useState(false);
   153→   153→  const [createForm, setCreateForm] = useState({ ...emptyCreateForm });
   154→   154→  const [editTarget, setEditTarget] = useState<UserRow | null>(null);
   155→   155→  const [editForm, setEditForm] = useState({ ...emptyEditForm });
   156→   156→  const [roleConfirmOpen, setRoleConfirmOpen] = useState(false);
   157→   157→  const [deleteTarget, setDeleteTarget] = useState<UserRow | null>(null);
   158→   158→  const [viewTarget, setViewTarget] = useState<UserRow | null>(null);
   159→   159→  const [donorStats, setDonorStats] = useState<{ count: number; sum: number } | null>(null);
   160→   160→
   161→   161→  /* -------------------------------- loading -------------------------------- */
   162→   162→
   163→   163→  const load = useCallback(async () => {
   164→   164→    setLoading(true);
   165→   165→    try {
   166→   166→      const data = await get<UserRow[] | { items?: UserRow[] }>("/api/users?limit=500");
   167→   167→      setRows(Array.isArray(data) ? data : (data.items ?? []));
   168→   168→    } catch (e) {
   169→   169→      toast.error(e instanceof Error ? e.message : "Failed to load users");
   170→   170→    } finally {
   171→   171→      setLoading(false);
   172→   172→    }
   173→   173→  }, []);
   174→   174→
   175→   175→  useEffect(() => {
   176→   176→    load();
   177→   177→  }, [load]);
   178→   178→
   179→   179→  useEffect(() => {
   180→   180→    const t = setTimeout(() => setSearch(q), 350);
   181→   181→    return () => clearTimeout(t);
   182→   182→  }, [q]);
   183→   183→
   184→   184→  const filtered = useMemo(() => {
   185→   185→    const needle = search.trim().toLowerCase();
   186→   186→    return rows.filter((u) => {
   187→   187→      if (role !== "ALL" && u.role !== role) return false;
   188→   188→      if (status !== "ALL" && u.status !== status) return false;
   189→   189→      if (needle && !`${u.name} ${u.email} ${u.phone ?? ""} ${u.city ?? ""}`.toLowerCase().includes(needle)) return false;
   190→   190→      return true;
   191→   191→    });
   192→   192→  }, [rows, search, role, status]);
   193→   193→
   194→   194→  /* -------------------------------- actions ------------------------------- */
   195→   195→
   196→   196→  const createUser = async () => {
   197→   197→    if (!createForm.name.trim() || !createForm.email.trim() || !createForm.password) {
   198→   198→      toast.error("Name, email and password are required");
   199→   199→      return;
   200→   200→    }
   201→   201→    if (!EMAIL_RE.test(createForm.email.trim())) {
   202→   202→      toast.error("Enter a valid email address");
   203→   203→      return;
   204→   204→    }
   205→   205→    if (createForm.password.length < 8) {
   206→   206→      toast.error("Password must be at least 8 characters");
   207→   207→      return;
   208→   208→    }
   209→   209→    setSaving(true);
   210→   210→    try {
   211→   211→      await post("/api/users", {
   212→   212→        name: createForm.name.trim(),
   213→   213→        email: createForm.email.trim(),
   214→   214→        password: createForm.password,
   215→   215→        role: createForm.role,
   216→   216→        city: createForm.city.trim() || undefined,
   217→   217→        phone: createForm.phone.trim() || undefined,
   218→   218→      });
   219→   219→      toast.success(`User ${createForm.name.trim()} created`);
   220→   220→      setCreateOpen(false);
   221→   221→      setCreateForm({ ...emptyCreateForm });
   222→   222→      await load();
   223→   223→    } catch (e) {
   224→   224→      toast.error(e instanceof Error ? e.message : "Failed to create user");
   225→   225→    } finally {
   226→   226→      setSaving(false);
   227→   227→    }
   228→   228→  };
   229→   229→
   230→   230→  const openEdit = (u: UserRow) => {
   231→   231→    setEditTarget(u);
   232→   232→    setEditForm({
   233→   233→      name: u.name ?? "",
   234→   234→      phone: u.phone ?? "",
   235→   235→      city: u.city ?? "",
   236→   236→      pan: u.pan ?? "",
   237→   237→      about: u.about ?? "",
   238→   238→      role: u.role ?? "DONOR",
   239→   239→      status: u.status ?? "ACTIVE",
   240→   240→      isActive: u.isActive ?? u.status === "ACTIVE",
   241→   241→      password: "",
   242→   242→    });
   243→   243→  };
   244→   244→
   245→   245→  const saveEdit = () => {
   246→   246→    if (!editTarget?.id) {
   247→   247→      toast.error("No user selected — close the dialog and try again");
   248→   248→      return;
   249→   249→    }
   250→   250→    if (!editForm.name.trim()) {
   251→   251→      toast.error("Name cannot be empty");
   252→   252→      return;
   253→   253→    }
   254→   254→    if (editForm.password && editForm.password.length < 8) {
   255→   255→      toast.error("Password must be at least 8 characters");
   256→   256→      return;
   257→   257→    }
   258→   258→    if (editForm.role !== editTarget.role) {
   259→   259→      setRoleConfirmOpen(true);
   260→   260→      return;
   261→   261→    }
   262→   262→    doSaveEdit();
   263→   263→  };
   264→   264→
   265→   265→  const doSaveEdit = async () => {
   266→   266→    if (!editTarget?.id) return;
   267→   267→    setRoleConfirmOpen(false);
   268→   268→    setSaving(true);
   269→   269→    try {
   270→   270→      await patch(`/api/users/${editTarget.id}`, {
   271→   271→        name: editForm.name.trim(),
   272→   272→        phone: editForm.phone.trim() || null,
   273→   273→        city: editForm.city.trim() || null,
   274→   274→        pan: editForm.pan.trim().toUpperCase() || null,
   275→   275→        about: editForm.about.trim() || null,
   276→   276→        role: editForm.role,
   277→   277→        status: editForm.status,
   278→   278→        isActive: editForm.isActive,
   279→   279→        ...(editForm.password ? { password: editForm.password } : {}),
   280→   280→      });
   281→   281→      toast.success(`${editForm.name.trim()} updated`);
   282→   282→      setEditTarget(null);
   283→   283→      await load();
   284→   284→    } catch (e) {
   285→   285→      toast.error(e instanceof Error ? e.message : "Failed to update user");
   286→   286→    } finally {
   287→   287→      setSaving(false);
   288→   288→    }
   289→   289→  };
   290→   290→
   291→   291→  const confirmDelete = async () => {
   292→   292→    if (!deleteTarget) return;
   293→   293→    if (deleteTarget.id === user?.id) {
   294→   294→      toast.error("You cannot delete your own account");
   295→   295→      setDeleteTarget(null);
   296→   296→      return;
   297→   297→    }
   298→   298→    setSaving(true);
   299→   299→    try {
   300→   300→      await del(`/api/users/${deleteTarget.id}`);
   301→   301→      toast.success("User deleted");
   302→   302→      setDeleteTarget(null);
   303→   303→      await load();
   304→   304→    } catch (e) {
   305→   305→      toast.error(e instanceof Error ? e.message : "Failed to delete user");
   306→   306→    } finally {
   307→   307→      setSaving(false);
   308→   308→    }
   309→   309→  };
   310→   310→
   311→   311→  const openView = (u: UserRow) => {
   312→   312→    setViewTarget(u);
   313→   313→    setDonorStats(null);
   314→   314→    if (u.role === "DONOR") {
   315→   315→      get<DonationLite[] | { items?: DonationLite[] }>("/api/donations?limit=500")
   316→   316→        .then((data) => {
   317→   317→          const list = Array.isArray(data) ? data : (data.items ?? []);
   318→   318→          const mine = list.filter((d) => d.donorId === u.id && d.paymentStatus === "COMPLETED");
   319→   319→          setDonorStats({
   320→   320→            count: mine.length,
   321→   321→            sum: mine.reduce((acc, d) => acc + (Number(d.amount) || 0), 0),
   322→   322→          });
   323→   323→        })
   324→   324→        .catch(() => setDonorStats(null));
   325→   325→    }
   326→   326→  };
   327→   327→
   328→   328→  const rowActions = (u: UserRow, withLabels = false) => {
   329→   329→    const self = u.id === user?.id;
   330→   330→    return (
   331→   331→      <div className={cn("items-center", withLabels ? "flex flex-wrap gap-2" : "flex justify-end gap-1")}>
   332→   332→        <Button
   333→   333→          size={withLabels ? "sm" : "icon"}
   334→   334→          variant="ghost"
   335→   335→          className={withLabels ? "" : "size-11 md:size-9"}
   336→   336→          aria-label={`View user ${u.name}`}
   337→   337→          onClick={() => openView(u)}
   338→   338→        >
   339→   339→          <Eye className="size-4" />
   340→   340→          {withLabels && <span>View</span>}
   341→   341→        </Button>
   342→   342→        <Button
   343→   343→          size={withLabels ? "sm" : "icon"}
   344→   344→          variant="ghost"
   345→   345→          className={withLabels ? "" : "size-11 md:size-9"}
   346→   346→          aria-label={`Edit user ${u.name}`}
   347→   347→          onClick={() => openEdit(u)}
   348→   348→        >
   349→   349→          <Pencil className="size-4" />
   350→   350→          {withLabels && <span>Edit</span>}
   351→   351→        </Button>
   352→   352→        <Button
   353→   353→          size={withLabels ? "sm" : "icon"}
   354→   354→          variant="ghost"
   355→   355→          className={cn(withLabels ? "text-red-600 hover:bg-red-500/10 hover:text-red-700" : "size-11 md:size-9 text-red-600 hover:bg-red-500/10 hover:text-red-700")}
   356→   356→          aria-label={self ? "You cannot delete your own account" : `Delete user ${u.name}`}
   357→   357→          disabled={self}
   358→   358→          onClick={() => setDeleteTarget(u)}
   359→   359→        >
   360→   360→          <Trash2 className="size-4" />
   361→   361→          {withLabels && <span>Delete</span>}
   362→   362→        </Button>
   363→   363→      </div>
   364→   364→    );
   365→   365→  };
   366→   366→
   367→   367→  /* --------------------------------- render -------------------------------- */
   368→   368→
   369→   369→  return (
   370→   370→    <div className="space-y-4 sm:space-y-6">
   371→   371→      <PageHeader
   372→   372→        title="Users & Roles"
   373→   373→        description="Manage accounts, roles and access across the trust."
   374→   374→        icon={<Users className="size-5" aria-hidden="true" />}
   375→   375→        actions={
   376→   376→          <Button onClick={() => setCreateOpen(true)} aria-label="Create new user">
   377→   377→            <Plus className="size-4" /> New user
   378→   378→          </Button>
   379→   379→        }
   380→   380→      />
   381→   381→
   382→   382→      {/* Filters */}
   383→   383→      <div className="flex flex-col gap-3 lg:flex-row lg:items-center">
   384→   384→        <div className="relative flex-1">
   385→   385→          <Search className="absolute left-3 top-1/2 size-4 -translate-y-1/2 text-muted-foreground" aria-hidden="true" />
   386→   386→          <Input value={q} onChange={(e) => setQ(e.target.value)} placeholder="Search name, email, phone or city…" className="pl-9" aria-label="Search users" />
   387→   387→        </div>
   388→   388→        <div className="flex flex-wrap gap-2">
   389→   389→          <Select value={role} onValueChange={setRole}>
   390→   390→            <SelectTrigger className="w-full sm:w-40" aria-label="Filter by role">
   391→   391→              <SelectValue placeholder="All roles" />
   392→   392→            </SelectTrigger>
   393→   393→            <SelectContent>
   394→   394→              <SelectItem value="ALL">All roles</SelectItem>
   395→   395→              {ROLES.map((r) => (
   396→   396→                <SelectItem key={r} value={r}>{titleCase(r)}</SelectItem>
   397→   397→              ))}
   398→   398→            </SelectContent>
   399→   399→          </Select>
   400→   400→          <Select value={status} onValueChange={setStatus}>
   401→   401→            <SelectTrigger className="w-full sm:w-40" aria-label="Filter by status">
   402→   402→              <SelectValue placeholder="All statuses" />
   403→   403→            </SelectTrigger>
   404→   404→            <SelectContent>
   405→   405→              <SelectItem value="ALL">All statuses</SelectItem>
   406→   406→              {STATUSES.map((s) => (
   407→   407→                <SelectItem key={s} value={s}>{titleCase(s)}</SelectItem>
   408→   408→              ))}
   409→   409→            </SelectContent>
   410→   410→          </Select>
   411→   411→          <span className="ml-auto self-center text-xs text-muted-foreground" aria-live="polite">
   412→   412→            {loading ? "Loading…" : `${filtered.length} of ${rows.length} users`}
   413→   413→          </span>
   414→   414→        </div>
   415→   415→      </div>
   416→   416→
   417→   417→      {/* Desktop table */}
   418→   418→      <div className="hidden rounded-xl border bg-card md:block">
   419→   419→        {loading ? (
   420→   420→          <div className="space-y-2 p-4">
   421→   421→            {Array.from({ length: 6 }).map((_, i) => (
   422→   422→              <Skeleton key={i} className="h-11 w-full" />
   423→   423→            ))}
   424→   424→          </div>
   425→   425→        ) : filtered.length === 0 ? (
   426→   426→          <div className="p-6">
   427→   427→            <EmptyState
   428→   428→              icon={<Users className="size-6" aria-hidden="true" />}
   429→   429→              title="No users found"
   430→   430→              hint="Adjust the filters or create a new account."
   431→   431→            >
   432→   432→              <Button size="sm" variant="outline" onClick={() => { setQ(""); setRole("ALL"); setStatus("ALL"); }}>
   433→   433→                Clear filters
   434→   434→              </Button>
   435→   435→            </EmptyState>
   436→   436→          </div>
   437→   437→        ) : (
   438→   438→          <div className="overflow-x-auto">
   439→   439→            <Table>
   440→   440→              <TableHeader>
   441→   441→                <TableRow>
   442→   442→                  <TableHead>User</TableHead>
   443→   443→                  <TableHead>Role</TableHead>
   444→   444→                  <TableHead>Status</TableHead>
   445→   445→                  <TableHead className="hidden md:table-cell">City</TableHead>
   446→   446→                  <TableHead className="hidden lg:table-cell">Phone</TableHead>
   447→   447→                  <TableHead className="hidden sm:table-cell">Joined</TableHead>
   448→   448→                  <TableHead className="text-right">Actions</TableHead>
   449→   449→                </TableRow>
   450→   450→              </TableHeader>
   451→   451→              <TableBody>
   452→   452→                {filtered.map((u) => {
   453→   453→                  const self = u.id === user?.id;
   454→   454→                  return (
   455→   455→                    <TableRow key={u.id}>
   456→   456→                      <TableCell>
   457→   457→                        <div className="flex items-center gap-2.5">
   458→   458→                          <span className="flex size-8 shrink-0 items-center justify-center rounded-full bg-primary/10 text-xs font-semibold text-primary" aria-hidden="true">
   459→   459→                            {initials(u.name)}
   460→   460→                          </span>
   461→   461→                          <div className="min-w-0">
   462→   462→                            <p className="truncate font-medium">
   463→   463→                              {u.name}
   464→   464→                              {self && <span className="ml-1.5 rounded bg-muted px-1 py-0.5 text-[10px] font-semibold text-muted-foreground">You</span>}
   465→   465→                            </p>
   466→   466→                            <p className="truncate text-xs text-muted-foreground">{u.email}</p>
   467→   467→                          </div>
   468→   468→                        </div>
   469→   469→                      </TableCell>
   470→   470→                      <TableCell>
   471→   471→                        <span className={cn("whitespace-nowrap rounded-full px-2 py-0.5 text-[11px] font-semibold", roleTone(u.role))}>{titleCase(u.role)}</span>
