   124→   124→
   125→   125→function isLowStock(i: InventoryItem): boolean {
   126→   126→  return i.quantity <= i.minStock;
   127→   127→}
   128→   128→
   129→   129→function isExpiring(i: InventoryItem): boolean {
   130→   130→  if (!i.expiryDate) return false;
   131→   131→  const d = new Date(i.expiryDate);
   132→   132→  if (Number.isNaN(d.getTime())) return false;
   133→   133→  const in30 = Date.now() + 30 * 24 * 60 * 60 * 1000;
   134→   134→  return d.getTime() <= in30;
   135→   135→}

   351→   351→              <RefreshCw className="size-4" aria-hidden="true" /> Refresh
   352→   352→            </Button>
   353→   353→            <Button onClick={openCreate}>
   354→   354→              <Plus className="size-4" aria-hidden="true" /> New item

   364→   364→          label="Low stock"

   372→   372→        <StatCard
