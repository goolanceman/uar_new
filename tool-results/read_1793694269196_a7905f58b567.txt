   131→  name: string;
   132→  email: string;
   133→  phone: string;
   134→  city: string;

   136→  about: string;
   137→  role: string;
   138→  status: string;
   139→  isActive: boolean;
   140→  password: string;

   144→  name: "",
   145→  email: "",
   146→  password: "",
   147→  role: "DONOR",

   149→  phone: "",

   153→  name: "",
   154→  email: "",
   155→  phone: "",
   156→  city: "",
   157→  pan: "",
   158→  about: "",
   159→  role: "DONOR",
   160→  status: "ACTIVE",
   161→  isActive: true,
   162→  password: "",

   164→
   165→function formFromRow(u: UserRow): EditForm {
   166→  return {
   167→    name: u.name ?? "",
   168→    email: u.email ?? "",
   169→    phone: u.phone ?? "",
   170→    city: u.city ?? "",
   171→    pan: u.pan ?? "",
   172→    about: u.about ?? "",
   173→    role: u.role ?? "DONOR",
   174→    status: u.status ?? "ACTIVE",
   175→    isActive: u.isActive ?? u.status === "ACTIVE",
   176→    password: "",
   177→  };
   178→}

   180→/* ----------------------------------- view ----------------------------------- */
   181→
   182→export default function UsersView() {
   183→  const { user } = useAuth();
   184→
   185→  const [rows, setRows] = useState<UserRow[]>([]);
   186→  const [loading, setLoading] = useState(true);
   187→  const [saving, setSaving] = useState(false);
   188→
   189→  // toolbar
   190→  const [q, setQ] = useState("");
   191→  const [search, setSearch] = useState("");
   192→  const [roleFilter, setRoleFilter] = useState("ALL");
   193→  const [statusFilter, setStatusFilter] = useState("ALL");
   194→
   195→  // pagination + view mode
   196→  const [page, setPage] = useState(1);
   197→  const [pageSize, setPageSize] = useState(25);
   198→  const [mode, setMode] = useViewPref("users");
   199→
   200→  // create / edit
   201→  const [createOpen, setCreateOpen] = useState(false);
   202→  const [createForm, setCreateForm] = useState<CreateForm>({ ...emptyCreateForm });
   203→  const [editOpen, setEditOpen] = useState(false);
   204→  const [editTarget, setEditTarget] = useState<UserRow | null>(null);
   205→  const [editForm, setEditForm] = useState<EditForm>({ ...emptyEditForm });
   206→  const [editSnapshot, setEditSnapshot] = useState<string>(JSON.stringify(emptyEditForm));
   207→
   208→  // inline role change confirm (table cell select)
   209→  const [roleChange, setRoleChange] = useState<{ target: UserRow; role: string } | null>(null);
   210→
   211→  // delete
   212→  const [deleteTarget, setDeleteTarget] = useState<UserRow | null>(null);
   213→
   214→  // detail sheet + linked data
   215→  const [viewTarget, setViewTarget] = useState<UserRow | null>(null);
   216→  const [donorStats, setDonorStats] = useState<
   217→    { count: number; sum: number } | "unavailable" | null
   218→  >(null);
   219→  const [linkedVolunteer, setLinkedVolunteer] = useState<VolunteerLite | null>(null);
   220→  const [linkedBeneficiary, setLinkedBeneficiary] = useState<BeneficiaryLite | null>(null);
   221→  const [activityCount, setActivityCount] = useState<number | null>(null);
   222→
   223→  const adminCount = useMemo(() => rows.filter((r) => r.role === "ADMIN").length, [rows]);
   224→
   225→  /* -------------------------------- loading -------------------------------- */
   226→
   227→  const load = useCallback(async () => {
   228→    setLoading(true);
   229→    try {
   230→      const data = await get<UserRow[] | { items?: UserRow[] }>("/api/users?limit=500");
   231→      setRows(Array.isArray(data) ? data : (data.items ?? []));
   232→    } catch (e) {
   233→      toast.error(e instanceof Error ? e.message : "Failed to load users");
   234→    } finally {
   235→      setLoading(false);
   236→    }
   237→  }, []);
   238→
   239→  useEffect(() => {
   240→    load();
   241→  }, [load]);
   242→
   243→  useEffect(() => {
   244→    const t = setTimeout(() => setSearch(q), 300);
   245→    return () => clearTimeout(t);
   246→  }, [q]);
   247→
   248→  const filtered = useMemo(() => {
   249→    const needle = search.trim().toLowerCase();
   250→    return rows.filter((u) => {
   251→      if (roleFilter !== "ALL" && u.role !== roleFilter) return false;
   252→      if (statusFilter !== "ALL" && u.status !== statusFilter) return false;
   253→      if (
   254→        needle &&
   255→        !`${u.name} ${u.email} ${u.phone ?? ""} ${u.city ?? ""}`.toLowerCase().includes(needle)
   256→      )
   257→        return false;
   258→      return true;
   259→    });
   260→  }, [rows, search, roleFilter, statusFilter]);
   261→
   262→  const pages = Math.max(1, Math.ceil(filtered.length / pageSize));
   263→  useEffect(() => {
   264→    setPage(1);
   265→  }, [search, roleFilter, statusFilter, pageSize]);
   266→  const paged = useMemo(
   267→    () => filtered.slice((page - 1) * pageSize, page * pageSize),
   268→    [filtered, page, pageSize]
   269→  );
   270→
   271→  const hasFilters = roleFilter !== "ALL" || statusFilter !== "ALL" || search.trim() !== "";
   272→
   273→  const clearFilters = () => {
   274→    setQ("");
   275→    setRoleFilter("ALL");
   276→    setStatusFilter("ALL");
   277→  };
   278→
   279→  /* -------------------------------- create ---------------------------------- */
   280→
   281→  const createDirty =
   282→    createOpen && JSON.stringify(createForm) !== JSON.stringify(emptyCreateForm);
   283→
   284→  const submitCreate = async () => {
   285→    const errors: string[] = [];
   286→    const nameErr = required("Name")(createForm.name);
   287→    if (nameErr) errors.push(nameErr);
   288→    const emailErr = emailRule()(createForm.email);
   289→    if (emailErr) errors.push(emailErr);
   290→    const passErr = minLen(6, "Password")(createForm.password);
   291→    if (passErr) errors.push(passErr);
   292→    if (errors.length) {
   293→      toast.error(errors[0]);
   294→      return;
   295→    }
   296→    setSaving(true);
   297→    try {
   298→      await post("/api/users", {
   299→        name: createForm.name.trim(),
   300→        email: createForm.email.trim().toLowerCase(),
   301→        password: createForm.password,
   302→        role: createForm.role,
   303→        city: createForm.city.trim() || undefined,
   304→        phone: createForm.phone.trim() || undefined,
   305→      });
   306→      toast.success(`User ${createForm.name.trim()} created`);
   307→      setCreateOpen(false);
   308→      setCreateForm({ ...emptyCreateForm });
   309→      await load();
   310→    } catch (e) {
   311→      toast.error(e instanceof Error ? e.message : "Failed to create user");
   312→    } finally {
   313→      setSaving(false);
   314→    }
   315→  };
   316→
   317→  /* --------------------------------- edit ----------------------------------- */
   318→
   319→  const openEdit = (u: UserRow) => {
   320→    setEditTarget(u);
   321→    const next = formFromRow(u);
   322→    setEditForm(next);
   323→    setEditSnapshot(JSON.stringify(next));
   324→    setEditOpen(true);
   325→  };
   326→
   327→  const editDirty = editOpen && JSON.stringify(editForm) !== editSnapshot;
   328→
   329→  const doSaveEdit = async () => {
   330→    if (!editTarget) return;
   331→    setSaving(true);
   332→    try {
   333→      await patch(`/api/users/${editTarget.id}`, {
   334→        name: editForm.name.trim(),
   335→        email: editForm.email.trim().toLowerCase(),
   336→        phone: editForm.phone.trim() || null,
   337→        city: editForm.city.trim() || null,
   338→        pan: editForm.pan.trim() ? editForm.pan.trim().toUpperCase() : null,
   339→        about: editForm.about.trim() || null,
   340→        role: editForm.role,
   341→        status: editForm.status,
   342→        isActive: editForm.isActive,
   343→        ...(editForm.password ? { password: editForm.password } : {}),
   344→      });
   345→      toast.success(`${editForm.name.trim()} updated`);
   346→      setEditOpen(false);
   347→      setEditTarget(null);
   348→      await load();
   349→    } catch (e) {
   350→      toast.error(e instanceof Error ? e.message : "Failed to update user");
   351→    } finally {
   352→      setSaving(false);
   353→    }
   354→  };
   355→
   356→  const submitEdit = () => {
   357→    if (!editTarget) return;
   358→    const errors: string[] = [];
   359→    const nameErr = required("Name")(editForm.name);
   360→    if (nameErr) errors.push(nameErr);
   361→    const emailErr = emailRule()(editForm.email);
   362→    if (emailErr) errors.push(emailErr);
   363→    const panErr = panRule(true)(editForm.pan);
   364→    if (panErr) errors.push(panErr);
   365→    const phoneErr = phoneRule(true)(editForm.phone);
   366→    if (phoneErr) errors.push(phoneErr);
   367→    if (editForm.password) {
   368→      const passErr = minLen(6, "New password")(editForm.password);
   369→      if (passErr) errors.push(passErr);
   370→    }
   371→    if (errors.length) {
   372→      toast.error(errors[0]);
   373→      return;
   374→    }
   375→    if (editForm.role !== editTarget.role) {
   376→      // confirm via the shared role-change dialog
   377→      setRoleChange({ target: editTarget, role: editForm.role });
   378→      return;
   379→    }
   380→    void doSaveEdit();
   381→  };
   382→
   383→  /* ---------------------------- inline role change --------------------------- */
   384→
   385→  const confirmRoleChange = async () => {
   386→    if (!roleChange) return;
   387→    const { target, role } = roleChange;
   388→    if (target.role === "ADMIN" && adminCount <= 1 && role !== "ADMIN") {
   389→      toast.error("You cannot change the role of the only administrator");
   390→      setRoleChange(null);
   391→      return;
   392→    }
   393→    setSaving(true);
   394→    try {
   395→      await patch(`/api/users/${target.id}`, { role });
   396→      toast.success(`${target.name} is now ${titleCase(role)}`);
   397→      setRoleChange(null);
   398→      await load();
   399→    } catch (e) {
   400→      toast.error(e instanceof Error ? e.message : "Failed to change role");
   401→    } finally {
   402→      setSaving(false);
   403→    }
   404→  };
   405→
   406→  /* --------------------------------- delete ---------------------------------- */
   407→
   408→  const isSelf = (u: UserRow) => u.id === user?.id;
   409→  const isLastAdmin = (u: UserRow) => u.role === "ADMIN" && adminCount <= 1;
   410→
   411→  const confirmDelete = async () => {
   412→    if (!deleteTarget) return;
   413→    if (isSelf(deleteTarget)) {
   414→      toast.error("You cannot delete your own account");
   415→      setDeleteTarget(null);
   416→      return;
   417→    }
   418→    if (isLastAdmin(deleteTarget)) {
   419→      toast.error("You cannot delete the only administrator account");
   420→      setDeleteTarget(null);
   421→      return;
   422→    }
   423→    setSaving(true);
   424→    try {
   425→      await del(`/api/users/${deleteTarget.id}`);
   426→      toast.success("User deleted");
   427→      setDeleteTarget(null);
   428→      setViewTarget((v) => (v && v.id === deleteTarget.id ? null : v));
   429→      await load();
   430→    } catch (e) {
   431→      toast.error(e instanceof Error ? e.message : "Failed to delete user");
   432→    } finally {
   433→      setSaving(false);
   434→    }
   435→  };
   436→
   437→  /* ------------------------------ detail sheet ------------------------------- */
   438→
   439→  const openView = (u: UserRow) => {
   440→    setViewTarget(u);
   441→    setDonorStats(null);
   442→    setLinkedVolunteer(null);
   443→    setLinkedBeneficiary(null);
   444→    setActivityCount(null);
   445→
   446→    if (u.role === "DONOR") {
   447→      // Aggregation is capped at the latest 500 donations (API limit) — close
   448→      // enough for a summary badge, not an audited statement.
   449→      get<DonationLite[] | { items?: DonationLite[] }>("/api/donations?limit=500")
   450→        .then((data) => {
   451→          const list = Array.isArray(data) ? data : (data.items ?? []);
   452→          const mine = list.filter(
   453→            (d) => d.donorId === u.id && d.paymentStatus === "COMPLETED"
   454→          );
   455→          setDonorStats({
   456→            count: mine.length,
   457→            sum: mine.reduce((acc, d) => acc + (Number(d.amount) || 0), 0),
   458→          });
   459→        })
   460→        .catch(() => setDonorStats("unavailable"));
   461→    }
   462→
   463→    if (u.role === "VOLUNTEER") {
   464→      get<VolunteerLite[]>("/api/volunteers")
   465→        .then((list) => {
   466→          const v = Array.isArray(list) ? list.find((x) => x.userId === u.id) : undefined;
   467→          setLinkedVolunteer(v ?? null);
   468→        })
   469→        .catch(() => setLinkedVolunteer(null));
   470→    }
   471→
   472→    if (u.role === "BENEFICIARY") {
   473→      get<BeneficiaryLite[]>("/api/beneficiaries")
   474→        .then((list) => {
   475→          const b = Array.isArray(list) ? list.find((x) => x.userId === u.id) : undefined;
   476→          setLinkedBeneficiary(b ?? null);
   477→        })
   478→        .catch(() => setLinkedBeneficiary(null));
   479→    }
   480→
   481→    // Activity count from the newest log pool (capped at 500 by the API).
   482→    get<{ items?: Array<{ userId?: string | null }> } | Array<{ userId?: string | null }>>(
   483→      "/api/activity-logs?limit=500"
   484→    )
   485→      .then((data) => {
   486→        const list = Array.isArray(data) ? data : (data.items ?? []);
   487→        setActivityCount(list.filter((l) => l.userId === u.id).length);
   488→      })
   489→      .catch(() => setActivityCount(null));
   490→  };
   491→
   492→  /* --------------------------------- actions UI ------------------------------ */
   493→
   494→  const inlineRoleSelect = (u: UserRow) => (
   495→    <Select
   496→      value={u.role}
   497→      onValueChange={(v) => {
   498→        if (v === u.role) return;
   499→        setRoleChange({ target: u, role: v });
   500→      }}
   501→    >
   502→      <SelectTrigger
   503→        className="h-8 w-36 text-xs"
   504→        aria-label={`Change role for ${u.name}`}
   505→        onClick={(e) => e.stopPropagation()}
   506→      >
   507→        <SelectValue />
   508→      </SelectTrigger>
   509→      <SelectContent onClick={(e) => e.stopPropagation()}>
   510→        {ROLES.map((r) => (
   511→          <SelectItem key={r} value={r}>
   512→            {titleCase(r)}
   513→          </SelectItem>
   514→        ))}
   515→      </SelectContent>
   516→    </Select>
   517→  );
   518→
   519→  const rowActions = (u: UserRow, withLabels = false) => {
   520→    const self = isSelf(u);
   521→    const lastAdmin = isLastAdmin(u);
   522→    return (
   523→      <div
   524→        className={cn(
   525→          "items-center",
   526→          withLabels ? "flex flex-wrap gap-2" : "flex justify-end gap-1"
   527→        )}
   528→      >
   529→        <Button
   530→          size={withLabels ? "sm" : "icon"}
   531→          variant="ghost"
   532→          className={withLabels ? "" : "size-11 md:size-9"}
   533→          aria-label={`View user ${u.name}`}
   534→          onClick={(e) => {
   535→            e.stopPropagation();
   536→            openView(u);
   537→          }}
   538→        >
   539→          <Eye className="size-4" />
   540→          {withLabels && <span>View</span>}
   541→        </Button>
   542→        <Button
   543→          size={withLabels ? "sm" : "icon"}
   544→          variant="ghost"
   545→          className={withLabels ? "" : "size-11 md:size-9"}
   546→          aria-label={`Edit user ${u.name}`}
   547→          onClick={(e) => {
   548→            e.stopPropagation();
   549→            openEdit(u);
   550→          }}
   551→        >
   552→          <Pencil className="size-4" />
   553→          {withLabels && <span>Edit</span>}
   554→        </Button>
   555→        <Button
   556→          size={withLabels ? "sm" : "icon"}
   557→          variant="ghost"
   558→          className={cn(
   559→            "text-red-600 hover:bg-red-500/10 hover:text-red-700",
   560→            !withLabels && "size-11 md:size-9"
   561→          )}
   562→          aria-label={
   563→            self
   564→              ? "You cannot delete your own account"
   565→              : lastAdmin
   566→                ? "You cannot delete the only administrator"
   567→                : `Delete user ${u.name}`
   568→          }
   569→          disabled={self || lastAdmin}
   570→          onClick={(e) => {
   571→            e.stopPropagation();
   572→            setDeleteTarget(u);
   573→          }}
   574→        >
   575→          <Trash2 className="size-4" />
   576→          {withLabels && <span>Delete</span>}
   577→        </Button>
   578→      </div>
   579→    );
   580→  };

   583→
   584→  const csvColumns: CsvColumn<UserRow>[] = useMemo(
   585→    () => [
   586→      { header: "Name", value: (u) => u.name },
   587→      { header: "Email", value: (u) => u.email },
   588→      { header: "Role", value: (u) => titleCase(u.role) },
   589→      { header: "Status", value: (u) => titleCase(u.status) },
   590→      { header: "Active", value: (u) => (u.isActive ? "Yes" : "No") },
   591→      { header: "Phone", value: (u) => u.phone ?? "" },
   592→      { header: "City", value: (u) => u.city ?? "" },
   593→      { header: "PAN", value: (u) => u.pan ?? "" },
   594→      { header: "Joined", value: (u) => formatDateTime(u.createdAt) },
   595→    ],
   596→    []
   597→  );
   598→
   599→  /* ------------------------------- table columns ----------------------------- */
   600→
   601→  const columns: DataTableColumn<UserRow>[] = [
   602→    { key: "user", header: "User" },
   603→    { key: "role", header: "Role" },
   604→    { key: "status", header: "Status" },
   605→    { key: "city", header: "City", hideBelow: "md" },
   606→    { key: "phone", header: "Phone", mono: true, hideBelow: "lg" },
