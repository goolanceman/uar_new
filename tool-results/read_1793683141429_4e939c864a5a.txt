   156→   156→function slugify(title: string): string {
   157→   157→  return title
   158→   158→    .toLowerCase()
   159→   159→    .trim()
   160→   160→    .replace(/[^a-z0-9\s-]/g, "")
   161→   161→    .replace(/[\s_]+/g, "-")
   162→   162→    .replace(/-+/g, "-")
   163→   163→    .replace(/^-|-$/g, "");
   164→   164→}
   165→   165→
   166→   166→function isoToDateInput(value?: string | null): string {
   167→   167→  if (!value) return "";
   168→   168→  const d = new Date(value);
   169→   169→  if (Number.isNaN(d.getTime())) return "";
   170→   170→  const y = d.getFullYear();
   171→   171→  const m = String(d.getMonth() + 1).padStart(2, "0");
   172→   172→  const day = String(d.getDate()).padStart(2, "0");
   173→   173→  return `${y}-${m}-${day}`;
   174→   174→}
   175→   175→
   176→   176→function dateInputToIso(value: string): string | null {
   177→   177→  if (!value) return null;
   178→   178→  const d = new Date(`${value}T12:00:00`);
   179→   179→  return Number.isNaN(d.getTime()) ? null : d.toISOString();
   180→   180→}

   189→   189→        <circle cx="50" cy="50" r={r} fill="none" stroke="currentColor" strokeOpacity="0.15" strokeWidth="8" />
   190→   190→        <circle
   191→   191→          cx="50"
   192→   192→          cy="50"
   193→   193→          r={r}
   194→   194→          fill="none"
   195→   195→          stroke="currentColor"
   196→   196→          strokeWidth="8"
   197→   197→          strokeLinecap="round"
   198→   198→          strokeDasharray={c}
   199→   199→          strokeDashoffset={c - (Math.min(100, pct) / 100) * c}
   200→   200→        />
   201→   201→      </svg>
