   199→function ageText(months: number): string {
   200→  const m = Math.max(0, Math.trunc(Number(months ?? 0)));
   201→  if (m < 12) return `${m} months`;
   202→  const y = Math.floor(m / 12);
   203→  const rest = m % 12;
   204→  return rest ? `${y}y ${rest}m` : `${y} year${y === 1 ? "" : "s"}`;
   205→}
   206→
   207→function validateAnimalForm(f: AnimalForm): Record<string, string | null> {
   208→  const age = f.ageMonths.trim() === "" ? NaN : Number(f.ageMonths);
   209→  const food = f.monthlyFood.trim() === "" ? 0 : Number(f.monthlyFood);
   210→  return {
   211→    name: required("Name")(f.name),
   212→    ageMonths:
   213→      Number.isFinite(age) && age >= 0 && age <= 600
   214→        ? null
   215→        : "Age must be between 0 and 600 months",
   216→    story: required("Story")(f.story),
   217→    location: required("Location")(f.location),
   218→    monthlyFood:
   219→      Number.isFinite(food) && food >= 0 && food <= 1_000_000
   220→        ? null
   221→        : "Enter a valid monthly food amount (₹0 or more)",
   222→  };
   223→}

   236→    return (

   251→}
