   135→   135→  const [citySearch, setCitySearch] = useState("");

   168→   168→      if (conditionFilter !== "ALL" && c.condition !== conditionFilter) return false;
   169→   169→      if (q && !`${c.city ?? ""}`.toLowerCase().includes(q) && !`${c.patientName}`.toLowerCase().includes(q))

   177→   177→    const adopted = causes.filter((c) => c.status === "ADOPTED").length;

   182→   182→    return { seeking, adopted, monthly, total: causes.length };

   193→   193→  const openEdit = (c: Cause) => {

   196→   196→      patientName: c.patientName ?? "",
   197→   197→      age: String(c.age ?? ""),
   198→   198→      city: c.city ?? "",
   199→   199→      condition: c.condition ?? "BP",
   200→   200→      monthlyCost: String(c.monthlyCost ?? ""),
   201→   201→      durationMonths: String(c.durationMonths ?? 12),
   202→   202→      story: c.story ?? "",

   205→   205→  };

   209→   209→    if (!form.city.trim()) return "City is required";

   213→   213→    const cost = Number(form.monthlyCost);

   224→   224→      return;

   229→   229→        patientName: form.patientName.trim(),
   230→   230→        age: Number(form.age),
   231→   231→        city: form.city.trim(),
   232→   232→        condition: form.condition,
   233→   233→        monthlyCost: Number(form.monthlyCost),
   234→   234→        durationMonths: Number(form.durationMonths),
   235→   235→        story: form.story.trim(),

   243→   243→      }
   244→   244→      setDialogOpen(false);
   245→   245→      setSheet(null);
   246→   246→      load();
   247→   247→    } catch (e) {
   248→   248→      toast.error(e instanceof Error ? e.message : "Save failed");
   249→   249→    } finally {
   250→   250→      setSaving(false);
   251→   251→    }

   262→   262→      toast.error(e instanceof Error ? e.message : "Update failed");
   263→   263→    }

   292→   292→        title: "Medicine Sponsorship Certificate",
   293→   293→        description: `Monthly medicine support for ${c.patientName} (${c.code})`,
   294→   294→        meta: JSON.stringify({ cause: c.code, patient: c.patientName, condition: c.condition }),
   295→   295→      });
   296→   296→      const code = res?.certificate?.code;
   297→   297→      if (code) {
   298→   298→        toast.success(`Certificate ${code} issued to ${c.adoptedByName}`, {
   299→   299→          action: {
   300→   300→            label: "Open",
   301→   301→            onClick: () => window.open(`/api/certificates/${code}/html`, "_blank"),
   302→   302→          },
   303→   303→        });
   304→   304→      } else {
   305→   305→        toast.success("Sponsorship certificate issued");
   306→   306→      }
   307→   307→    } catch (e) {
   308→   308→      toast.error(e instanceof Error ? e.message : "Certificate failed");
   309→   309→    } finally {
   310→   310→      setIssuing(null);
   311→   311→    }

   313→   313→

   323→   323→        icon={<HeartPulse className="size-5" aria-hidden="true" />}
   324→   324→        actions={
   325→   325→          <>
   326→   326→            <Button variant="outline" onClick={load} aria-label="Refresh causes">
   327→   327→              <RefreshCw className="size-4" aria-hidden="true" /> Refresh
   328→   328→            </Button>
   329→   329→            <Button onClick={openCreate}>
   330→   330→              <Plus className="size-4" aria-hidden="true" /> New cause
   331→   331→            </Button>
   332→   332→          </>
   333→   333→        }
   334→   334→      />
   335→   335→
   336→   336→      {/* stats */}
   337→   337→      <div className="grid gap-4 sm:grid-cols-2 xl:grid-cols-4">
   338→   338→        <StatCard label="Total causes" value={stats.total} accent="teal" icon={<HeartPulse className="size-5" aria-hidden="true" />} loading={loading} />
   339→   339→        <StatCard label="Seeking / partial" value={stats.seeking} accent="amber" icon={<HandCoins className="size-5" aria-hidden="true" />} loading={loading} />
   340→   340→        <StatCard label="Adopted" value={stats.adopted} accent="emerald" icon={<UserCheck className="size-5" aria-hidden="true" />} loading={loading} />
   341→   341→        <StatCard label="Monthly pledged" value={formatINR(stats.monthly)} accent="pink" icon={<HandCoins className="size-5" aria-hidden="true" />} loading={loading} hint="Sum of active pledges" />
   342→   342→      </div>
   343→   343→
   344→   344→      {/* filters */}
   345→   345→      <div className="flex flex-col gap-2 sm:flex-row sm:items-center">
   346→   346→        <Select value={statusFilter} onValueChange={setStatusFilter}>
   347→   347→          <SelectTrigger className="w-full sm:w-44" aria-label="Filter by status">
   348→   348→            <SelectValue />
   349→   349→          </SelectTrigger>
   350→   350→          <SelectContent>
   351→   351→            <SelectItem value="ALL">All statuses</SelectItem>
   352→   352→            <SelectItem value="SEEKING">Seeking</SelectItem>
   353→   353→            <SelectItem value="PARTIAL">Partial</SelectItem>
   354→   354→            <SelectItem value="ADOPTED">Adopted</SelectItem>

   383→   383→      {/* cards */}

   420→   420→              >
