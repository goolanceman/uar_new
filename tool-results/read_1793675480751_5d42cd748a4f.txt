    40→    40→function pickNum(o: Rec | undefined, keys: string[], fb = 0): number {
    41→    41→  if (!o) return fb;
    42→    42→  for (const k of keys) {
    43→    43→    const v = o[k];
    44→    44→    if (typeof v === "number" && Number.isFinite(v)) return v;
    45→    45→    if (typeof v === "string" && v !== "" && Number.isFinite(Number(v))) return Number(v);
    46→    46→  }
    47→    47→  return fb;
    48→    48→}
    49→    49→
    50→    50→function pickStr(o: Rec | undefined, keys: string[], fb = ""): string {
    51→    51→  if (!o) return fb;
    52→    52→  for (const k of keys) {
    53→    53→    const v = o[k];
    54→    54→    if (typeof v === "string" && v) return v;
    55→    55→  }
    56→    56→  return fb;
    57→    57→}
    58→    58→
    59→    59→function asArray(v: unknown): Rec[] {
    60→    60→  if (Array.isArray(v)) return v as Rec[];
    61→    61→  if (v && typeof v === "object") {
    62→    62→    const o = v as Rec;
    63→    63→    if (Array.isArray(o.items)) return o.items as Rec[];
    64→    64→    if (Array.isArray(o.data)) return o.data as Rec[];
    65→    65→  }
    66→    66→  return [];
    67→    67→}

    69→    69→function downloadCsv(filename: string, rows: (string | number)[][]) {
    70→    70→  const csv = rows
    71→    71→    .map((r) => r.map((c) => `"${String(c ?? "").replace(/"/g, '""')}"`).join(","))
    72→    72→    .join("\n");
    73→    73→  const blob = new Blob([`\uFEFF${csv}`], { type: "text/csv;charset=utf-8;" });
    74→    74→  const url = URL.createObjectURL(blob);
    75→    75→  const a = document.createElement("a");
    76→    76→  a.href = url;
    77→    77→  a.download = filename;
    78→    78→  a.click();
    79→    79→  URL.revokeObjectURL(url);
    80→    80→}

   232→   232→  const [to, setTo] = useState("");
   233→   233→  const [data, setData] = useState<Rec | null>(null);
   234→   234→  const [loading, setLoading] = useState(true);
   235→   235→
   236→   236→  const load = useCallback(async () => {
   237→   237→    setLoading(true);
   238→   238→    try {
   239→   239→      const params = new URLSearchParams({ type });
   240→   240→      if (from) params.set("from", from);
   241→   241→      if (to) params.set("to", to);

   243→   243→      setData(res ?? {});

   245→   245→      toast.error(e instanceof Error ? e.message : "Failed to load report");
   246→   246→      setData({});
   247→   247→    } finally {
   248→   248→      setLoading(false);
   249→   249→    }
   250→   250→  }, [type, from, to]);

   252→   252→  useEffect(() => {
   253→   253→    load();

   275→   275→        ["Receipt", "Donor", "Campaign", "Amount", "Type", "Status", "Date"],
