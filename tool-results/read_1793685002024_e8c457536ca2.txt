   151→/** Impact hint for the current amount + campaign (from campaign-data tiers). */
   152→function impactHint(amount: number, slug: string | null | undefined): string {
   153→  const tiers: DonationTier[] = slug
   154→    ? (DONATION_TIERS[slug] ?? GENERIC_IMPACT_HINTS)
   155→    : GENERIC_IMPACT_HINTS;
   156→  const sorted = [...tiers].sort((a, b) => b.amount - a.amount);
   157→  const hit = sorted.find((t) => amount >= t.amount) ?? sorted[sorted.length - 1];
   158→  return hit?.impact ?? "every rupee is QR-traced to a doorstep";
   159→}
