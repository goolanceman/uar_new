   110→   110→function randInt(min: number, max: number): number {

   114→   114→function pickCongestion(): Congestion {
   115→   115→  const r = Math.random();
   116→   116→  if (r < 0.25) return "LOW";
   117→   117→  if (r < 0.55) return "MODERATE";
   118→   118→  if (r < 0.85) return "HEAVY";
   119→   119→  return "GRIDLOCK";
   120→   120→}
