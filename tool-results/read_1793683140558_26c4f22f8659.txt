   143→   143→
   144→   144→function parseItems(json: string | null | undefined): TplItem[] {
   145→   145→  if (!json) return [];
   146→   146→  try {
   147→   147→    const parsed = JSON.parse(json) as { name?: string; qty?: number | string; unit?: string }[];
   148→   148→    if (!Array.isArray(parsed)) return [];
   149→   149→    return parsed.map((i) => ({
   150→   150→      name: String(i?.name ?? ""),
   151→   151→      qty: String(i?.qty ?? ""),
   152→   152→      unit: String(i?.unit ?? "pcs"),
   153→   153→    }));
   154→   154→  } catch {
   155→   155→    return [];
   156→   156→  }
   157→   157→}

   160→   160→
   161→   161→export default function KitsView() {
   162→   162→  const [templates, setTemplates] = useState<KitTemplate[]>([]);
   163→   163→  const [batches, setBatches] = useState<KitBatch[]>([]);
   164→   164→  const [loading, setLoading] = useState(true);
   165→   165→
   166→   166→  /* template dialog */

   170→   170→  const [savingTpl, setSavingTpl] = useState(false);

   172→   172→

   175→   175→  const [wizardStep, setWizardStep] = useState(1);
   176→   176→  const [wizardTemplateId, setWizardTemplateId] = useState("");
   177→   177→  const [wizardName, setWizardName] = useState("");

   179→   179→  const [wizardInstructions, setWizardInstructions] = useState("");
   180→   180→  const [savingBatch, setSavingBatch] = useState(false);
   181→   181→  const [deleteBatch, setDeleteBatch] = useState<KitBatch | null>(null);
   182→   182→
   183→   183→  /* batch detail sheet */
   184→   184→  const [sheet, setSheet] = useState<KitBatch | null>(null);
   185→   185→  const [units, setUnits] = useState<KitUnit[]>([]);
   186→   186→  const [unitsLoading, setUnitsLoading] = useState(false);
   187→   187→  const [unitStatus, setUnitStatus] = useState("ALL");

   189→   189→
   190→   190→  /* QR label dialog */

   193→   193→  const [qrLoading, setQrLoading] = useState(false);
   194→   194→
   195→   195→  const unitOptions = useMasters("UNIT");
   196→   196→
   197→   197→  const load = useCallback(async () => {

   206→   206→    }

   208→   208→
   209→   209→  useEffect(() => {
   210→   210→    load();
   211→   211→  }, [load]);

   217→   217→  }, [batches, templates]);

   219→   219→  /* ---------------------------- template actions ---------------------------- */
   220→   220→

   222→   222→    setEditingTpl(null);
   223→   223→    setTplForm(EMPTY_TPL_FORM);

   229→   229→    setTplForm({
   230→   230→      name: t.name ?? "",
   231→   231→      description: t.description ?? "",
   232→   232→      costPerUnit: String(t.costPerUnit ?? ""),
   233→   233→      items: parseItems(t.items).length ? parseItems(t.items) : [{ name: "", qty: "1", unit: "kg" }],

   235→   235→    setTplDialog(true);
   236→   236→  };
   237→   237→
   238→   238→  const computedCost = useMemo(() => {
   239→   239→    const qty = tplForm.items.reduce((s, i) => s + (Number(i.qty) || 0), 0);
   240→   240→    return { qty, perUnit: Number(tplForm.costPerUnit) || 0 };
   241→   241→  }, [tplForm]);
   242→   242→
   243→   243→  const saveTpl = async () => {
   244→   244→    if (!tplForm.name.trim()) return toast.error("Template name is required");
   245→   245→    const items = tplForm.items.filter((i) => i.name.trim());
   246→   246→    if (items.length === 0) return toast.error("Add at least one item with a name");
   247→   247→    setSavingTpl(true);
   248→   248→    try {
   249→   249→      const body = {
   250→   250→        name: tplForm.name.trim(),
   251→   251→        description: tplForm.description.trim() || undefined,
   252→   252→        items: items.map((i) => ({ name: i.name.trim(), qty: Number(i.qty) || 1, unit: i.unit })),
   253→   253→        costPerUnit: Number(tplForm.costPerUnit) || 0,
   254→   254→      };
   255→   255→      if (editingTpl) {
   256→   256→        await patch(`/api/kit-templates/${editingTpl.id}`, body);
   257→   257→        toast.success(`Template "${body.name}" updated`);
   258→   258→      } else {
   259→   259→        await post("/api/kit-templates", body);
   260→   260→        toast.success(`Template "${body.name}" created`);
   261→   261→      }
   262→   262→      setTplDialog(false);
   263→   263→      load();
   264→   264→    } catch (e) {
   265→   265→      toast.error(e instanceof Error ? e.message : "Save failed");
   266→   266→    } finally {
   267→   267→      setSavingTpl(false);
   268→   268→    }
   269→   269→  };
   270→   270→
   271→   271→  const removeTpl = async () => {
   272→   272→    if (!deleteTpl) return;
   273→   273→    try {
   274→   274→      await del(`/api/kit-templates/${deleteTpl.id}`);
   275→   275→      toast.success(`Template "${deleteTpl.name}" deleted`);
   276→   276→      setDeleteTpl(null);
   277→   277→      load();
   278→   278→    } catch (e) {
   279→   279→      toast.error(e instanceof Error ? e.message : "Delete failed");
   280→   280→    }
   281→   281→  };
   282→   282→
   283→   283→  /* ------------------------------ batch wizard ------------------------------ */
   284→   284→
   285→   285→  const openWizard = () => {
   286→   286→    setWizardStep(1);
   287→   287→    setWizardTemplateId("");
   288→   288→    setWizardName("");
   289→   289→    setWizardUnits("25");
   290→   290→    setWizardInstructions("");
   291→   291→    setWizardOpen(true);
   292→   292→  };
   293→   293→
   294→   294→  const createBatch = async () => {
   295→   295→    if (!wizardTemplateId) return toast.error("Pick a template first");
   296→   296→    if (!wizardName.trim()) return toast.error("Give the batch a name");
   297→   297→    const units = Math.trunc(Number(wizardUnits));
   298→   298→    if (!Number.isFinite(units) || units < 1 || units > 500) {
   299→   299→      toast.error("Total units must be between 1 and 500");
   300→   300→      return;
   301→   301→    }
   302→   302→    setSavingBatch(true);
   303→   303→    try {
   304→   304→      await post("/api/kit-batches", {
   305→   305→        templateId: wizardTemplateId,
   306→   306→        name: wizardName.trim(),
   307→   307→        totalUnits: units,
   308→   308→        instructions: wizardInstructions.trim() || undefined,
   309→   309→      });
   310→   310→      toast.success(`Batch "${wizardName.trim()}" created with ${units} QR-coded units`);
   311→   311→      setWizardOpen(false);
   312→   312→      load();
   313→   313→    } catch (e) {
   314→   314→      toast.error(e instanceof Error ? e.message : "Batch creation failed");
   315→   315→    } finally {
   316→   316→      setSavingBatch(false);
   317→   317→    }
   318→   318→  };
   319→   319→
   320→   320→  const removeBatch = async () => {
   321→   321→    if (!deleteBatch) return;
   322→   322→    try {
   323→   323→      await del(`/api/kit-batches/${deleteBatch.id}`);
   324→   324→      toast.success(`Batch "${deleteBatch.name}" deleted`);
   325→   325→      setDeleteBatch(null);
   326→   326→      setSheet(null);
   327→   327→      load();
   328→   328→    } catch (e) {
   329→   329→      toast.error(e instanceof Error ? e.message : "Delete failed");
   330→   330→    }
   331→   331→  };
   332→   332→
   333→   333→  /* ------------------------------ unit sheet -------------------------------- */
   334→   334→
   335→   335→  const openSheet = (b: KitBatch) => {
   336→   336→    setSheet(b);
   337→   337→    setUnitStatus("ALL");
   338→   338→    setUnitSearch("");
   339→   339→    setUnitsLoading(true);
   340→   340→    get<KitUnit[]>(`/api/kit-units?batchId=${b.id}&limit=500`)
   341→   341→      .then((rows) => setUnits(Array.isArray(rows) ? rows : []))
   342→   342→      .catch((e) => {
   343→   343→        setUnits([]);
   344→   344→        toast.error(e instanceof Error ? e.message : "Failed to load units");
   345→   345→      })
   346→   346→      .finally(() => setUnitsLoading(false));
   347→   347→  };
   348→   348→
   349→   349→  const filteredUnits = useMemo(() => {
   350→   350→    const q = unitSearch.trim().toUpperCase();
   351→   351→    return units.filter((u) => {
   352→   352→      if (unitStatus !== "ALL" && u.status !== unitStatus) return false;
   353→   353→      if (q && !u.code.includes(q)) return false;
   354→   354→      return true;
   355→   355→    });
   356→   356→  }, [units, unitStatus, unitSearch]);
   357→   357→
   358→   358→  const readyUnits = useMemo(() => units.filter((u) => u.status === "READY"), [units]);
   359→   359→
   360→   360→  /* ------------------------------ QR printing ------------------------------- */
   361→   361→
   362→   362→  useEffect(() => {
   363→   363→    if (!qrOpen) return;
   364→   364→    let alive = true;
   365→   365→    const run = async () => {

   367→   367→      const picked = readyUnits.slice(0, QR_SHEET_LIMIT);

   371→   371→          const dataUrl = await QRCode.toDataURL(u.code, { margin: 1, width: 140 });

   373→   373→        } catch {
   374→   374→          // skip a unit whose QR fails to render
   375→   375→        }
   376→   376→      }
   377→   377→      if (alive) {
   378→   378→        setQrLabels(out);
   379→   379→        setQrLoading(false);
   380→   380→      }
   381→   381→    };
   382→   382→    run();
   383→   383→    return () => {
   384→   384→      alive = false;
   385→   385→    };
   386→   386→  }, [qrOpen, readyUnits]);
   387→   387→
   388→   388→  /* --------------------------------- render -------------------------------- */
   389→   389→
   390→   390→  return (
   391→   391→    <div className="space-y-6">
   392→   392→      <PageHeader
   393→   393→        title="Ration Kits"
   394→   394→        description="Kit templates, QR-coded batches and distribution tracking."
   395→   395→        icon={<ClipboardList className="size-5" aria-hidden="true" />}
   396→   396→        actions={
   397→   397→          <>
   398→   398→            <Button variant="outline" onClick={load} aria-label="Refresh kits">
   399→   399→              <RefreshCw className="size-4" aria-hidden="true" /> Refresh
   400→   400→            </Button>
   401→   401→            <Button onClick={openWizard}>
   402→   402→              <Plus className="size-4" aria-hidden="true" /> New batch
   403→   403→            </Button>
   404→   404→          </>
   405→   405→        }
   406→   406→      />
   407→   407→
   408→   408→      <div className="grid gap-4 sm:grid-cols-2 xl:grid-cols-4">
   409→   409→        <StatCard label="Templates" value={totals.templates} accent="emerald" icon={<ClipboardList className="size-5" aria-hidden="true" />} loading={loading} />
   410→   410→        <StatCard label="Batches" value={totals.batches} accent="teal" icon={<Package className="size-5" aria-hidden="true" />} loading={loading} />
   411→   411→        <StatCard label="Units assembled" value={totals.totalUnits.toLocaleString("en-IN")} accent="amber" icon={<Package className="size-5" aria-hidden="true" />} loading={loading} />
   412→   412→        <StatCard label="Units scanned" value={totals.scanned.toLocaleString("en-IN")} accent="lime" icon={<QrCode className="size-5" aria-hidden="true" />} loading={loading} hint="At beneficiary hand-off" />
   413→   413→      </div>
   414→   414→
   415→   415→      <Tabs defaultValue="batches">
   416→   416→        <TabsList>
   417→   417→          <TabsTrigger value="batches">Batches ({batches.length})</TabsTrigger>
   418→   418→          <TabsTrigger value="templates">Templates ({templates.length})</TabsTrigger>
   419→   419→        </TabsList>
   420→   420→
   421→   421→        {/* ============================== BATCHES ============================== */}
   422→   422→        <TabsContent value="batches" className="mt-4">
   423→   423→          {loading ? (
   424→   424→            <div className="grid gap-4 sm:grid-cols-2 xl:grid-cols-3" aria-busy="true">
   425→   425→              {[0, 1, 2].map((i) => (
   426→   426→                <Skeleton key={i} className="h-56 rounded-xl" />
   427→   427→              ))}
   428→   428→            </div>
   429→   429→          ) : batches.length === 0 ? (
   430→   430→            <EmptyState
   431→   431→              icon={<Package className="size-8" aria-hidden="true" />}
   432→   432→              title="No batches yet"
   433→   433→              hint="Run the wizard to assemble a batch of QR-coded ration kits from a template."
   434→   434→            >
   435→   435→              <Button onClick={openWizard}>
   436→   436→                <Plus className="size-4" aria-hidden="true" /> New batch
   437→   437→              </Button>
   438→   438→            </EmptyState>
   439→   439→          ) : (
   440→   440→            <div className="grid gap-4 sm:grid-cols-2 xl:grid-cols-3">
   441→   441→              {batches.map((b) => {
   442→   442→                const pct =
   443→   443→                  b.totalUnits > 0 ? Math.min(100, Math.round(((b.scannedUnits ?? 0) / b.totalUnits) * 100)) : 0;
   444→   444→                return (
   445→   445→                  <button
   446→   446→                    key={b.id}
   447→   447→                    type="button"
   448→   448→                    onClick={() => openSheet(b)}
   449→   449→                    aria-label={`Open batch ${b.name}`}
   450→   450→                    className="flex flex-col gap-3 rounded-xl border bg-card p-4 text-left shadow-sm transition hover:border-emerald-500/40 hover:shadow-md focus-visible:outline-2 focus-visible:outline-primary"
   451→   451→                  >
   452→   452→                    <div className="flex items-start gap-2">
   453→   453→                      <div className="min-w-0 flex-1">
   454→   454→                        <p className="truncate font-semibold leading-tight">{b.name}</p>
   455→   455→                        <p className="truncate text-xs text-muted-foreground">{b.template?.name ?? "Template"}</p>
   456→   456→                      </div>
   457→   457→                      <Badge variant="outline" className={cn("shrink-0", BATCH_STATUS[b.status] ?? "")}>
   458→   458→                        {titleCase(b.status)}
   459→   459→                      </Badge>
