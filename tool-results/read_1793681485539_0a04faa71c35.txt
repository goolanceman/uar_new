   129→function statusTone(status: string): string {
   130→  switch (status) {
   131→    case "COMPLETED":
   132→      return "bg-emerald-500/10 text-emerald-700 dark:text-emerald-400";
   133→    case "PENDING":
   134→      return "bg-amber-500/10 text-amber-700 dark:text-amber-400";
   135→    case "FAILED":
   136→      return "bg-red-500/10 text-red-700 dark:text-red-400";
   137→    case "REFUNDED":
   138→      return "bg-muted text-muted-foreground";
   139→    default:
   140→      return "bg-muted text-muted-foreground";
   141→  }
   142→}

   157→function downloadCsv(filename: string, rows: (string | number)[][]) {
   158→  const csv = rows
   159→    .map((r) => r.map((c) => `"${String(c ?? "").replace(/"/g, '""')}"`).join(","))
   160→    .join("\n");
   161→  const blob = new Blob([`\uFEFF${csv}`], { type: "text/csv;charset=utf-8;" });
   162→  const url = URL.createObjectURL(blob);
   163→  const a = document.createElement("a");
   164→  a.href = url;
   165→  a.download = filename;
   166→  a.click();
   167→  URL.revokeObjectURL(url);
   168→}

   170→/** Normalises an Indian mobile number → 10 digits, or null when invalid. */
   171→function normalisePhone(raw: string): string | null {
   172→  const digits = raw.replace(/\D/g, "");
   173→  const trimmed = digits.length > 10 && digits.startsWith("91") ? digits.slice(2) : digits;
   174→  return trimmed.length === 10 ? trimmed : null;
   175→}
