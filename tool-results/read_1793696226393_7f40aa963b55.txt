   324→   423→  if (!json) return [];

   326→   425→    const parsed = JSON.parse(json);
   327→   426→    return Array.isArray(parsed) ? (parsed as ParsedItem[]) : [];

   333→   432→function itemsSummary(json: string | null | undefined): string {
   334→   433→  const items = parseItems(json);
   335→   434→  if (!items.length) return "—";
   336→   435→  return items
   337→   436→    .map((i) => {
   338→   437→      const name = String(i?.name ?? i ?? "").trim();
   339→   438→      const qty = i?.qty;
   340→   439→      const unit = i?.unit ? ` ${i.unit}` : "";
   341→   440→      return `${name} ×${qty ?? 1}${unit}`;
   342→   441→    })
   343→   442→    .join(", ");
   344→   443→}
