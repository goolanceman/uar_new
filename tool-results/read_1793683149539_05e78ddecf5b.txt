   117→function parseList(s?: string | null): string[] {
   118→  if (!s) return [];
   119→  try {
   120→    const parsed = JSON.parse(s);
   121→    if (Array.isArray(parsed)) return parsed.filter((x) => typeof x === "string" && x);
   122→  } catch {
   123→    // plain comma-separated string
   124→  }
   125→  return s.split(",").map((x) => x.trim()).filter(Boolean);
   126→}
   127→
   128→function parseTempCreds(s?: string | null): TempCreds | null {
   129→  if (!s) return null;
   130→  try {
   131→    const parsed = JSON.parse(s) as TempCreds;
   132→    if (parsed && (parsed.email || parsed.password)) return parsed;
   133→  } catch {
   134→    // not JSON
   135→  }
   136→  return null;
   137→}

   139→function appStatusTone(status: string): string {
   140→  switch (status) {
   141→    case "APPROVED":
   142→    case "ACTIVE":
   143→      return "bg-emerald-500/10 text-emerald-700 dark:text-emerald-400";
   144→    case "PENDING":
   145→      return "bg-amber-500/10 text-amber-700 dark:text-amber-400";
   146→    case "REJECTED":
   147→    case "SUSPENDED":
   148→      return "bg-red-500/10 text-red-700 dark:text-red-400";
   149→    default:
   150→      return "bg-muted text-muted-foreground";
   151→  }
   152→}
   153→
   154→function priorityTone(priority: string): string {
   155→  switch (priority) {
   156→    case "HIGH":
   157→      return "bg-red-500/10 text-red-700 dark:text-red-400";
   158→    case "MEDIUM":
   159→      return "bg-amber-500/10 text-amber-700 dark:text-amber-400";
   160→    default:
   161→      return "bg-muted text-muted-foreground";
   162→  }
   163→}

   186→  const load = useCallback(async () => {

   189→      const qs = filter === "ALL" ? "" : `status=${filter}`;
   190→      const data = await get<Application[] | { items?: Application[] }>(`/api/volunteer-applications?${qs}`);
   191→      const list = (Array.isArray(data) ? data : (data.items ?? [])).slice();
   192→      list.sort((a, b) => {
   193→        const rank = (s: string) => (s === "PENDING" ? 0 : 1);
   194→        if (rank(a.status) !== rank(b.status)) return rank(a.status) - rank(b.status);
   195→        return new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime();
   196→      });

   198→    } catch (e) {
   199→      toast.error(e instanceof Error ? e.message : "Failed to load applications");

   215→        { status: "APPROVED", reviewNote: reviewNote.trim() || undefined }

   540→      const [v, t] = await Promise.all([
   541→        get<VolunteerRow[] | { items?: VolunteerRow[] }>("/api/volunteers"),
   542→        get<TaskRow[] | { items?: TaskRow[] }>("/api/volunteer-tasks?limit=500").catch(() => [] as TaskRow[]),
   543→      ]);
   544→      setVols(Array.isArray(v) ? v : (v.items ?? []));
   545→      setTasks(Array.isArray(t) ? t : ((t as { items?: TaskRow[] })?.items ?? []));
   546→    } catch (e) {
   547→      toast.error(e instanceof Error ? e.message : "Failed to load volunteers");
   548→    } finally {
   549→      setLoading(false);
   550→    }
   551→  }, []);
   552→
   553→  useEffect(() => {
   554→    load();
   555→  }, [load]);
   556→
   557→  const filtered = useMemo(() => {
   558→    const needle = q.trim().toLowerCase();
   559→    if (!needle) return vols;
   560→    return vols.filter((v) => {
   561→      const hay = `${v.user?.name ?? ""} ${v.user?.email ?? ""} ${v.assignedArea ?? ""}`.toLowerCase();
   562→      return hay.includes(needle);
   563→    });
   564→  }, [vols, q]);
   565→
   566→  const openView = (v: VolunteerRow) => {
   567→    setViewTarget(v);
   568→    setViewTasks(tasks.filter((t) => t.volunteerId === v.id));
   569→    setAttendanceCount(null);
   570→    get<unknown>("/api/volunteer-attendances?volunteerId=" + encodeURIComponent(v.id))
   571→      .then((d) => {
   572→        if (Array.isArray(d)) setAttendanceCount(d.length);
   573→        else if (d && typeof d === "object" && Array.isArray((d as { items?: unknown[] }).items)) {
   574→          setAttendanceCount(((d as { items: unknown[] }).items).length);
   575→        } else setAttendanceCount(null);
   576→      })
   577→      .catch(() => setAttendanceCount(null));
   578→  };
   579→
   580→  const setVolunteerStatus = async (v: VolunteerRow, next: "ACTIVE" | "SUSPENDED" | "INACTIVE") => {
   581→    setBusyId(v.id);
   582→    try {
   583→      await patch(`/api/volunteers/${v.id}`, { status: next });
   584→      toast.success(`${v.user?.name ?? "Volunteer"} → ${titleCase(next)}`);
   585→    } catch {
   586→      // Volunteers API has no PATCH today — fall back to the linked user account
   587→      const map: Record<string, { status?: string; isActive: boolean }> = {
   588→        ACTIVE: { status: "ACTIVE", isActive: true },
   589→        SUSPENDED: { status: "SUSPENDED", isActive: false },
   590→        INACTIVE: { isActive: false },
   591→      };
   592→      try {
   593→        await patch(`/api/users/${v.userId}`, map[next]);
   594→        toast.success(`${v.user?.name ?? "Volunteer"} account → ${next === "INACTIVE" ? "Inactive" : titleCase(next)}`);
   595→      } catch (e2) {
   596→        toast.error(e2 instanceof Error ? e2.message : "Failed to update volunteer");
   597→        setBusyId(null);
   598→        return;
   599→      }
   600→    }
   601→    setBusyId(null);
   602→    setViewTarget((prev) => (prev && prev.id === v.id ? { ...prev, status: next } : prev));
   603→    await load();
   604→  };
   605→
   606→  const volName = (v: VolunteerRow) => v.user?.name ?? "Volunteer";
   607→
   608→  return (
   609→    <div className="space-y-4">
   610→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center">
   611→        <div className="relative flex-1">
   612→          <Search className="absolute left-3 top-1/2 size-4 -translate-y-1/2 text-muted-foreground" aria-hidden="true" />
   613→          <Input value={q} onChange={(e) => setQ(e.target.value)} placeholder="Search name, email or area…" className="pl-9" aria-label="Search volunteers" />
   614→        </div>
   615→        <span className="text-xs text-muted-foreground" aria-live="polite">
   616→          {loading ? "Loading…" : `${filtered.length} volunteer${filtered.length === 1 ? "" : "s"}`}
   617→        </span>
   618→      </div>
   619→
   620→      {/* Desktop table */}
   621→      <div className="hidden rounded-xl border bg-card md:block">
   622→        {loading ? (
   623→          <div className="space-y-2 p-4">
   624→            {Array.from({ length: 5 }).map((_, i) => (
   625→              <Skeleton key={i} className="h-11 w-full" />
   626→            ))}
   627→          </div>
   628→        ) : filtered.length === 0 ? (
   629→          <div className="p-6">
   630→            <EmptyState
   631→              icon={<Users className="size-6" aria-hidden="true" />}
   632→              title="No volunteers found"
   633→              hint="Approve applications in the first tab to onboard volunteers."
   634→            />
   635→          </div>
   636→        ) : (
   637→          <div className="overflow-x-auto">
   638→            <Table>
   639→              <TableHeader>
   640→                <TableRow>
   641→                  <TableHead>Volunteer</TableHead>
   642→                  <TableHead>Status</TableHead>
   643→                  <TableHead className="hidden lg:table-cell">Assigned area</TableHead>
   644→                  <TableHead className="hidden xl:table-cell">Skills</TableHead>
   645→                  <TableHead className="hidden sm:table-cell">Joined</TableHead>
   646→                  <TableHead className="text-right">Actions</TableHead>
   647→                </TableRow>
   648→              </TableHeader>
   649→              <TableBody>
   650→                {filtered.map((v) => (
   651→                  <TableRow key={v.id}>
   652→                    <TableCell>
   653→                      <div className="flex items-center gap-2.5">
   654→                        <span className="flex size-8 shrink-0 items-center justify-center rounded-full bg-primary/10 text-xs font-semibold text-primary" aria-hidden="true">
   655→                          {initials(volName(v))}
   656→                        </span>
   657→                        <div className="min-w-0">
   658→                          <p className="truncate font-medium">{volName(v)}</p>
   659→                          <p className="truncate text-xs text-muted-foreground">{v.user?.email ?? "—"}</p>
   660→                        </div>
   661→                      </div>
   662→                    </TableCell>
   663→                    <TableCell>
   664→                      <span className={cn("whitespace-nowrap rounded-full px-2 py-0.5 text-[11px] font-semibold", appStatusTone(v.status))}>
   665→                        {titleCase(v.status)}
   666→                      </span>
   667→                    </TableCell>
   668→                    <TableCell className="hidden lg:table-cell">{v.assignedArea || "—"}</TableCell>
   669→                    <TableCell className="hidden max-w-48 xl:table-cell">
   670→                      <div className="flex flex-wrap gap-1">
   671→                        {parseList(v.skills).slice(0, 3).map((s) => <Chip key={s}>{s}</Chip>)}
   672→                        {parseList(v.skills).length > 3 && <span className="text-[11px] text-muted-foreground">+{parseList(v.skills).length - 3}</span>}
   673→                      </div>
   674→                    </TableCell>
   675→                    <TableCell className="hidden whitespace-nowrap text-sm sm:table-cell">{formatDate(v.joinedAt)}</TableCell>
   676→                    <TableCell className="text-right">
   677→                      <div className="flex justify-end gap-1">
   678→                        <Button
   679→                          size="icon"
   680→                          variant="ghost"
   681→                          className="size-11 md:size-9"
   682→                          aria-label={`View volunteer ${volName(v)}`}
   683→                          onClick={() => openView(v)}
   684→                        >
   685→                          <Eye className="size-4" />
   686→                        </Button>
   687→                        {v.status !== "ACTIVE" && (
   688→                          <Button
   689→                            size="sm"
   690→                            variant="outline"
   691→                            disabled={busyId === v.id}
   692→                            aria-label={`Activate volunteer ${volName(v)}`}
   693→                            onClick={() => setVolunteerStatus(v, "ACTIVE")}
   694→                          >
   695→                            Activate
   696→                          </Button>
   697→                        )}
   698→                        {v.status === "ACTIVE" && (
   699→                          <Button
   700→                            size="sm"
   701→                            variant="outline"
   702→                            disabled={busyId === v.id}
   703→                            aria-label={`Suspend volunteer ${volName(v)}`}
   704→                            onClick={() => setVolunteerStatus(v, "SUSPENDED")}
   705→                          >
   706→                            Suspend
   707→                          </Button>
   708→                        )}
   709→                        {v.status !== "INACTIVE" && (
   710→                          <Button
   711→                            size="sm"
   712→                            variant="ghost"
   713→                            disabled={busyId === v.id}
   714→                            aria-label={`Deactivate volunteer ${volName(v)}`}
   715→                            onClick={() => setVolunteerStatus(v, "INACTIVE")}
   716→                          >
   717→                            Deactivate
   718→                          </Button>
   719→                        )}
   720→                      </div>
   721→                    </TableCell>
   722→                  </TableRow>
   723→                ))}
   724→              </TableBody>
   725→            </Table>
   726→          </div>
   727→        )}
   728→      </div>
   729→
   730→      {/* Mobile cards */}
   731→      <div className="space-y-3 md:hidden">
   732→        {loading ? (
   733→          Array.from({ length: 4 }).map((_, i) => <Skeleton key={i} className="h-28 rounded-xl" />)
   734→        ) : filtered.length === 0 ? (
   735→          <EmptyState
   736→            icon={<Users className="size-6" aria-hidden="true" />}
   737→            title="No volunteers found"
   738→            hint="Approve applications in the first tab to onboard volunteers."
   739→          />
   740→        ) : (
   741→          filtered.map((v) => (
   742→            <article key={v.id} className="space-y-2.5 rounded-xl border bg-card p-4">
   743→              <div className="flex items-start justify-between gap-2">
   744→                <div className="flex min-w-0 items-center gap-2.5">
   745→                  <span className="flex size-9 shrink-0 items-center justify-center rounded-full bg-primary/10 text-sm font-semibold text-primary" aria-hidden="true">
   746→                    {initials(volName(v))}
   747→                  </span>
   748→                  <div className="min-w-0">
   749→                    <p className="truncate text-sm font-semibold">{volName(v)}</p>
   750→                    <p className="truncate text-xs text-muted-foreground">{v.user?.email ?? "—"}</p>
   751→                  </div>
   752→                </div>
   753→                <span className={cn("shrink-0 rounded-full px-2 py-0.5 text-[11px] font-semibold", appStatusTone(v.status))}>
   754→                  {titleCase(v.status)}
   755→                </span>
   756→              </div>
   757→              {v.assignedArea && (
   758→                <p className="flex items-center gap-1.5 text-xs text-muted-foreground">
   759→                  <MapPin className="size-3.5" aria-hidden="true" /> {v.assignedArea}
   760→                </p>
   761→              )}
   762→              <div className="flex flex-wrap gap-2 border-t pt-2">
   763→                <Button size="sm" variant="outline" onClick={() => openView(v)} aria-label={`View volunteer ${volName(v)}`}>
   764→                  <Eye className="size-4" /> View
   765→                </Button>
   766→                {v.status === "ACTIVE" ? (
   767→                  <Button size="sm" variant="outline" disabled={busyId === v.id} onClick={() => setVolunteerStatus(v, "SUSPENDED")} aria-label={`Suspend volunteer ${volName(v)}`}>
   768→                    Suspend
   769→                  </Button>
   770→                ) : (
   771→                  <Button size="sm" variant="outline" disabled={busyId === v.id} onClick={() => setVolunteerStatus(v, "ACTIVE")} aria-label={`Activate volunteer ${volName(v)}`}>
   772→                    Activate
   773→                  </Button>
   774→                )}
   775→              </div>
   776→            </article>
   777→          ))
   778→        )}
   779→      </div>
   780→
   781→      {/* Volunteer detail sheet */}
   782→      <Sheet open={!!viewTarget} onOpenChange={(o) => !o && setViewTarget(null)}>
   783→        <SheetContent className="flex w-full flex-col gap-0 overflow-y-auto sm:max-w-lg">
   784→          <SheetHeader className="text-left">
   785→            <SheetTitle className="flex items-center gap-2">
   786→              <Users className="size-4 text-primary" aria-hidden="true" />
   787→              {viewTarget ? volName(viewTarget) : "Volunteer"}
   788→            </SheetTitle>
   789→            <SheetDescription>
   790→              {viewTarget ? `${viewTarget.user?.email ?? ""} · ${titleCase(viewTarget.status)}` : ""}
   791→            </SheetDescription>
   792→          </SheetHeader>
   793→          {viewTarget && (
   794→            <div className="flex-1 space-y-4 px-4 pb-6">
   795→              <SectionCard title="Profile" icon={<UserCheck className="size-4" aria-hidden="true" />}>
   796→                <div className="grid grid-cols-2 gap-x-4 gap-y-4">
   797→                  <DataField label="Status" value={<span className={cn("rounded-full px-2 py-0.5 text-[11px] font-semibold", appStatusTone(viewTarget.status))}>{titleCase(viewTarget.status)}</span>} />
   798→                  <DataField label="Volunteer ID" value={viewTarget.id} mono copy />
   799→                  <DataField label="Assigned area" value={viewTarget.assignedArea} />
   800→                  <DataField label="Emergency contact" value={viewTarget.emergencyContact} copy />
   801→                  <DataField label="Phone" value={viewTarget.user?.phone} />
   802→                  <DataField label="City" value={viewTarget.user?.city} />
   803→                  <DataField label="Joined" value={formatDate(viewTarget.joinedAt)} />
   804→                  <DataField label="Attendance records" value={attendanceCount === null ? "—" : String(attendanceCount)} />
   805→                </div>
   806→                <div className="mt-4 grid gap-3">
   807→                  <div>
   808→                    <p className="text-[11px] font-medium uppercase tracking-wide text-muted-foreground">Skills</p>
   809→                    <div className="mt-1.5 flex flex-wrap gap-1.5">
   810→                      {parseList(viewTarget.skills).length > 0
   811→                        ? parseList(viewTarget.skills).map((s) => <Chip key={s}>{s}</Chip>)
   812→                        : <span className="text-sm text-muted-foreground">—</span>}
   813→                    </div>
   814→                  </div>
   815→                  <div>
   816→                    <p className="text-[11px] font-medium uppercase tracking-wide text-muted-foreground">Availability</p>
   817→                    <div className="mt-1.5 flex flex-wrap gap-1.5">
   818→                      {parseList(viewTarget.availability).length > 0
   819→                        ? parseList(viewTarget.availability).map((s) => <Chip key={s}>{titleCase(s)}</Chip>)
   820→                        : <span className="text-sm text-muted-foreground">—</span>}
   821→                    </div>
   822→                  </div>
   823→                </div>
   824→              </SectionCard>
   825→
   826→              <SectionCard
   827→                title="Assigned tasks"
   828→                description="All tasks linked to this volunteer"
   829→                icon={<ClipboardList className="size-4" aria-hidden="true" />}
   830→              >
   831→                {viewTasks === null ? (
   832→                  <Skeleton className="h-20 w-full" />
   833→                ) : viewTasks.length === 0 ? (
   834→                  <EmptyState
   835→                    icon={<ClipboardList className="size-6" aria-hidden="true" />}
   836→                    title="No tasks yet"
   837→                    hint="Assign tasks from the admin tasks API — they will appear here."
   838→                  />
   839→                ) : (
   840→                  <ul className="max-h-60 space-y-2 overflow-y-auto scrollbar-thin" aria-label="Volunteer tasks">
   841→                    {viewTasks.map((t) => (
   842→                      <li key={t.id} className="rounded-lg border p-3">
   843→                        <div className="flex items-start justify-between gap-2">
   844→                          <p className="min-w-0 truncate text-sm font-medium">{t.title}</p>
   845→                          <span className={cn("shrink-0 rounded-full px-2 py-0.5 text-[11px] font-semibold", priorityTone(t.priority))}>
   846→                            {titleCase(t.priority)}
   847→                          </span>
   848→                        </div>
   849→                        <p className="mt-1 flex flex-wrap items-center gap-x-3 gap-y-0.5 text-xs text-muted-foreground">
   850→                          <span className="flex items-center gap-1"><CalendarClock className="size-3" aria-hidden="true" /> {formatDateTime(t.scheduledDate)}</span>
   851→                          {t.location && <span className="flex items-center gap-1"><MapPin className="size-3" aria-hidden="true" /> {t.location}</span>}
   852→                          <span>{titleCase(t.type)}</span>
   853→                          <span className="font-semibold">{titleCase(t.status)}</span>
   854→                        </p>
   855→                      </li>
   856→                    ))}
   857→                  </ul>
   858→                )}
   859→              </SectionCard>
   860→
   861→              <div className="flex flex-wrap gap-2">
   862→                {viewTarget.status !== "ACTIVE" && (
   863→                  <Button size="sm" disabled={busyId === viewTarget.id} onClick={() => setVolunteerStatus(viewTarget, "ACTIVE")} aria-label="Activate volunteer">
   864→                    <Check className="size-4" /> Activate
   865→                  </Button>
   866→                )}
   867→                {viewTarget.status === "ACTIVE" && (
   868→                  <Button size="sm" variant="outline" disabled={busyId === viewTarget.id} onClick={() => setVolunteerStatus(viewTarget, "SUSPENDED")} aria-label="Suspend volunteer">
   869→                    Suspend
   870→                  </Button>
   871→                )}
   872→                {viewTarget.status !== "INACTIVE" && (
   873→                  <Button size="sm" variant="ghost" disabled={busyId === viewTarget.id} onClick={() => setVolunteerStatus(viewTarget, "INACTIVE")} aria-label="Deactivate volunteer">
   874→                    Deactivate
   875→                  </Button>
   876→                )}
   877→              </div>
   878→            </div>
   879→          )}
   880→        </SheetContent>
   881→      </Sheet>
   882→    </div>
   883→  );
   884→}
