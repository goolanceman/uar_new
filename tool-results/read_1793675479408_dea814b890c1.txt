   500→/* Toolkit — interactive checklists with progress                      */
   501→/* ================================================================== */
   502→
   503→function ToolkitSection({
   504→  accent,
   505→  toolkit,
   506→}: {
   507→  accent: string;
   508→  toolkit: { title: string; items: string[] }[];
   509→}) {
   510→  const a = accentClasses(accent);
   511→  const [checked, setChecked] = useState<Set<string>>(new Set());
   512→
   513→  const total = toolkit.reduce((n, g) => n + g.items.length, 0);
   514→  const done = checked.size;
   515→
   516→  const toggle = (key: string) => {
   517→    setChecked((prev) => {
   518→      const next = new Set(prev);
   519→      if (next.has(key)) next.delete(key);
   520→      else next.add(key);
   521→      return next;
   522→    });
   523→  };
   524→
   525→  return (
   526→    <Section
   527→      title="Your toolkit"
   528→      subtitle="Tick items off as you secure your home, family or workplace. Progress is saved as you go."
   529→    >
   530→      <div className="mb-5 flex items-center gap-3">
   531→        <Progress
   532→          value={total ? (done / total) * 100 : 0}
   533→          className="h-2.5 max-w-xs"
   534→          aria-label={`${done} of ${total} checklist items done`}
   535→        />
   536→        <span className="text-sm font-semibold text-muted-foreground">
   537→          {done}/{total}
   538→        </span>
   539→      </div>
   540→      <div className="grid gap-4 md:grid-cols-2 lg:grid-cols-3">
   541→        {toolkit.map((g, gi) => {
   542→          const groupDone = g.items.filter((_, ii) => checked.has(`${gi}-${ii}`)).length;
   543→          return (
   544→            <Card key={gi} className="flex flex-col p-5">
   545→              <div className="flex items-center justify-between gap-2">
   546→                <h3 className="font-semibold leading-snug">{g.title}</h3>
   547→                <Badge variant="outline" className={cn("shrink-0", a.border200, a.text700)}>
   548→                  {groupDone}/{g.items.length}
   549→                </Badge>
   550→              </div>
   551→              <ul className="mt-4 space-y-3">
   552→                {g.items.map((item, ii) => {
   553→                  const key = `${gi}-${ii}`;
   554→                  return (
   555→                    <li key={key}>
   556→                      <label className="flex cursor-pointer items-start gap-2.5">
   557→                        <Checkbox
   558→                          checked={checked.has(key)}
   559→                          onCheckedChange={() => toggle(key)}
   560→                          className="mt-0.5"
   561→                          aria-label={item}
   562→                        />
   563→                        <span
   564→                          className={cn(
   565→                            "text-sm leading-relaxed",
   566→                            checked.has(key) && "text-muted-foreground line-through"
   567→                          )}
   568→                        >
   569→                          {item}
   570→                        </span>
   571→                      </label>
   572→                    </li>
   573→                  );
   574→                })}
   575→              </ul>
   576→            </Card>
   577→          );
   578→        })}
   579→      </div>
   580→    </Section>
   581→  );

   684→function AdoptDialog({
   685→  cause,
   686→  onClose,
   687→  onDone,
   688→}: {
   689→  cause: Cause | null;
   690→  onClose: () => void;
   691→  onDone: () => void;
   692→}) {
   693→  const [amount, setAmount] = useState("");
   694→  const [months, setMonths] = useState("6");
   695→  const [name, setName] = useState("");
   696→  const [busy, setBusy] = useState(false);
   697→
   698→  useEffect(() => {
   699→    if (cause) {
   700→      setAmount(String(Math.round(cause.monthlyCost)));
   701→      setMonths("6");
   702→      setName("");
   703→      setBusy(false);
   704→    }
   705→  }, [cause]);
   706→
   707→  const submit = async () => {
   708→    if (!cause) return;
   709→    const amt = Number(amount);
   710→    if (!amt || amt <= 0) {
   711→      toast.error("Enter a valid monthly amount");
   712→      return;
   713→    }
   714→    if (!name.trim()) {
   715→      toast.error("Please enter your name");
   716→      return;
   717→    }
   718→    setBusy(true);
   719→    try {
   720→      await post(`/api/causes/${cause.id}/pledges`, {
   721→        amountPerMonth: amt,
   722→        months: Number(months) || 6,
   723→        donorName: name.trim(),
   724→      });
   725→      toast.success(
   726→        `Adoption pledged for ${cause.patientName} — our team will call you within 48 hours.`
   727→      );
   728→      onClose();
   729→      onDone();
   730→    } catch (e) {
   731→      toast.error(e instanceof Error ? e.message : "Could not create pledge");
   732→    } finally {
   733→      setBusy(false);
   734→    }
   735→  };
   736→
   737→  return (
   738→    <Dialog open={!!cause} onOpenChange={(o) => !o && onClose()}>
   739→      <DialogContent className="sm:max-w-md">
   740→        {cause && (
   741→          <>
   742→            <DialogHeader>
   743→              <DialogTitle>
   744→                Adopt {cause.patientName}&rsquo;s medicines
   745→              </DialogTitle>
   746→              <DialogDescription>
   747→                {cause.code} · {titleCase(cause.condition)} · {cause.city} · needs{" "}
   748→                {formatINR(cause.monthlyCost)}/mo for {cause.durationMonths} months
   749→              </DialogDescription>
   750→            </DialogHeader>
   751→            <div className="space-y-4 py-2">
   752→              <div className="space-y-1.5">
   753→                <Label htmlFor="pledge-amount">Amount per month (₹)</Label>
   754→                <Input
   755→                  id="pledge-amount"
   756→                  type="number"
   757→                  min={1}
   758→                  value={amount}
   759→                  onChange={(e) => setAmount(e.target.value)}
   760→                />
   761→                <p className="text-xs text-muted-foreground">
   762→                  Full cost is {formatINR(cause.monthlyCost)} — partial adoptions are welcome.
   763→                </p>
   764→              </div>
   765→              <div className="space-y-1.5">
   766→                <Label>Commitment</Label>
   767→                <Select value={months} onValueChange={setMonths}>
   768→                  <SelectTrigger aria-label="Months">
   769→                    <SelectValue />
   770→                  </SelectTrigger>
   771→                  <SelectContent>
   772→                    <SelectItem value="3">3 months</SelectItem>
   773→                    <SelectItem value="6">6 months</SelectItem>
   774→                    <SelectItem value="12">12 months</SelectItem>
   775→                  </SelectContent>
   776→                </Select>
   777→              </div>
   778→              <div className="space-y-1.5">
   779→                <Label htmlFor="pledge-name">Your name</Label>
   780→                <Input
   781→                  id="pledge-name"
   782→                  value={name}
   783→                  onChange={(e) => setName(e.target.value)}
   784→                  placeholder="As you'd like to be known"
   785→                />
   786→              </div>
   787→            </div>
   788→            <DialogFooter>
   789→              <Button variant="outline" onClick={onClose} disabled={busy}>
   790→                Cancel
   791→              </Button>
   792→              <Button onClick={submit} disabled={busy} className="bg-teal-600 hover:bg-teal-700">
   793→                {busy ? "Pledging…" : "Confirm adoption"}
   794→              </Button>
   795→            </DialogFooter>
   796→          </>
   797→        )}
   798→      </DialogContent>
   799→    </Dialog>
   800→  );
   801→}
