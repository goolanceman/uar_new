   110→function randInt(min: number, max: number): number {
   111→  return min + Math.floor(Math.random() * (max - min + 1));

   113→
   114→function pickCongestion(): Congestion {
   115→  const r = Math.random();
   116→  if (r < 0.25) return "LOW";
   117→  if (r < 0.55) return "MODERATE";
   118→  if (r < 0.85) return "HEAVY";
   119→  return "GRIDLOCK";
   120→}

   201→    for (let i = 0; i < SEGMENTS.length; i++) {

   452→
   453→interface CorridorStats {
   454→  missions?: { total?: number; completed?: number; active?: number };
   455→  vehiclesAlerted?: number;
   456→  timeSavedMin?: number;
   457→  avgSavedSec?: number;
   458→  byMethod?: { method: string; count: number; savedSec: number }[];
   459→}
   460→
   461→function CorridorLiveStats() {
   462→  const [stats, setStats] = useState<CorridorStats | null>(null);
   463→  const [failed, setFailed] = useState(false);
   464→
   465→  useEffect(() => {
   466→    let alive = true;
   467→    get<CorridorStats>("/api/corridor-stats")
   468→      .then((d) => alive && setStats(d ?? null))
   469→      .catch(() => {
   470→        if (alive) {
   471→          setFailed(true);
   472→        }
   473→      });
   474→    return () => {
   475→      alive = false;
   476→    };
   477→  }, []);
   478→
   479→  if (failed) {
   480→    return (
   481→      <Card className="p-5">
   482→        <p className="flex items-start gap-2 text-sm text-muted-foreground">
   483→          <Radio className="mt-0.5 size-4 shrink-0 text-red-600 dark:text-red-400" aria-hidden />
   484→          Live ops-desk counters are visible to signed-in volunteers and admins. The pilot
   485→          benchmarks below and the simulation above tell the story meanwhile — every corridor
   486→          event is timestamped per segment and fully auditable.
   487→        </p>
   488→      </Card>
   489→    );
   490→  }
   491→
   492→  if (stats === null) {
   493→    return (
   494→      <div className="grid gap-3 sm:grid-cols-2 lg:grid-cols-4">
   495→        {Array.from({ length: 4 }).map((_, i) => (
   496→          <Skeleton key={i} className="h-24 rounded-xl" />
   497→        ))}
   498→      </div>
   499→    );
   500→  }
   501→
   502→  const kpis = [
   503→    { label: "corridor missions", value: stats.missions?.total ?? 0 },
   504→    { label: "completed", value: stats.missions?.completed ?? 0 },
   505→    { label: "minutes saved", value: stats.timeSavedMin ?? 0 },
   506→    { label: "vehicles alerted", value: stats.vehiclesAlerted ?? 0 },
   507→  ];
   508→
   509→  return (
   510→    <div>
   511→      <div className="grid grid-cols-2 gap-3 lg:grid-cols-4">
   512→        {kpis.map((k) => (
   513→          <Card key={k.label} className="p-4 text-center sm:p-5">
   514→            <p className="text-2xl font-extrabold text-red-600 dark:text-red-400">
   515→              {k.value.toLocaleString("en-IN")}
   516→            </p>
   517→            <p className="mt-1 text-xs text-muted-foreground">{k.label}</p>
   518→          </Card>
   519→        ))}
   520→      </div>
   521→      {stats.byMethod && stats.byMethod.length > 0 && (
   522→        <div className="mt-3 flex flex-wrap items-center gap-2">
   523→          <span className="text-[11px] font-bold uppercase tracking-wide text-muted-foreground">
   524→            Alerts by layer
   525→          </span>
   526→          {stats.byMethod.map((m) => (
   527→            <Badge key={m.method} variant="outline" className="gap-1 text-[10px]">
   528→              {m.method} · {m.count}
   529→            </Badge>
   530→          ))}
   531→        </div>
   532→      )}
   533→    </div>
   534→  );
   535→}
