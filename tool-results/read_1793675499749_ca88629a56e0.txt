    45→
    46→interface Segment {
    47→  name: string;
    48→  km: number;

    51→const SEGMENTS: Segment[] = [
    52→  { name: "Hebbal Flyover", km: 3.1 },
    53→  { name: "ORR · Nagawara", km: 2.6 },

    66→}

   100→  },
   101→};
   102→
   103→const TOTAL_KM = SEGMENTS.reduce((n, s) => n + s.km, 0);
   104→
   105→const sleep = (ms: number) => new Promise<void>((r) => setTimeout(r, ms));
   106→
   107→function randInt(min: number, max: number): number {
   108→  return min + Math.floor(Math.random() * (max - min + 1));
   109→}
   110→
   111→function pickCongestion(): Congestion {
   112→  const r = Math.random();
   113→  if (r < 0.25) return "LOW";
   114→  if (r < 0.55) return "MODERATE";
   115→  if (r < 0.85) return "HEAVY";
   116→  return "GRIDLOCK";
   117→}
   118→
   119→interface LogEntry {
   120→  id: number;
   121→  segment: string;
   122→  method: string;
   123→  vehicles: number;
   124→  saved: number;
   125→  congestion: Congestion;
   126→}
   127→
   128→export function LiveCorridorDemo() {
   129→  const [phase, setPhase] = useState<"idle" | "running" | "done">("idle");
   130→  const [cong, setCong] = useState<Congestion[]>(SEGMENTS.map(() => "LOW"));
   131→  const [revealed, setRevealed] = useState<boolean[]>(SEGMENTS.map(() => false));
   132→  const [cars, setCars] = useState<number[]>(SEGMENTS.map(() => 4));
   133→  const [alerted, setAlerted] = useState<boolean[]>(SEGMENTS.map(() => false));
   134→  const [pulse, setPulse] = useState(-1);
   135→  const [ambX, setAmbX] = useState(1);
   136→  const [log, setLog] = useState<LogEntry[]>([]);
   137→  const [savedSec, setSavedSec] = useState(0);
   138→  const [alertedTotal, setAlertedTotal] = useState(0);
   139→  const [method, setMethod] = useState<string>("—");
   140→  const cancelled = useRef(false);
   141→  const logId = useRef(0);
   142→
   143→  useEffect(() => {
   144→    return () => {
   145→      cancelled.current = true;
   146→    };
   147→  }, []);
   148→
   149→  const reset = () => {
   150→    cancelled.current = true;
   151→    setPhase("idle");
   152→    setCong(SEGMENTS.map(() => "LOW"));
   153→    setRevealed(SEGMENTS.map(() => false));
   154→    setCars(SEGMENTS.map(() => 4));
   155→    setAlerted(SEGMENTS.map(() => false));
   156→    setPulse(-1);
   157→    setAmbX(1);
   158→    setLog([]);
   159→    setSavedSec(0);
   160→    setAlertedTotal(0);
   161→    setMethod("—");
   162→    // allow a fresh run on next tick
   163→    setTimeout(() => {
   164→      cancelled.current = false;
   165→    }, 0);
   166→  };
   167→
   168→  const run = async () => {
   169→    if (phase === "running") return;
   170→    cancelled.current = false;
   171→
   172→    // Generate today's traffic
   173→    const nextCong: Congestion[] = SEGMENTS.map(() => pickCongestion());
   174→    const nextCars: number[] = SEGMENTS.map(() => randInt(3, 6));
   175→
   176→    setPhase("running");
   177→    setCong(nextCong);
   178→    setCars(nextCars);
   179→    setRevealed(SEGMENTS.map(() => false));
   180→    setAlerted(SEGMENTS.map(() => false));
   181→    setLog([]);
   182→    setSavedSec(0);
   183→    setAlertedTotal(0);
   184→    setMethod("Scanning traffic…");
   185→    setPulse(-1);
   186→    setAmbX(1);
   187→
   188→    await sleep(700);
   189→    if (cancelled.current) return;
   190→    setRevealed(SEGMENTS.map(() => true));
   191→
   192→    await sleep(600);
   193→    if (cancelled.current) return;
   194→
   195→    let totalSaved = 0;
   196→    let totalAlerted = 0;
   197→
   198→    for (let i = 0; i < SEGMENTS.length; i++) {
   199→      // Ambulance rolls into the segment
   200→      setAmbX(i * 20 + 1);
   201→      await sleep(150);
   202→      if (cancelled.current) return;
   203→
   204→      // Corridor alert fires for this segment
   205→      const meta = META[nextCong[i]];
   206→      const vehicles = nextCars[i] + randInt(1, 3);
   207→      const saved = randInt(meta.saved[0], meta.saved[1]);
   208→      totalSaved += saved;
   209→      totalAlerted += vehicles;
   210→
   211→      setAlerted((prev) => prev.map((v, idx) => idx === i || v));
   212→      setPulse(i);
   213→      setMethod(meta.method);
   214→      setSavedSec(totalSaved);
   215→      setAlertedTotal(totalAlerted);
   216→      setLog((prev) => [
   217→        {
   218→          id: ++logId.current,
   219→          segment: SEGMENTS[i].name,
   220→          method: meta.method,
   221→          vehicles,
   222→          saved,
   223→          congestion: nextCong[i],
   224→        },
   225→        ...prev,
   226→      ]);
   227→
   228→      await sleep(650); // cars react & shift
   229→      if (cancelled.current) return;
   230→      setPulse(-1);
   231→
   232→      // Cross the segment
   233→      setAmbX(i * 20 + 19);
   234→      await sleep(1800);
   235→      if (cancelled.current) return;
   236→    }
   237→
   238→    setMethod("Mission complete");
   239→    setPhase("done");
   240→    toast.success(
   241→      `Corridor run complete — ${totalSaved}s saved, ${totalAlerted} vehicles alerted across ${SEGMENTS.length} segments.`
   242→    );
   243→  };
   244→
   245→  const totalDistance = Math.round(((ambX - 1) / 98) * TOTAL_KM * 10) / 10;
   246→
   247→  return (
   248→    <Card className="overflow-hidden p-0">
   249→      {/* Header */}
   250→      <div className="flex flex-wrap items-center justify-between gap-2 border-b bg-muted/40 px-4 py-3 sm:px-5">
   251→        <div className="flex items-center gap-2">
   252→          <span className="flex size-8 items-center justify-center rounded-lg bg-red-100 text-red-700">
   253→            <Ambulance className="size-4" aria-hidden />
   254→          </span>
   255→          <div>
   256→            <p className="text-sm font-bold leading-tight">Live corridor simulation</p>
   257→            <p className="text-[11px] text-muted-foreground">
   258→              Hebbal → Tin Factory · {TOTAL_KM} km · randomised traffic
   259→            </p>
   260→          </div>
   261→        </div>
   262→        <Badge className={cn("border-0", method === "—" ? "bg-zinc-200 text-zinc-600" : method === "Mission complete" ? "bg-emerald-600 text-white" : "bg-red-600 text-white")}>
   263→          <Radio className="mr-1 size-3" aria-hidden />
   264→          {method}
   265→        </Badge>
   266→      </div>
   267→
   268→      {/* Segment labels + congestion chips */}
   269→      <div className="grid grid-cols-5 gap-1 px-3 pt-3 sm:px-5">
   270→        {SEGMENTS.map((s, i) => (
   271→          <div key={s.name} className="text-center">
   272→            <p className="truncate text-[10px] font-semibold text-muted-foreground sm:text-xs">
   273→              {s.name}
   274→            </p>
   275→            <span
   276→              className={cn(
   277→                "mt-1 inline-block rounded-full px-1.5 py-0.5 text-[9px] font-bold sm:text-[10px]",
   278→                revealed[i] ? META[cong[i]].chip : "bg-muted text-muted-foreground"
   279→              )}
   280→            >
   281→              {revealed[i] ? META[cong[i]].label : "—"}
   282→            </span>
   283→          </div>
   284→        ))}
   285→      </div>
   286→
   287→      {/* Road */}
   288→      <div className="px-3 pt-2 sm:px-5">
   289→        <div className="relative h-32 overflow-hidden rounded-xl bg-zinc-300 sm:h-36" role="img" aria-label="Road simulation with an ambulance crossing five segments">
   290→          {/* Asphalt + lanes */}
   291→          <div className="absolute inset-x-0 top-6 bottom-6 bg-zinc-500" />
   292→          <div className="absolute inset-x-0 top-[calc(50%+1px)] border-t-2 border-dashed border-white/70" aria-hidden />
   293→
   294→          {/* Congestion tints */}
   295→          {SEGMENTS.map((s, i) => (
   296→            <div
   297→              key={s.name}
   298→              className="absolute top-6 bottom-6 transition-colors duration-700"
   299→              style={{
   300→                left: `${i * 20}%`,
   301→                width: "20%",
   302→                backgroundColor: revealed[i] ? META[cong[i]].color : "transparent",
   303→                opacity: revealed[i] ? 0.35 : 0,
   304→              }}
   305→            />
   306→          ))}
   307→
   308→          {/* Segment dividers */}
   309→          {[20, 40, 60, 80].map((p) => (
   310→            <div
   311→              key={p}
   312→              aria-hidden
   313→              className="absolute top-4 bottom-4 w-px bg-white/40"
   314→              style={{ left: `${p}%` }}
   315→            />
   316→          ))}
   317→
   318→          {/* Pulse ring at alerting segment */}
   319→          {pulse >= 0 && (
   320→            <span
   321→              aria-hidden
   322→              className="absolute top-1/2 size-10 -translate-x-1/2 -translate-y-1/2 animate-ping rounded-full border-4 border-white/80 bg-white/30"
   323→              style={{ left: `${pulse * 20 + 10}%` }}
   324→            />
   325→          )}
   326→
   327→          {/* Cars */}
   328→          {SEGMENTS.map((s, i) =>
   329→            Array.from({ length: cars[i] }).map((_, j) => {
   330→              const pos = i * 20 + ((j + 0.8) / (cars[i] + 0.6)) * 20;
   331→              const isAlerted = alerted[i];
   332→              return (
   333→                <span
   334→                  key={`${i}-${j}`}
   335→                  className={cn(
   336→                    "absolute flex size-5 -translate-x-1/2 items-center justify-center rounded-full transition-all duration-700 sm:size-6",
   337→                    isAlerted ? "bg-emerald-500 text-white shadow-md" : "bg-zinc-400 text-zinc-600"
   338→                  )}
   339→                  style={{
   340→                    left: `${pos}%`,
   341→                    top: isAlerted ? "4px" : j % 2 === 0 ? "38%" : "62%",
   342→                  }}
   343→                >
   344→                  <Car className="size-3" aria-hidden />
   345→                </span>
   346→              );
   347→            })
   348→          )}
   349→
   350→          {/* Ambulance */}
   351→          <span
   352→            className="absolute top-1/2 z-10 flex size-9 -translate-x-1/2 -translate-y-1/2 items-center justify-center rounded-lg bg-white text-red-600 shadow-lg transition-[left] duration-[1800ms] ease-linear sm:size-10"
   353→            style={{ left: `${ambX}%` }}
   354→            aria-label="Ambulance"
   355→          >
   356→            <Ambulance className="size-5" aria-hidden />
   357→            <span className="absolute inset-0 rounded-lg ring-2 ring-red-500/60" aria-hidden />
   358→          </span>
   359→
   360→          {/* Coverage note */}
   361→          <div className="absolute bottom-1 right-2 text-[10px] font-medium text-white/90">
   362→            2 km ahead broadcast · 4-layer mesh
   363→          </div>
   364→        </div>
   365→      </div>
   366→
   367→      {/* Counters */}
   368→      <div className="grid grid-cols-2 gap-3 px-3 pt-4 sm:grid-cols-4 sm:px-5">
   369→        <div className="rounded-lg border bg-card p-3">
   370→          <p className="flex items-center gap-1 text-[11px] font-medium text-muted-foreground">
   371→            <Timer className="size-3.5" aria-hidden /> Time saved
   372→          </p>
   373→          <p className="text-lg font-bold text-emerald-700">{savedSec}s</p>
   374→        </div>
   375→        <div className="rounded-lg border bg-card p-3">
   376→          <p className="flex items-center gap-1 text-[11px] font-medium text-muted-foreground">
   377→            <Waves className="size-3.5" aria-hidden /> Vehicles alerted
   378→          </p>
   379→          <p className="text-lg font-bold">{alertedTotal}</p>
   380→        </div>
   381→        <div className="rounded-lg border bg-card p-3">
   382→          <p className="flex items-center gap-1 text-[11px] font-medium text-muted-foreground">
   383→            <Gauge className="size-3.5" aria-hidden /> Distance
   384→          </p>
   385→          <p className="text-lg font-bold">{totalDistance} km</p>
   386→        </div>
   387→        <div className="rounded-lg border bg-card p-3">
   388→          <p className="flex items-center gap-1 text-[11px] font-medium text-muted-foreground">
   389→            <Radio className="size-3.5" aria-hidden /> Alert method
   390→          </p>
   391→          <p className="truncate text-sm font-bold">{method}</p>
   392→        </div>
   393→      </div>
   394→
   395→      {/* Controls */}
   396→      <div className="flex flex-wrap items-center gap-2 px-3 pt-4 sm:px-5">
   397→        <Button
   398→          onClick={run}
   399→          disabled={phase === "running"}
   400→          className="bg-red-600 text-white hover:bg-red-700"
   401→        >
   402→          <Play className="size-4" aria-hidden />
   403→          {phase === "running" ? "Mission running…" : phase === "done" ? "Run again" : "Run mission"}
   404→        </Button>
   405→        <Button variant="outline" onClick={reset} disabled={phase === "running"}>
   406→          <RotateCcw className="size-4" aria-hidden /> Reset
   407→        </Button>
   408→        <p className="text-xs text-muted-foreground">
   409→          GRIDLOCK → V2X 5G + BLE · HEAVY → Cell broadcast · MODERATE → BLE mesh · LOW → FM RDS
   410→        </p>
   411→      </div>
   412→
   413→      {/* Event log */}
   414→      <div className="px-3 pb-4 pt-4 sm:px-5">
   415→        <p className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   416→          Per-segment event log
   417→        </p>
   418→        <div className="mt-2 max-h-48 overflow-y-auto rounded-lg border scrollbar-thin">
   419→          {log.length === 0 ? (
   420→            <p className="px-3 py-4 text-center text-xs text-muted-foreground">
   421→              No corridor events yet — press “Run mission”.
   422→            </p>
   423→          ) : (
   424→            <ul className="divide-y">
   425→              {log.map((e) => (
   426→                <li key={e.id} className="flex items-center gap-2 px-3 py-2 text-xs">
   427→                  <span className={cn("rounded px-1.5 py-0.5 text-[10px] font-bold", META[e.congestion].chip)}>
   428→                    {META[e.congestion].label}
   429→                  </span>
   430→                  <span className="min-w-0 flex-1 truncate font-medium">{e.segment}</span>
   431→                  <span className="hidden font-semibold text-zinc-600 sm:inline">{e.method}</span>
   432→                  <span className="text-muted-foreground">{e.vehicles} veh</span>
   433→                  <span className="font-bold text-emerald-700">−{e.saved}s</span>
   434→                </li>
   435→              ))}
   436→            </ul>
   437→          )}
   438→        </div>
   439→      </div>
   440→    </Card>
   441→  );
   442→}
