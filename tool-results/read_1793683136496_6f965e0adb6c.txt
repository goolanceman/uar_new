   141→function statusTone(status: string): string {
   142→  switch (status) {
   143→    case "ACTIVE":
   144→      return "bg-emerald-500/10 text-emerald-700 dark:text-emerald-400";
   145→    case "PAUSED":
   146→      return "bg-amber-500/10 text-amber-700 dark:text-amber-400";
   147→    case "COMPLETED":
   148→      return "bg-teal-500/10 text-teal-700 dark:text-teal-400";
   149→    case "CANCELLED":
   150→      return "bg-red-500/10 text-red-700 dark:text-red-400";
   151→    default:
   152→      return "bg-muted text-muted-foreground";
   153→  }
   154→}
   155→
   156→function slugify(title: string): string {
   157→  return title
   158→    .toLowerCase()
   159→    .trim()
   160→    .replace(/[^a-z0-9\s-]/g, "")
   161→    .replace(/[\s_]+/g, "-")
   162→    .replace(/-+/g, "-")
   163→    .replace(/^-|-$/g, "");
   164→}
   165→
   166→function isoToDateInput(value?: string | null): string {
   167→  if (!value) return "";
   168→  const d = new Date(value);
   169→  if (Number.isNaN(d.getTime())) return "";
   170→  const y = d.getFullYear();
   171→  const m = String(d.getMonth() + 1).padStart(2, "0");
   172→  const day = String(d.getDate()).padStart(2, "0");
   173→  return `${y}-${m}-${day}`;
   174→}
   175→
   176→function dateInputToIso(value: string): string | null {
   177→  if (!value) return null;
   178→  const d = new Date(`${value}T12:00:00`);
   179→  return Number.isNaN(d.getTime()) ? null : d.toISOString();
   180→}

   182→function ProgressRing({ pct, accent }: { pct: number; accent: string }) {
   183→  const r = 42;
   184→  const c = 2 * Math.PI * r;
   185→  const tone = ACCENTS[accent]?.chip ?? ACCENTS.emerald.chip;
   186→  return (
   187→    <div className={cn("relative mx-auto size-28", tone)}>
   188→      <svg viewBox="0 0 100 100" className="size-full -rotate-90" aria-hidden="true">
   189→        <circle cx="50" cy="50" r={r} fill="none" stroke="currentColor" strokeOpacity="0.15" strokeWidth="8" />
   190→        <circle
   191→          cx="50"
   192→          cy="50"
   193→          r={r}
   194→          fill="none"
   195→          stroke="currentColor"
   196→          strokeWidth="8"
   197→          strokeLinecap="round"
   198→          strokeDasharray={c}
   199→          strokeDashoffset={c - (Math.min(100, pct) / 100) * c}
   200→        />
   201→      </svg>
   202→      <span className="absolute inset-0 flex items-center justify-center text-lg font-bold">
   203→        {pct}%
   204→      </span>
   205→    </div>
   206→  );
   207→}

   229→export default function CampaignsView() {
   230→  const [rows, setRows] = useState<Campaign[]>([]);
   231→  const [loading, setLoading] = useState(true);
   232→  const [saving, setSaving] = useState(false);
   233→  const [busyId, setBusyId] = useState<string | null>(null);
   234→
   235→  // filters
   236→  const [q, setQ] = useState("");
   237→  const [status, setStatus] = useState("ALL");
   238→  const [category, setCategory] = useState("ALL");
   239→
   240→  // dialogs / sheets
   241→  const [dialogOpen, setDialogOpen] = useState(false);
   242→  const [editingId, setEditingId] = useState<string | null>(null);
   243→  const [form, setForm] = useState({ ...emptyForm });
   244→  const [viewTarget, setViewTarget] = useState<Campaign | null>(null);
   245→  const [viewDonationCount, setViewDonationCount] = useState<number | null>(null);
   246→  const [deleteTarget, setDeleteTarget] = useState<Campaign | null>(null);
   247→
   248→  const load = useCallback(async () => {
   249→    setLoading(true);
   250→    try {
   251→      const data = await get<Campaign[] | { items?: Campaign[] }>("/api/campaigns?all=1");
   252→      setRows(Array.isArray(data) ? data : (data.items ?? []));
   253→    } catch (e) {
   254→      toast.error(e instanceof Error ? e.message : "Failed to load campaigns");
   255→    } finally {
   256→      setLoading(false);
   257→    }
   258→  }, []);
   259→
   260→  useEffect(() => {
   261→    load();
   262→  }, [load]);
   263→
   264→  const filtered = useMemo(() => {
   265→    const needle = q.trim().toLowerCase();
   266→    return rows.filter((c) => {
   267→      if (status !== "ALL" && c.status !== status) return false;
   268→      if (category !== "ALL" && c.category !== category) return false;
   269→      if (needle) {
   270→        const hay = `${c.title} ${c.slug ?? ""} ${c.tagline ?? ""} ${c.location ?? ""}`.toLowerCase();
   271→        if (!hay.includes(needle)) return false;
   272→      }
   273→      return true;
   274→    });
   275→  }, [rows, q, status, category]);
   276→
   277→  const openCreate = () => {
   278→    setEditingId(null);
   279→    setForm({ ...emptyForm, startDate: isoToDateInput(new Date().toISOString()) });
   280→    setDialogOpen(true);
   281→  };
   282→
   283→  const openEdit = (c: Campaign) => {
   284→    setEditingId(c.id);
   285→    setForm({
   286→      title: c.title,
   287→      slug: c.slug ?? "",
   288→      slugTouched: true,
   289→      tagline: c.tagline ?? "",
   290→      description: c.description ?? "",
   291→      category: c.category ?? "GENERAL",
   292→      targetAmount: c.targetAmount !== undefined ? String(c.targetAmount) : "",
   293→      startDate: isoToDateInput(c.startDate),
   294→      endDate: isoToDateInput(c.endDate),
   295→      location: c.location ?? "",
   296→      icon: c.icon ?? "heart",
   297→      accent: c.accent ?? "emerald",
   298→      helpline: c.helpline ?? "",
   299→      featured: Boolean(c.featured),
   300→      status: c.status ?? "ACTIVE",
   301→    });
   302→    setDialogOpen(true);
   303→  };
   304→
   305→  const openView = (c: Campaign) => {
   306→    setViewTarget(c);
   307→    setViewDonationCount(null);
   308→    get<unknown[] | { items?: unknown[] }>(`/api/donations?campaignId=${encodeURIComponent(c.id)}&limit=500`)
   309→      .then((d) => setViewDonationCount(Array.isArray(d) ? d.length : (d.items?.length ?? 0)))
   310→      .catch(() => setViewDonationCount(null));
   311→  };
   312→
   313→  const save = async () => {
   314→    if (!form.title.trim() || !form.description.trim() || !Number(form.targetAmount)) {
   315→      toast.error("Title, description and a valid target amount are required");
   316→      return;
   317→    }
   318→    if (form.startDate && form.endDate && form.endDate < form.startDate) {
   319→      toast.error("End date cannot be before the start date");
   320→      return;
   321→    }
   322→    setSaving(true);
   323→    try {
   324→      const body = {
   325→        title: form.title.trim(),
   326→        slug: (form.slug.trim() || slugify(form.title)) || undefined,
   327→        tagline: form.tagline.trim() || undefined,
   328→        description: form.description.trim(),
   329→        category: form.category,
   330→        targetAmount: Number(form.targetAmount),
   331→        startDate: form.startDate ? dateInputToIso(form.startDate) : undefined,
   332→        endDate: dateInputToIso(form.endDate),
   333→        location: form.location.trim() || undefined,
   334→        icon: form.icon,
   335→        accent: form.accent,
   336→        helpline: form.helpline.trim() || undefined,
   337→        featured: form.featured,
   338→        status: form.status,
   339→      };
   340→      if (editingId) {
   341→        await patch(`/api/campaigns/${editingId}`, body);
   342→        toast.success("Campaign updated");
   343→      } else {
   344→        await post("/api/campaigns", body);
   345→        toast.success("Campaign created");
   346→      }
   347→      setDialogOpen(false);
   348→      await load();
   349→    } catch (e) {
   350→      toast.error(e instanceof Error ? e.message : "Failed to save campaign");
   351→    } finally {
   352→      setSaving(false);
   353→    }
   354→  };
   355→
   356→  const setStatusFor = async (c: Campaign, next: string) => {
   357→    setBusyId(c.id);
   358→    try {
   359→      await patch(`/api/campaigns/${c.id}`, { status: next });
   360→      toast.success(`${c.title} → ${titleCase(next)}`);
   361→      await load();
   362→      setViewTarget((prev) => (prev && prev.id === c.id ? { ...prev, status: next } : prev));
   363→    } catch (e) {
   364→      toast.error(e instanceof Error ? e.message : "Failed to update status");
   365→    } finally {
   366→      setBusyId(null);
   367→    }
   368→  };
   369→
   370→  const requestDelete = (c: Campaign) => {
   371→    if ((c.raisedAmount ?? 0) > 0) {
   372→      toast.error(
   373→        `${c.title} has ${formatINR(c.raisedAmount)} raised — campaigns with funds cannot be deleted. Pause it or mark it Complete instead.`
   374→      );
   375→      return;
   376→    }
   377→    setDeleteTarget(c);
   378→  };
   379→
   380→  const confirmDelete = async () => {
   381→    if (!deleteTarget) return;
   382→    setSaving(true);
   383→    try {
   384→      await del(`/api/campaigns/${deleteTarget.id}`);
   385→      toast.success("Campaign deleted");
   386→      setDeleteTarget(null);
   387→      await load();
   388→    } catch (e) {
   389→      toast.error(e instanceof Error ? e.message : "Failed to delete campaign");
   390→    } finally {
   391→      setSaving(false);
   392→    }
   393→  };
   394→
   395→  const statusActions = (c: Campaign, withLabels = false) => {
   396→    const actions: { label: string; next: string; show: boolean; cls: string }[] = [
   397→      { label: "Activate", next: "ACTIVE", show: c.status === "PAUSED" || c.status === "DRAFT", cls: "" },
   398→      { label: "Pause", next: "PAUSED", show: c.status === "ACTIVE", cls: "" },
   399→      { label: "Complete", next: "COMPLETED", show: c.status === "ACTIVE" || c.status === "PAUSED", cls: "text-teal-600 hover:bg-teal-500/10 hover:text-teal-700" },
   400→    ];
   401→    return actions.filter((a) => a.show).map((a) => (
   402→      <Button
   403→        key={a.next}
   404→        size="sm"
   405→        variant="outline"
   406→        className={a.cls}
   407→        disabled={busyId === c.id}
   408→        aria-label={`${a.label} campaign ${c.title}`}
   409→        onClick={() => setStatusFor(c, a.next)}
   410→      >
   411→        {busyId === c.id && <Loader2 className="size-3.5 animate-spin" />}
   412→        {a.label}
   413→      </Button>
   414→    ));
   415→  };
   416→
   417→  return (
   418→    <div className="space-y-4 sm:space-y-6">
   419→      <PageHeader
   420→        title="Campaigns"
   421→        description="Fundraising drives with live progress, status control and public visibility."
   422→        icon={<Sprout className="size-5" aria-hidden="true" />}
   423→        actions={
   424→          <Button onClick={openCreate} aria-label="Create new campaign">
   425→            <Plus className="size-4" /> New campaign
   426→          </Button>
   427→        }
   428→      />
   429→
   430→      {/* Filters */}
   431→      <div className="flex flex-col gap-3 lg:flex-row lg:items-center">
   432→        <div className="relative flex-1">
   433→          <Search className="absolute left-3 top-1/2 size-4 -translate-y-1/2 text-muted-foreground" aria-hidden="true" />
   434→          <Input value={q} onChange={(e) => setQ(e.target.value)} placeholder="Search title, tagline or location…" className="pl-9" aria-label="Search campaigns" />
   435→        </div>
   436→        <div className="flex flex-wrap gap-2">
   437→          <Select value={status} onValueChange={setStatus}>
   438→            <SelectTrigger className="w-full sm:w-40" aria-label="Filter by status">
   439→              <SelectValue placeholder="All statuses" />
   440→            </SelectTrigger>
   441→            <SelectContent>
   442→              <SelectItem value="ALL">All statuses</SelectItem>
   443→              {STATUSES.map((s) => (
   444→                <SelectItem key={s} value={s}>{titleCase(s)}</SelectItem>
   445→              ))}
   446→            </SelectContent>
   447→          </Select>
   448→          <Select value={category} onValueChange={setCategory}>
   449→            <SelectTrigger className="w-full sm:w-44" aria-label="Filter by category">
   450→              <SelectValue placeholder="All categories" />
   451→            </SelectTrigger>
   452→            <SelectContent>
   453→              <SelectItem value="ALL">All categories</SelectItem>
   454→              {CATEGORIES.map((c) => (
   455→                <SelectItem key={c} value={c}>{titleCase(c)}</SelectItem>
   456→              ))}
   457→            </SelectContent>
   458→          </Select>
   459→          <span className="ml-auto self-center text-xs text-muted-foreground" aria-live="polite">
   460→            {loading ? "Loading…" : `${filtered.length} of ${rows.length} campaigns`}
   461→          </span>
   462→        </div>
   463→      </div>
   464→
   465→      {/* Card grid */}
   466→      {loading ? (
   467→        <div className="grid gap-4 sm:grid-cols-2 xl:grid-cols-3">
   468→          {Array.from({ length: 6 }).map((_, i) => (
   469→            <Skeleton key={i} className="h-64 rounded-xl" />
   470→          ))}
   471→        </div>
   472→      ) : filtered.length === 0 ? (
   473→        <EmptyState
   474→          icon={<Sprout className="size-6" aria-hidden="true" />}
   475→          title={rows.length === 0 ? "No campaigns yet" : "No campaigns match the filters"}
   476→          hint={rows.length === 0 ? "Launch your first fundraising drive to start collecting donations." : "Try a different search term or clear the filters."}
   477→        >
   478→          {rows.length === 0 ? (
   479→            <Button size="sm" onClick={openCreate}>
   480→              <Plus className="size-4" /> New campaign
   481→            </Button>
   482→          ) : (
   483→            <Button size="sm" variant="outline" onClick={() => { setQ(""); setStatus("ALL"); setCategory("ALL"); }}>
   484→              Clear filters
   485→            </Button>
   486→          )}
   487→        </EmptyState>
   488→      ) : (
   489→        <div className="grid gap-4 sm:grid-cols-2 xl:grid-cols-3">
   490→          {filtered.map((c) => {
   491→            const Icon = ICONS[c.icon] ?? Heart;
   492→            const accent = ACCENTS[c.accent] ?? ACCENTS.emerald;
   493→            const pct = c.targetAmount > 0 ? Math.min(100, Math.round((c.raisedAmount / c.targetAmount) * 100)) : 0;
   494→            return (
   495→              <article key={c.id} className="flex flex-col gap-3 rounded-xl border bg-card p-4">
   496→                <div className="flex items-start gap-3">
   497→                  <span className={cn("flex size-10 shrink-0 items-center justify-center rounded-xl", accent.chip)}>
   498→                    <Icon className="size-5" aria-hidden="true" />
   499→                  </span>
   500→                  <div className="min-w-0 flex-1">
   501→                    <div className="flex items-center gap-1.5">
   502→                      <span className={cn("size-2 shrink-0 rounded-full", accent.dot)} aria-hidden="true" />
   503→                      {c.featured && <Star className="size-3.5 fill-amber-400 text-amber-500" aria-label="Featured campaign" />}
   504→                      <h3 className="truncate text-sm font-semibold leading-tight">{c.title}</h3>
   505→                    </div>
   506→                    {c.slug && <p className="truncate font-mono text-[11px] text-muted-foreground">/{c.slug}</p>}
   507→                  </div>
   508→                  <span className={cn("shrink-0 rounded-full px-2 py-0.5 text-[11px] font-semibold", statusTone(c.status))}>
   509→                    {titleCase(c.status)}
   510→                  </span>
   511→                </div>
   512→
   513→                {c.tagline && <p className="line-clamp-2 text-xs text-muted-foreground">{c.tagline}</p>}
   514→
   515→                <div className="space-y-1.5">
   516→                  <Progress value={pct} className={accent.bar} aria-label={`${c.title} funding progress: ${pct}%`} />
