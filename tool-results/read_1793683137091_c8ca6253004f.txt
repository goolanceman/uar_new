   219→
   220→function parseRoute(json: string | null | undefined): { name: string; congestion: string; lengthKm: number }[] {
   221→  if (!json) return [];
   222→  try {
   223→    const parsed = JSON.parse(json) as { name?: string; congestion?: string; lengthKm?: number }[];
   224→    return Array.isArray(parsed)
   225→      ? parsed.map((s) => ({
   226→          name: String(s?.name ?? "Segment"),
   227→          congestion: String(s?.congestion ?? "MODERATE"),
   228→          lengthKm: Number(s?.lengthKm ?? 0),
   229→        }))
   230→      : [];
   231→  } catch {
   232→    return [];
   233→  }
   234→}

   314→      code: a.code ?? "",
   315→      type: a.type ?? "BLS",

   330→    try {
   331→      const body = {
   332→        code: ambForm.code.trim().toUpperCase(),
   333→        type: ambForm.type,
   334→        base: ambForm.base.trim(),
   335→        driverName: ambForm.driverName.trim(),
   336→        driverPhone: ambForm.driverPhone.trim() || undefined,
   337→        crew: ambForm.crew.trim() || undefined,
   338→        equipment: ambForm.equipment.trim() || undefined,
   339→      };

   341→        await patch(`/api/ambulances/${editingAmb.id}`, body);
   342→        toast.success(`Ambulance ${body.code} updated`);
   343→      } else {
   344→        await post("/api/ambulances", body);
   345→        toast.success(`Ambulance ${body.code} added to the fleet`);
   346→      }
   347→      setAmbDialog(false);
   348→      load();
   349→    } catch (e) {
   350→      toast.error(e instanceof Error ? e.message : "Save failed");

   353→    }

   360→      toast.success(`Ambulance ${deleteAmb.code} removed`);
   361→      setDeleteAmb(null);

   363→    } catch (e) {

   365→    }

   372→    setMissionDialog(true);

   387→        pickup: missionForm.pickup.trim(),
   388→        destination: missionForm.destination.trim(),
   389→        route: missionForm.route

   395→      load();
   396→    } catch (e) {
   397→      toast.error(e instanceof Error ? e.message : "Create failed");

   399→      setSavingMission(false);
   400→    }

   494→        icon={<AmbulanceIcon className="size-4" aria-hidden="true" />}

   498→            {[0, 1, 2].map((i) => (
   499→              <Skeleton key={i} className="h-12 rounded-lg" />
   500→            ))}
   501→          </div>

   503→          <EmptyState

   508→            <Button variant="outline" onClick={openAmbCreate}>
   509→              <Plus className="size-4" aria-hidden="true" /> Add ambulance

   512→        ) : (
