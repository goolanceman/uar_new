    95→function normalize(raw: unknown): SettingsMap {
    96→  if (Array.isArray(raw)) {
    97→    const out: SettingsMap = {};
    98→    for (const item of raw as Record<string, unknown>[]) {
    99→      if (item && typeof item.key === "string") out[item.key] = String(item.value ?? "");
   100→    }
   101→    return out;
   102→  }
   103→  if (raw && typeof raw === "object") {
   104→    const out: SettingsMap = {};
   105→    for (const [k, v] of Object.entries(raw as Record<string, unknown>)) {
   106→      if (typeof v === "string" || typeof v === "number" || typeof v === "boolean") out[k] = String(v);
   107→    }
   108→    return out;
   109→  }
   110→  return {};
   111→}

   178→function TestChip({ result }: { result: TestResult | undefined }) {
   179→  if (!result) return null;
   180→  if (result.state === "running") {
   181→    return (
   182→      <Badge variant="outline" className="gap-1 border-amber-500/30 bg-amber-500/10 text-amber-700 dark:text-amber-400">

   184→      </Badge>

   188→    return (
   189→      <Badge variant="outline" className="gap-1 border-emerald-500/30 bg-emerald-500/10 text-emerald-700 dark:text-emerald-400">

   191→        OK · {result.latencyMs ?? 0}ms
