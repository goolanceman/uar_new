    95→    95→function normalize(raw: unknown): SettingsMap {
    96→    96→  if (Array.isArray(raw)) {
    97→    97→    const out: SettingsMap = {};
    98→    98→    for (const item of raw as Record<string, unknown>[]) {
    99→    99→      if (item && typeof item.key === "string") out[item.key] = String(item.value ?? "");
   100→   100→    }
   101→   101→    return out;
   102→   102→  }
   103→   103→  if (raw && typeof raw === "object") {
   104→   104→    const out: SettingsMap = {};
   105→   105→    for (const [k, v] of Object.entries(raw as Record<string, unknown>)) {
   106→   106→      if (typeof v === "string" || typeof v === "number" || typeof v === "boolean") out[k] = String(v);
   107→   107→    }
   108→   108→    return out;
   109→   109→  }
   110→   110→  return {};
   111→   111→}
