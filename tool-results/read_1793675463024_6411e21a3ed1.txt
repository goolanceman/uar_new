    37→type Rec = Record<string, unknown>;
    38→type ReportType = "overview" | "donations" | "distributions" | "causes" | "animals";
    39→
    40→function pickNum(o: Rec | undefined, keys: string[], fb = 0): number {
    41→  if (!o) return fb;
    42→  for (const k of keys) {
    43→    const v = o[k];
    44→    if (typeof v === "number" && Number.isFinite(v)) return v;
    45→    if (typeof v === "string" && v !== "" && Number.isFinite(Number(v))) return Number(v);
    46→  }
    47→  return fb;
    48→}
    49→
    50→function pickStr(o: Rec | undefined, keys: string[], fb = ""): string {
    51→  if (!o) return fb;
    52→  for (const k of keys) {
    53→    const v = o[k];
    54→    if (typeof v === "string" && v) return v;
    55→  }
    56→  return fb;
    57→}
    58→
    59→function asArray(v: unknown): Rec[] {
    60→  if (Array.isArray(v)) return v as Rec[];
    61→  if (v && typeof v === "object") {
    62→    const o = v as Rec;
    63→    if (Array.isArray(o.items)) return o.items as Rec[];
    64→    if (Array.isArray(o.data)) return o.data as Rec[];
    65→  }
    66→  return [];
    67→}
    68→
    69→function downloadCsv(filename: string, rows: (string | number)[][]) {
    70→  const csv = rows
    71→    .map((r) => r.map((c) => `"${String(c ?? "").replace(/"/g, '""')}"`).join(","))
    72→    .join("\n");
    73→  const blob = new Blob([`\uFEFF${csv}`], { type: "text/csv;charset=utf-8;" });
    74→  const url = URL.createObjectURL(blob);
    75→  const a = document.createElement("a");
    76→  a.href = url;
    77→  a.download = filename;
    78→  a.click();
    79→  URL.revokeObjectURL(url);
    80→}

    82→function statusBadgeClass(status: string): string {
    83→  switch (status) {
    84→    case "ADOPTED":
    85→    case "COMPLETED":
    86→      return "bg-emerald-100 text-emerald-700";
    87→    case "PARTIAL":
    88→    case "FOSTERED":
    89→      return "bg-amber-100 text-amber-700";
    90→    case "PAUSED":
    91→      return "bg-zinc-100 text-zinc-600";
    92→    case "CRITICAL":
    93→      return "bg-red-100 text-red-700";
    94→    default:
    95→      return "bg-teal-100 text-teal-700";
    96→  }
    97→}

   229→export default function ReportsView() {
   230→  const [type, setType] = useState<ReportType>("overview");
   231→  const [from, setFrom] = useState("");
   232→  const [to, setTo] = useState("");
   233→  const [data, setData] = useState<Rec | null>(null);
   234→  const [loading, setLoading] = useState(true);
   235→
   236→  const load = useCallback(async () => {
   237→    setLoading(true);
   238→    try {
   239→      const params = new URLSearchParams({ type });
   240→      if (from) params.set("from", from);
   241→      if (to) params.set("to", to);
   242→      const res = await get<Rec>(`/api/reports?${params.toString()}`);
   243→      setData(res ?? {});
   244→    } catch (e) {
   245→      toast.error(e instanceof Error ? e.message : "Failed to load report");
   246→      setData({});
   247→    } finally {
   248→      setLoading(false);
   249→    }
   250→  }, [type, from, to]);
   251→
   252→  useEffect(() => {
   253→    load();
   254→  }, [load]);
   255→
   256→  const donationsRows = useMemo(() => asArray(data?.donations ?? (type === "donations" ? data : null)), [data, type]);
   257→  const distributionRows = useMemo(() => asArray(data?.distributions ?? (type === "distributions" ? data : null)), [data, type]);
   258→  const causeRows = useMemo(() => asArray(data?.causes ?? (type === "causes" ? data : null)), [data, type]);
   259→  const animalRows = useMemo(() => asArray(data?.animals ?? (type === "animals" ? data : null)), [data, type]);

   270→          pickNum(m, ["distributions", "distributionCount", "distributionValue", "count"]),
   271→        ]),
   272→      ]);

   275→        ["Receipt", "Donor", "Campaign", "Amount", "Type", "Status", "Date"],
   276→        ...donationsRows.map((d) => [
   277→          pickStr(d, ["receiptNumber"]),
   278→          pickStr(d, ["donorName"]),
   279→          typeof d.campaign === "object" && d.campaign ? pickStr(d.campaign as Rec, ["title"]) : pickStr(d, ["campaignTitle"]),
   280→          pickNum(d, ["amount"]),
   281→          pickStr(d, ["type"]),
   282→          pickStr(d, ["paymentStatus"]),
   283→          formatDate(pickStr(d, ["createdAt"])),
   284→        ]),
   285→      ]);
   286→    } else if (type === "distributions") {
   287→      downloadCsv(`uar-report-distributions-${stamp}.csv`, [
   288→        ["Type", "Quantity", "Value", "Location", "Date"],
   289→        ...distributionRows.map((d) => [
   290→          pickStr(d, ["type"]),
   291→          pickNum(d, ["quantity"]),
   292→          pickNum(d, ["value"]),
   293→          pickStr(d, ["location"]),
   294→          formatDate(pickStr(d, ["distributedAt", "createdAt"])),
   295→        ]),
   296→      ]);
   297→    } else if (type === "causes") {
   298→      downloadCsv(`uar-report-causes-${stamp}.csv`, [
   299→        ["Code", "Patient", "City", "Condition", "Monthly Cost", "Duration (months)", "Raised", "Status"],
   300→        ...causeRows.map((c) => [
   301→          pickStr(c, ["code"]),

   310→      ]);

   313→        ["Name", "Species", "Location", "Monthly Food", "Status"],

   315→          pickStr(a, ["name"]),
   316→          pickStr(a, ["species"]),
   317→          pickStr(a, ["location"]),
   318→          pickNum(a, ["monthlyFood"]),
   319→          pickStr(a, ["status"]),

   340→          <TabsList className="h-auto w-full flex-wrap justify-start sm:w-auto">
