   230→
   231→function kindMeta(kind?: string | null) {
   232→  const k = (kind ?? "").toUpperCase() as KitKind;
   233→  return KIND_META[k] ?? KIND_META.CUSTOM;
   234→}
   235→
   236→function prefixFor(t: { kind?: string | null; codePrefix?: string | null }): string {
   237→  const p = (t.codePrefix ?? "").trim();
   238→  if (p) return p;
   239→  const k = (t.kind ?? "").toUpperCase() as KitKind;
   240→  return KIND_META[k]?.prefix ?? "UAR-RK";
   241→}

   245→const EMPTY_TPL_FORM: TemplateForm = {
   246→  name: "",
   247→  kind: "RATION",
   248→  description: "",
   249→  codePrefix: "",
   250→  image: "",

   256→const QR_SHEET_LIMIT = 200;
   257→
   258→function parseItems(json: string | null | undefined): TplItem[] {
   259→  if (!json) return [];
   260→  try {
   261→    const parsed = JSON.parse(json) as { name?: string; qty?: number | string; unit?: string }[];
   262→    if (!Array.isArray(parsed)) return [];
   263→    return parsed.map((i) => ({
   264→      name: String(i?.name ?? ""),
   265→      qty: String(i?.qty ?? ""),
   266→      unit: String(i?.unit ?? "pcs"),
   267→    }));
   268→  } catch {
   269→    return [];
   270→  }
   271→}
   272→
   273→function scanPct(b: Pick<KitBatch, "totalUnits" | "scannedUnits">): number {
   274→  return b.totalUnits > 0
   275→    ? Math.min(100, Math.round(((b.scannedUnits ?? 0) / b.totalUnits) * 100))
   276→    : 0;
   277→}
