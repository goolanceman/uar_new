   157→   157→function downloadCsv(filename: string, rows: (string | number)[][]) {
   158→   158→  const csv = rows
   159→   159→    .map((r) => r.map((c) => `"${String(c ?? "").replace(/"/g, '""')}"`).join(","))
   160→   160→    .join("\n");
   161→   161→  const blob = new Blob([`\uFEFF${csv}`], { type: "text/csv;charset=utf-8;" });
   162→   162→  const url = URL.createObjectURL(blob);
   163→   163→  const a = document.createElement("a");
   164→   164→  a.href = url;
   165→   165→  a.download = filename;
   166→   166→  a.click();
   167→   167→  URL.revokeObjectURL(url);
   168→   168→}

   170→   170→/** Normalises an Indian mobile number → 10 digits, or null when invalid. */
   171→   171→function normalisePhone(raw: string): string | null {
   172→   172→  const digits = raw.replace(/\D/g, "");
   173→   173→  const trimmed = digits.length > 10 && digits.startsWith("91") ? digits.slice(2) : digits;
   174→   174→  return trimmed.length === 10 ? trimmed : null;
   175→   175→}
