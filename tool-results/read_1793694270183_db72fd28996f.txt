   377→function pad2(n: number): string {
   378→  return String(n).padStart(2, "0");
   379→}
   380→
   381→/** Local-time value for <input type="datetime-local"> */
   382→function nowLocalInput(): string {
   383→  const d = new Date();
   384→  return `${d.getFullYear()}-${pad2(d.getMonth() + 1)}-${pad2(d.getDate())}T${pad2(d.getHours())}:${pad2(d.getMinutes())}`;
   385→}
   386→
   387→function toLocalInput(iso: string | null | undefined): string {
   388→  if (!iso) return "";
   389→  const d = new Date(iso);
   390→  if (Number.isNaN(d.getTime())) return "";
   391→  return `${d.getFullYear()}-${pad2(d.getMonth() + 1)}-${pad2(d.getDate())}T${pad2(d.getHours())}:${pad2(d.getMinutes())}`;
   392→}
   393→
   394→function todayIso(): string {
   395→  const d = new Date();
   396→  return `${d.getFullYear()}-${pad2(d.getMonth() + 1)}-${pad2(d.getDate())}`;
   397→}
   398→
   399→/** Financial-year window (Apr → Mar) as ISO strings for API filters. */
   400→function fyRange(): { from: string; to: string } {
   401→  const now = new Date();
   402→  const y = now.getMonth() + 1 >= 4 ? now.getFullYear() : now.getFullYear() - 1;
   403→  return { from: `${y}-04-01T00:00:00`, to: `${y + 1}-03-31T23:59:59.999` };
   404→}
   405→
   406→function rangeToParams(range: DateRange): { from?: string; to?: string } {
   407→  if (range === "all") return {};
   408→  if (range === "fy") return fyRange();
   409→  const days = range === "30" ? 30 : 90;
   410→  const from = new Date(Date.now() - days * 86_400_000);
   411→  return {
   412→    from: `${from.getFullYear()}-${pad2(from.getMonth() + 1)}-${pad2(from.getDate())}T00:00:00`,
   413→  };
   414→}

   416→interface ParsedItem {
   417→  name?: string;
   418→  qty?: number | string;
   419→  unit?: string;
   420→}
   421→
   422→function parseItems(json: string | null | undefined): ParsedItem[] {
   423→  if (!json) return [];
   424→  try {
   425→    const parsed = JSON.parse(json);
   426→    return Array.isArray(parsed) ? (parsed as ParsedItem[]) : [];
   427→  } catch {
   428→    return [];
   429→  }
   430→}
   431→
   432→function itemsSummary(json: string | null | undefined): string {
   433→  const items = parseItems(json);
   434→  if (!items.length) return "—";
   435→  return items
   436→    .map((i) => {
   437→      const name = String(i?.name ?? i ?? "").trim();
   438→      const qty = i?.qty;
   439→      const unit = i?.unit ? ` ${i.unit}` : "";
   440→      return `${name} ×${qty ?? 1}${unit}`;
   441→    })
   442→    .join(", ");
   443→}
