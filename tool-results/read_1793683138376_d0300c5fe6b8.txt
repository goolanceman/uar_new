    76→/* ---------------------------------- types ---------------------------------- */
    77→
    78→interface TplItem {
    79→  name: string;
    80→  qty: string;
    81→  unit: string;
    82→}
    83→
    84→interface KitTemplate {
    85→  id: string;
    86→  name: string;
    87→  description?: string | null;
    88→  items: string;
    89→  costPerUnit: number;
    90→  _count?: { batches: number };
    91→}
    92→
    93→interface KitBatch {
    94→  id: string;
    95→  templateId: string;
    96→  name: string;
    97→  totalUnits: number;
    98→  scannedUnits: number;
    99→  instructions?: string | null;
   100→  status: string;
   101→  template?: { name: string; costPerUnit: number } | null;
   102→}
   103→
   104→interface KitUnit {
   105→  id: string;
   106→  batchId: string;
   107→  code: string;
   108→  status: string;
   109→  scannedAt?: string | null;
   110→  scannedByName?: string | null;
   111→  location?: string | null;
   112→}
   113→
   114→interface TemplateForm {
   115→  name: string;
   116→  description: string;
   117→  costPerUnit: string;
   118→  items: TplItem[];
   119→}
   120→
   121→/* ------------------------------- chip styles ------------------------------- */
   122→
   123→const BATCH_STATUS: Record<string, string> = {
   124→  READY: "border-emerald-500/30 bg-emerald-500/10 text-emerald-700 dark:text-emerald-400",
   125→  IN_PROGRESS: "border-amber-500/30 bg-amber-500/10 text-amber-700 dark:text-amber-400",
   126→  DEPLETED: "border-rose-500/30 bg-rose-500/10 text-rose-700 dark:text-rose-400",
   127→};
   128→
   129→const UNIT_STATUS: Record<string, string> = {
   130→  READY: "border-emerald-500/30 bg-emerald-500/10 text-emerald-700 dark:text-emerald-400",
   131→  SCANNED: "border-teal-500/30 bg-teal-500/10 text-teal-700 dark:text-teal-400",
   132→  EXPIRED: "border-zinc-500/30 bg-zinc-500/10 text-zinc-600 dark:text-zinc-400",
   133→};
   134→
   135→const EMPTY_TPL_FORM: TemplateForm = {
   136→  name: "",
   137→  description: "",
   138→  costPerUnit: "",
   139→  items: [{ name: "", qty: "1", unit: "kg" }],
   140→};
   141→
   142→const QR_SHEET_LIMIT = 200;
   143→
   144→function parseItems(json: string | null | undefined): TplItem[] {
   145→  if (!json) return [];
   146→  try {
   147→    const parsed = JSON.parse(json) as { name?: string; qty?: number | string; unit?: string }[];
   148→    if (!Array.isArray(parsed)) return [];
   149→    return parsed.map((i) => ({
   150→      name: String(i?.name ?? ""),
   151→      qty: String(i?.qty ?? ""),
   152→      unit: String(i?.unit ?? "pcs"),
   153→    }));
   154→  } catch {
   155→    return [];
   156→  }
   157→}

   159→/* ---------------------------------- view ----------------------------------- */
   160→
   161→export default function KitsView() {
   162→  const [templates, setTemplates] = useState<KitTemplate[]>([]);
   163→  const [batches, setBatches] = useState<KitBatch[]>([]);
   164→  const [loading, setLoading] = useState(true);
   165→
   166→  /* template dialog */
   167→  const [tplDialog, setTplDialog] = useState(false);
   168→  const [editingTpl, setEditingTpl] = useState<KitTemplate | null>(null);
   169→  const [tplForm, setTplForm] = useState<TemplateForm>(EMPTY_TPL_FORM);
   170→  const [savingTpl, setSavingTpl] = useState(false);
   171→  const [deleteTpl, setDeleteTpl] = useState<KitTemplate | null>(null);
   172→
   173→  /* batch wizard */
   174→  const [wizardOpen, setWizardOpen] = useState(false);
   175→  const [wizardStep, setWizardStep] = useState(1);
   176→  const [wizardTemplateId, setWizardTemplateId] = useState("");
   177→  const [wizardName, setWizardName] = useState("");
   178→  const [wizardUnits, setWizardUnits] = useState("25");
   179→  const [wizardInstructions, setWizardInstructions] = useState("");
   180→  const [savingBatch, setSavingBatch] = useState(false);
   181→  const [deleteBatch, setDeleteBatch] = useState<KitBatch | null>(null);
   182→
   183→  /* batch detail sheet */
   184→  const [sheet, setSheet] = useState<KitBatch | null>(null);
   185→  const [units, setUnits] = useState<KitUnit[]>([]);
   186→  const [unitsLoading, setUnitsLoading] = useState(false);
   187→  const [unitStatus, setUnitStatus] = useState("ALL");
   188→  const [unitSearch, setUnitSearch] = useState("");
   189→
   190→  /* QR label dialog */
   191→  const [qrOpen, setQrOpen] = useState(false);
   192→  const [qrLabels, setQrLabels] = useState<{ code: string; dataUrl: string }[]>([]);
   193→  const [qrLoading, setQrLoading] = useState(false);
   194→
   195→  const unitOptions = useMasters("UNIT");
   196→
   197→  const load = useCallback(async () => {
   198→    try {
   199→      const [tpl, bat] = await Promise.all([get<KitTemplate[]>("/api/kit-templates"), get<KitBatch[]>("/api/kit-batches")]);
   200→      setTemplates(Array.isArray(tpl) ? tpl : []);
   201→      setBatches(Array.isArray(bat) ? bat : []);
   202→    } catch (e) {
   203→      toast.error(e instanceof Error ? e.message : "Failed to load kits data");
   204→    } finally {
   205→      setLoading(false);
   206→    }
   207→  }, []);
   208→
   209→  useEffect(() => {
   210→    load();
   211→  }, [load]);
   212→
   213→  const totals = useMemo(() => {
   214→    const totalUnits = batches.reduce((s, b) => s + (b.totalUnits ?? 0), 0);
   215→    const scanned = batches.reduce((s, b) => s + (b.scannedUnits ?? 0), 0);
   216→    return { totalUnits, scanned, batches: batches.length, templates: templates.length };
   217→  }, [batches, templates]);
   218→
   219→  /* ---------------------------- template actions ---------------------------- */
   220→
   221→  const openTplCreate = () => {
   222→    setEditingTpl(null);
   223→    setTplForm(EMPTY_TPL_FORM);
   224→    setTplDialog(true);
   225→  };
   226→
   227→  const openTplEdit = (t: KitTemplate) => {
   228→    setEditingTpl(t);
   229→    setTplForm({
   230→      name: t.name ?? "",
   231→      description: t.description ?? "",
   232→      costPerUnit: String(t.costPerUnit ?? ""),
   233→      items: parseItems(t.items).length ? parseItems(t.items) : [{ name: "", qty: "1", unit: "kg" }],
   234→    });
   235→    setTplDialog(true);
   236→  };
   237→
   238→  const computedCost = useMemo(() => {
   239→    const qty = tplForm.items.reduce((s, i) => s + (Number(i.qty) || 0), 0);
   240→    return { qty, perUnit: Number(tplForm.costPerUnit) || 0 };
   241→  }, [tplForm]);
   242→
   243→  const saveTpl = async () => {
   244→    if (!tplForm.name.trim()) return toast.error("Template name is required");
   245→    const items = tplForm.items.filter((i) => i.name.trim());
   246→    if (items.length === 0) return toast.error("Add at least one item with a name");
   247→    setSavingTpl(true);
   248→    try {
   249→      const body = {
   250→        name: tplForm.name.trim(),
   251→        description: tplForm.description.trim() || undefined,
   252→        items: items.map((i) => ({ name: i.name.trim(), qty: Number(i.qty) || 1, unit: i.unit })),
   253→        costPerUnit: Number(tplForm.costPerUnit) || 0,
   254→      };
   255→      if (editingTpl) {
   256→        await patch(`/api/kit-templates/${editingTpl.id}`, body);
   257→        toast.success(`Template "${body.name}" updated`);
   258→      } else {
   259→        await post("/api/kit-templates", body);
   260→        toast.success(`Template "${body.name}" created`);
   261→      }
   262→      setTplDialog(false);
   263→      load();
   264→    } catch (e) {
   265→      toast.error(e instanceof Error ? e.message : "Save failed");
   266→    } finally {
   267→      setSavingTpl(false);
   268→    }
   269→  };
   270→
   271→  const removeTpl = async () => {
   272→    if (!deleteTpl) return;
   273→    try {
   274→      await del(`/api/kit-templates/${deleteTpl.id}`);
   275→      toast.success(`Template "${deleteTpl.name}" deleted`);
   276→      setDeleteTpl(null);
   277→      load();
   278→    } catch (e) {
   279→      toast.error(e instanceof Error ? e.message : "Delete failed");
   280→    }
   281→  };
   282→
   283→  /* ------------------------------ batch wizard ------------------------------ */
   284→
   285→  const openWizard = () => {
   286→    setWizardStep(1);
   287→    setWizardTemplateId("");
   288→    setWizardName("");
   289→    setWizardUnits("25");
   290→    setWizardInstructions("");
   291→    setWizardOpen(true);
   292→  };
   293→
   294→  const createBatch = async () => {
   295→    if (!wizardTemplateId) return toast.error("Pick a template first");
   296→    if (!wizardName.trim()) return toast.error("Give the batch a name");
   297→    const units = Math.trunc(Number(wizardUnits));
   298→    if (!Number.isFinite(units) || units < 1 || units > 500) {
   299→      toast.error("Total units must be between 1 and 500");
   300→      return;
   301→    }
   302→    setSavingBatch(true);
   303→    try {
   304→      await post("/api/kit-batches", {
   305→        templateId: wizardTemplateId,
   306→        name: wizardName.trim(),
   307→        totalUnits: units,
   308→        instructions: wizardInstructions.trim() || undefined,
   309→      });
   310→      toast.success(`Batch "${wizardName.trim()}" created with ${units} QR-coded units`);
   311→      setWizardOpen(false);
   312→      load();
   313→    } catch (e) {
   314→      toast.error(e instanceof Error ? e.message : "Batch creation failed");
   315→    } finally {
   316→      setSavingBatch(false);
   317→    }
   318→  };
   319→
   320→  const removeBatch = async () => {
   321→    if (!deleteBatch) return;
   322→    try {
   323→      await del(`/api/kit-batches/${deleteBatch.id}`);
   324→      toast.success(`Batch "${deleteBatch.name}" deleted`);
   325→      setDeleteBatch(null);
   326→      setSheet(null);
   327→      load();
   328→    } catch (e) {
   329→      toast.error(e instanceof Error ? e.message : "Delete failed");
   330→    }
   331→  };
   332→
   333→  /* ------------------------------ unit sheet -------------------------------- */
   334→
   335→  const openSheet = (b: KitBatch) => {
   336→    setSheet(b);
   337→    setUnitStatus("ALL");
   338→    setUnitSearch("");
   339→    setUnitsLoading(true);
   340→    get<KitUnit[]>(`/api/kit-units?batchId=${b.id}&limit=500`)
   341→      .then((rows) => setUnits(Array.isArray(rows) ? rows : []))
   342→      .catch((e) => {
   343→        setUnits([]);
   344→        toast.error(e instanceof Error ? e.message : "Failed to load units");
   345→      })
   346→      .finally(() => setUnitsLoading(false));
   347→  };
   348→
   349→  const filteredUnits = useMemo(() => {
   350→    const q = unitSearch.trim().toUpperCase();
   351→    return units.filter((u) => {
   352→      if (unitStatus !== "ALL" && u.status !== unitStatus) return false;
   353→      if (q && !u.code.includes(q)) return false;
   354→      return true;
   355→    });
   356→  }, [units, unitStatus, unitSearch]);
   357→
   358→  const readyUnits = useMemo(() => units.filter((u) => u.status === "READY"), [units]);
   359→
   360→  /* ------------------------------ QR printing ------------------------------- */
   361→
   362→  useEffect(() => {
   363→    if (!qrOpen) return;
   364→    let alive = true;
   365→    const run = async () => {
   366→      setQrLoading(true);
   367→      const picked = readyUnits.slice(0, QR_SHEET_LIMIT);
   368→      const out: { code: string; dataUrl: string }[] = [];
   369→      for (const u of picked) {
   370→        try {
   371→          const dataUrl = await QRCode.toDataURL(u.code, { margin: 1, width: 140 });
   372→          out.push({ code: u.code, dataUrl });
   373→        } catch {
   374→          // skip a unit whose QR fails to render
   375→        }
   376→      }
   377→      if (alive) {
   378→        setQrLabels(out);
   379→        setQrLoading(false);
   380→      }
   381→    };
   382→    run();
   383→    return () => {
   384→      alive = false;
   385→    };
   386→  }, [qrOpen, readyUnits]);
   387→
   388→  /* --------------------------------- render -------------------------------- */
   389→
   390→  return (
   391→    <div className="space-y-6">
   392→      <PageHeader
   393→        title="Ration Kits"
   394→        description="Kit templates, QR-coded batches and distribution tracking."
   395→        icon={<ClipboardList className="size-5" aria-hidden="true" />}
   396→        actions={
   397→          <>
   398→            <Button variant="outline" onClick={load} aria-label="Refresh kits">
   399→              <RefreshCw className="size-4" aria-hidden="true" /> Refresh
   400→            </Button>
   401→            <Button onClick={openWizard}>
   402→              <Plus className="size-4" aria-hidden="true" /> New batch
   403→            </Button>
   404→          </>
   405→        }
   406→      />
   407→
   408→      <div className="grid gap-4 sm:grid-cols-2 xl:grid-cols-4">
   409→        <StatCard label="Templates" value={totals.templates} accent="emerald" icon={<ClipboardList className="size-5" aria-hidden="true" />} loading={loading} />
   410→        <StatCard label="Batches" value={totals.batches} accent="teal" icon={<Package className="size-5" aria-hidden="true" />} loading={loading} />
   411→        <StatCard label="Units assembled" value={totals.totalUnits.toLocaleString("en-IN")} accent="amber" icon={<Package className="size-5" aria-hidden="true" />} loading={loading} />
   412→        <StatCard label="Units scanned" value={totals.scanned.toLocaleString("en-IN")} accent="lime" icon={<QrCode className="size-5" aria-hidden="true" />} loading={loading} hint="At beneficiary hand-off" />
   413→      </div>
   414→
   415→      <Tabs defaultValue="batches">
   416→        <TabsList>
   417→          <TabsTrigger value="batches">Batches ({batches.length})</TabsTrigger>
   418→          <TabsTrigger value="templates">Templates ({templates.length})</TabsTrigger>
   419→        </TabsList>
   420→
   421→        {/* ============================== BATCHES ============================== */}
   422→        <TabsContent value="batches" className="mt-4">
   423→          {loading ? (
   424→            <div className="grid gap-4 sm:grid-cols-2 xl:grid-cols-3" aria-busy="true">
   425→              {[0, 1, 2].map((i) => (
   426→                <Skeleton key={i} className="h-56 rounded-xl" />
   427→              ))}
   428→            </div>
   429→          ) : batches.length === 0 ? (
   430→            <EmptyState
   431→              icon={<Package className="size-8" aria-hidden="true" />}
   432→              title="No batches yet"
   433→              hint="Run the wizard to assemble a batch of QR-coded ration kits from a template."
   434→            >
   435→              <Button onClick={openWizard}>
   436→                <Plus className="size-4" aria-hidden="true" /> New batch
   437→              </Button>
   438→            </EmptyState>
   439→          ) : (
   440→            <div className="grid gap-4 sm:grid-cols-2 xl:grid-cols-3">
   441→              {batches.map((b) => {
   442→                const pct =
   443→                  b.totalUnits > 0 ? Math.min(100, Math.round(((b.scannedUnits ?? 0) / b.totalUnits) * 100)) : 0;
   444→                return (
   445→                  <button
   446→                    key={b.id}
   447→                    type="button"
   448→                    onClick={() => openSheet(b)}
   449→                    aria-label={`Open batch ${b.name}`}
   450→                    className="flex flex-col gap-3 rounded-xl border bg-card p-4 text-left shadow-sm transition hover:border-emerald-500/40 hover:shadow-md focus-visible:outline-2 focus-visible:outline-primary"
   451→                  >
   452→                    <div className="flex items-start gap-2">
   453→                      <div className="min-w-0 flex-1">
   454→                        <p className="truncate font-semibold leading-tight">{b.name}</p>
   455→                        <p className="truncate text-xs text-muted-foreground">{b.template?.name ?? "Template"}</p>
   456→                      </div>
   457→                      <Badge variant="outline" className={cn("shrink-0", BATCH_STATUS[b.status] ?? "")}>
   458→                        {titleCase(b.status)}
   459→                      </Badge>
   460→                    </div>
   461→                    <div className="space-y-1.5">
   462→                      <div className="flex justify-between text-xs text-muted-foreground">
   463→                        <span>
   464→                          {b.scannedUnits ?? 0} / {b.totalUnits} scanned
   465→                        </span>
   466→                        <span>{pct}%</span>
   467→                      </div>
   468→                      <Progress value={pct} className="h-2 [&>div]:bg-emerald-600" aria-label={`${pct}% scanned`} />
   469→                    </div>
   470→                    {b.instructions ? (
   471→                      <p className="line-clamp-2 text-xs text-muted-foreground">{b.instructions}</p>
   472→                    ) : null}
   473→                    <p className="mt-auto text-xs text-muted-foreground">
   474→                      ~{formatINR((b.template?.costPerUnit ?? 0) * b.totalUnits)} of kit value
   475→                    </p>
   476→                  </button>
   477→                );
   478→              })}
   479→            </div>
   480→          )}
   481→        </TabsContent>
   482→
   483→        {/* ============================= TEMPLATES ============================= */}
   484→        <TabsContent value="templates" className="mt-4">
   485→          {loading ? (
   486→            <div className="grid gap-4 sm:grid-cols-2 xl:grid-cols-3" aria-busy="true">
   487→              {[0, 1, 2].map((i) => (
   488→                <Skeleton key={i} className="h-56 rounded-xl" />
   489→              ))}
   490→            </div>
   491→          ) : templates.length === 0 ? (
   492→            <EmptyState
   493→              icon={<ClipboardList className="size-8" aria-hidden="true" />}
   494→              title="No kit templates yet"
   495→              hint="Define what goes into a standard ration kit — items, quantities and cost."
   496→            >
   497→              <Button onClick={openTplCreate}>
   498→                <Plus className="size-4" aria-hidden="true" /> New template
   499→              </Button>
   500→            </EmptyState>
   501→          ) : (
   502→            <div className="grid gap-4 sm:grid-cols-2 xl:grid-cols-3">
   503→              {templates.map((t) => {
   504→                const items = parseItems(t.items);
   505→                return (
   506→                  <div key={t.id} className="flex flex-col rounded-xl border bg-card p-4 shadow-sm">
   507→                    <div className="flex items-start gap-2">
   508→                      <div className="min-w-0 flex-1">
   509→                        <p className="truncate font-semibold leading-tight">{t.name}</p>
   510→                        <p className="text-xs text-muted-foreground">
   511→                          {items.length} item{items.length === 1 ? "" : "s"} · {formatINR(t.costPerUnit)} per kit
   512→                        </p>
   513→                      </div>
   514→                      <Badge variant="outline" className="shrink-0">
   515→                        {t._count?.batches ?? 0} batch{(t._count?.batches ?? 0) === 1 ? "" : "es"}
   516→                      </Badge>
   517→                    </div>
   518→                    {t.description ? (
   519→                      <p className="mt-2 line-clamp-2 text-xs text-muted-foreground">{t.description}</p>
   520→                    ) : null}
   521→                    <ul className="mt-3 max-h-36 space-y-1 overflow-y-auto pr-1 text-xs scrollbar-thin">
   522→                      {items.map((i, idx) => (
   523→                        <li key={idx} className="flex items-center justify-between gap-2 rounded-md bg-muted/50 px-2 py-1">
   524→                          <span className="min-w-0 truncate">{i.name}</span>
   525→                          <span className="shrink-0 font-medium text-muted-foreground">
   526→                            {i.qty} {i.unit}
   527→                          </span>
   528→                        </li>
   529→                      ))}
   530→                    </ul>
   531→                    <div className="mt-auto flex gap-1.5 border-t pt-3">
   532→                      <Button variant="outline" size="sm" onClick={() => openTplEdit(t)} aria-label={`Edit ${t.name}`}>
   533→                        <Pencil className="size-3.5" aria-hidden="true" /> Edit
   534→                      </Button>
   535→                      <Button
   536→                        variant="outline"
   537→                        size="sm"
   538→                        className="ml-auto text-red-600 hover:bg-red-500/10 dark:text-red-400"
   539→                        onClick={() => setDeleteTpl(t)}

  1028→      <AlertDialog open={!!deleteBatch} onOpenChange={(o) => !o && setDeleteBatch(null)}>
