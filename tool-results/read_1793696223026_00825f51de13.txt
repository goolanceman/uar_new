   236→   236→function prefixFor(t: { kind?: string | null; codePrefix?: string | null }): string {
   237→   237→  const p = (t.codePrefix ?? "").trim();
   238→   238→  if (p) return p;
   239→   239→  const k = (t.kind ?? "").toUpperCase() as KitKind;
   240→   240→  return KIND_META[k]?.prefix ?? "UAR-RK";

   258→   258→function parseItems(json: string | null | undefined): TplItem[] {
   259→   259→  if (!json) return [];
   260→   260→  try {
   261→   261→    const parsed = JSON.parse(json) as { name?: string; qty?: number | string; unit?: string }[];
   262→   262→    if (!Array.isArray(parsed)) return [];
   263→   263→    return parsed.map((i) => ({
   264→   264→      name: String(i?.name ?? ""),
   265→   265→      qty: String(i?.qty ?? ""),
   266→   266→      unit: String(i?.unit ?? "pcs"),
   267→   267→    }));
   268→   268→  } catch {
   269→   269→    return [];
   270→   270→  }
   271→   271→}

   273→   273→function scanPct(b: Pick<KitBatch, "totalUnits" | "scannedUnits">): number {
   274→   274→  return b.totalUnits > 0
   275→   275→    ? Math.min(100, Math.round(((b.scannedUnits ?? 0) / b.totalUnits) * 100))
   276→   276→    : 0;
   277→   277→}
