   117→   117→function parseList(s?: string | null): string[] {
   118→   118→  if (!s) return [];
   119→   119→  try {
   120→   120→    const parsed = JSON.parse(s);
   121→   121→    if (Array.isArray(parsed)) return parsed.filter((x) => typeof x === "string" && x);
   122→   122→  } catch {
   123→   123→    // plain comma-separated string
   124→   124→  }
   125→   125→  return s.split(",").map((x) => x.trim()).filter(Boolean);
   126→   126→}
   127→   127→
   128→   128→function parseTempCreds(s?: string | null): TempCreds | null {
   129→   129→  if (!s) return null;
   130→   130→  try {
   131→   131→    const parsed = JSON.parse(s) as TempCreds;
   132→   132→    if (parsed && (parsed.email || parsed.password)) return parsed;
   133→   133→  } catch {
   134→   134→    // not JSON
   135→   135→  }
   136→   136→  return null;
   137→   137→}

   139→   139→function appStatusTone(status: string): string {
   140→   140→  switch (status) {
   141→   141→    case "APPROVED":
   142→   142→    case "ACTIVE":
   143→   143→      return "bg-emerald-500/10 text-emerald-700 dark:text-emerald-400";
   144→   144→    case "PENDING":

   146→   146→    case "REJECTED":

   149→   149→    default:
   150→   150→      return "bg-muted text-muted-foreground";
   151→   151→  }
