   399→   399→/** Financial-year window (Apr → Mar) as ISO strings for API filters. */
   400→   400→function fyRange(): { from: string; to: string } {
   401→   401→  const now = new Date();
   402→   402→  const y = now.getMonth() + 1 >= 4 ? now.getFullYear() : now.getFullYear() - 1;
   403→   403→  return { from: `${y}-04-01T00:00:00`, to: `${y + 1}-03-31T23:59:59.999` };
   404→   404→}

   406→   406→function rangeToParams(range: DateRange): { from?: string; to?: string } {
   407→   407→  if (range === "all") return {};
   408→   408→  if (range === "fy") return fyRange();
   409→   409→  const days = range === "30" ? 30 : 90;
   410→   410→  const from = new Date(Date.now() - days * 86_400_000);
   411→   411→  return {
   412→   412→    from: `${from.getFullYear()}-${pad2(from.getMonth() + 1)}-${pad2(from.getDate())}T00:00:00`,
   413→   413→  };
   414→   414→}

   416→   416→interface ParsedItem {
   417→   417→  name?: string;
   418→   418→  qty?: number | string;
   419→   419→  unit?: string;
   420→   420→}
   421→   421→
   422→   422→function parseItems(json: string | null | undefined): ParsedItem[] {
   423→   423→  if (!json) return [];
   424→   424→  try {
   425→   425→    const parsed = JSON.parse(json);
   426→   426→    return Array.isArray(parsed) ? (parsed as ParsedItem[]) : [];
   427→   427→  } catch {
   428→   428→    return [];
   429→   429→  }
   430→   430→}
   431→   431→
   432→   432→function itemsSummary(json: string | null | undefined): string {
   433→   433→  const items = parseItems(json);
   434→   434→  if (!items.length) return "—";
   435→   435→  return items
   436→   436→    .map((i) => {
   437→   437→      const name = String(i?.name ?? i ?? "").trim();
   438→   438→      const qty = i?.qty;
   439→   439→      const unit = i?.unit ? ` ${i.unit}` : "";
   440→   440→      return `${name} ×${qty ?? 1}${unit}`;
   441→   441→    })
   442→   442→    .join(", ");
   443→   443→}
