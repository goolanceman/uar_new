    48→  DialogFooter,
    49→  DialogHeader,

    51→} from "@/components/ui/dialog";

    60→  AlertDialogTitle,

    72→
    73→interface Beneficiary {
    74→  id: string;
    75→  name: string;
    76→  phone?: string | null;
    77→  email?: string | null;
    78→  address?: string | null;
    79→  city?: string | null;
    80→  state?: string | null;
    81→  pincode?: string | null;
    82→  category: string;
    83→  urgencyLevel: string;
    84→  familyMembers: number;
    85→  monthlyIncome?: number | null;
    86→  monthlyNeed?: number | null;
    87→  needs?: string | null;
    88→  documents?: string | null;
    89→  notes?: string | null;
    90→  isActive: boolean;

    92→  createdAt?: string;

    96→  id: string;
    97→  type: string;
    98→  beneficiaryId?: string | null;

   100→  value?: number | null;
   101→  location?: string | null;
   102→  distributedAt: string;
   103→  beneficiary?: { name: string } | null;
   104→  volunteer?: { user?: { name: string } | null } | null;
   105→}

   108→  name: string;
   109→  phone: string;
   110→  email: string;
   111→  address: string;
   112→  city: string;
   113→  state: string;
   114→  pincode: string;
   115→  category: string;
   116→  urgencyLevel: string;
   117→  familyMembers: string;
   118→  monthlyIncome: string;
   119→  monthlyNeed: string;
   120→  needs: string;
   121→  documents: string;
   122→  notes: string;
   123→}

   128→  CRITICAL: "border-red-500/30 bg-red-500/10 text-red-700 dark:text-red-400 animate-pulse",
   129→  HIGH: "border-orange-500/30 bg-orange-500/10 text-orange-700 dark:text-orange-400",
   130→  MEDIUM: "border-amber-500/30 bg-amber-500/10 text-amber-700 dark:text-amber-400",
   131→  LOW: "border-emerald-500/30 bg-emerald-500/10 text-emerald-700 dark:text-emerald-400",
   132→};
   133→
   134→const CATEGORIES = [
   135→  "WIDOW",
   136→  "ORPHAN",
   137→  "STUDENT",
   138→  "ELDERLY",
   139→  "DISABLED",
   140→  "SINGLE_MOTHER",
   141→  "POOR_FAMILY",
   142→  "PATIENT",
   143→  "DAILY_WAGE",
   144→  "OTHER",
   145→];
   146→
   147→const URGENCIES = ["CRITICAL", "HIGH", "MEDIUM", "LOW"];
   148→
   149→const EMPTY_FORM: BenForm = {
   150→  name: "",
   151→  phone: "",
   152→  email: "",
   153→  address: "",
   154→  city: "",
   155→  state: "Karnataka",
   156→  pincode: "",
   157→  category: "OTHER",
   158→  urgencyLevel: "MEDIUM",
   159→  familyMembers: "1",
   160→  monthlyIncome: "",
   161→  monthlyNeed: "",
   162→  needs: "",
   163→  documents: "",
   164→  notes: "",
   165→};

   167→/* ---------------------------------- view ----------------------------------- */
   168→
   169→export default function BeneficiariesView() {
   170→  const [rows, setRows] = useState<Beneficiary[]>([]);
   171→  const [loading, setLoading] = useState(true);
   172→  const [categoryFilter, setCategoryFilter] = useState("ALL");
   173→  const [urgencyFilter, setUrgencyFilter] = useState("ALL");
   174→  const [search, setSearch] = useState("");
   175→
   176→  const [sheet, setSheet] = useState<Beneficiary | null>(null);
   177→  const [sheetDistributions, setSheetDistributions] = useState<DistributionRow[]>([]);
   178→  const [sheetDistributionsLoading, setSheetDistributionsLoading] = useState(false);
   179→
   180→  const [dialogOpen, setDialogOpen] = useState(false);
   181→  const [editing, setEditing] = useState<Beneficiary | null>(null);
   182→  const [form, setForm] = useState<BenForm>(EMPTY_FORM);
   183→  const [saving, setSaving] = useState(false);
   184→  const [deleteTarget, setDeleteTarget] = useState<Beneficiary | null>(null);
   185→  const [toggling, setToggling] = useState<string | null>(null);
   186→
   187→  const load = useCallback(async () => {
   188→    try {
   189→      const data = await get<Beneficiary[]>("/api/beneficiaries");
   190→      setRows(Array.isArray(data) ? data : []);
   191→    } catch (e) {
   192→      toast.error(e instanceof Error ? e.message : "Failed to load beneficiaries");
   193→    } finally {
   194→      setLoading(false);
   195→    }
   196→  }, []);
   197→
   198→  useEffect(() => {
   199→    load();
   200→  }, [load]);
   201→
   202→  const filtered = useMemo(() => {
   203→    const q = search.trim().toLowerCase();
   204→    return rows.filter((b) => {
   205→      if (categoryFilter !== "ALL" && b.category !== categoryFilter) return false;
   206→      if (urgencyFilter !== "ALL" && b.urgencyLevel !== urgencyFilter) return false;
   207→      if (q) {
   208→        const hay = `${b.name} ${b.city ?? ""} ${b.phone ?? ""}`.toLowerCase();
   209→        if (!hay.includes(q)) return false;
   210→      }
   211→      return true;
   212→    });
   213→  }, [rows, categoryFilter, urgencyFilter, search]);
   214→
   215→  const stats = useMemo(
   216→    () => ({
   217→      total: rows.length,
   218→      critical: rows.filter((b) => b.urgencyLevel === "CRITICAL").length,
   219→      active: rows.filter((b) => b.isActive).length,
   220→      monthlyNeed: rows.reduce((s, b) => s + (b.monthlyNeed ?? 0), 0),
   221→    }),
   222→    [rows]
   223→  );
   224→
   225→  /* open sheet + load its distributions */
   226→  const openSheet = (b: Beneficiary) => {
   227→    setSheet(b);
   228→    setSheetDistributionsLoading(true);
   229→    get<DistributionRow[]>("/api/distributions?limit=200")
   230→      .then((data) =>
   231→        setSheetDistributions(
   232→          (Array.isArray(data) ? data : []).filter((d) => d.beneficiaryId === b.id)
   233→        )
   234→      )
   235→      .catch(() => setSheetDistributions([]))
   236→      .finally(() => setSheetDistributionsLoading(false));
   237→  };
   238→
   239→  /* ------------------------------ form actions ----------------------------- */
   240→
   241→  const openCreate = () => {
   242→    setEditing(null);
   243→    setForm(EMPTY_FORM);
   244→    setDialogOpen(true);
   245→  };
   246→
   247→  const openEdit = (b: Beneficiary) => {
   248→    setEditing(b);
   249→    setForm({
   250→      name: b.name ?? "",
   251→      phone: b.phone ?? "",
   252→      email: b.email ?? "",
   253→      address: b.address ?? "",
   254→      city: b.city ?? "",
   255→      state: b.state ?? "",
   256→      pincode: b.pincode ?? "",
   257→      category: b.category ?? "OTHER",
   258→      urgencyLevel: b.urgencyLevel ?? "MEDIUM",
   259→      familyMembers: String(b.familyMembers ?? 1),
   260→      monthlyIncome: b.monthlyIncome != null ? String(b.monthlyIncome) : "",
   261→      monthlyNeed: b.monthlyNeed != null ? String(b.monthlyNeed) : "",
   262→      needs: b.needs ?? "",
   263→      documents: b.documents ?? "",
   264→      notes: b.notes ?? "",
   265→    });
   266→    setDialogOpen(true);
   267→  };
   268→
   269→  const validate = (): string | null => {
   270→    if (!form.name.trim()) return "Name is required";
   271→    if (form.phone && !/^\d{10}$/.test(form.phone.replace(/\D/g, ""))) return "Phone must be a 10-digit number";
   272→    if (form.email && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(form.email)) return "Email looks invalid";
   273→    if (form.pincode && !/^\d{6}$/.test(form.pincode)) return "Pincode must be exactly 6 digits";
   274→    const fam = Number(form.familyMembers);
   275→    if (!Number.isFinite(fam) || fam < 0) return "Family members must be 0 or more";
   276→    const inc = Number(form.monthlyIncome);
   277→    if (form.monthlyIncome && (!Number.isFinite(inc) || inc < 0)) return "Monthly income must be 0 or more";
   278→    const need = Number(form.monthlyNeed);
   279→    if (form.monthlyNeed && (!Number.isFinite(need) || need < 0)) return "Monthly need must be 0 or more";
   280→    return null;
   281→  };
   282→
   283→  const save = async () => {
   284→    const problem = validate();
   285→    if (problem) {
   286→      toast.error(problem);
   287→      return;
   288→    }
   289→    setSaving(true);
   290→    try {
   291→      const body = {
   292→        name: form.name.trim(),
   293→        phone: form.phone.trim() || undefined,
   294→        email: form.email.trim() || undefined,
   295→        address: form.address.trim() || undefined,
   296→        city: form.city.trim() || undefined,
   297→        state: form.state.trim() || undefined,
   298→        pincode: form.pincode.trim() || undefined,
   299→        category: form.category,
   300→        urgencyLevel: form.urgencyLevel,
   301→        familyMembers: Math.trunc(Number(form.familyMembers) || 1),
   302→        monthlyIncome: form.monthlyIncome === "" ? undefined : Number(form.monthlyIncome),
   303→        monthlyNeed: form.monthlyNeed === "" ? undefined : Number(form.monthlyNeed),
   304→        needs: form.needs.trim() || undefined,
   305→        documents: form.documents.trim() || undefined,
   306→        notes: form.notes.trim() || undefined,
   307→      };
   308→      if (editing) {
   309→        await patch(`/api/beneficiaries/${editing.id}`, body);
   310→        toast.success(`${body.name} updated`);
   311→      } else {
   312→        await post("/api/beneficiaries", body);
   313→        toast.success(`${body.name} onboarded`);
   314→      }
   315→      setDialogOpen(false);
   316→      setSheet(null);
   317→      load();
   318→    } catch (e) {
   319→      toast.error(e instanceof Error ? e.message : "Save failed");
   320→    } finally {
   321→      setSaving(false);
   322→    }
   323→  };
   324→
   325→  const toggleActive = async (b: Beneficiary) => {
   326→    setToggling(b.id);
   327→    try {
   328→      await patch(`/api/beneficiaries/${b.id}`, { isActive: !b.isActive });
   329→      toast.success(`${b.name} ${b.isActive ? "deactivated" : "activated"}`);
   330→      load();
   331→    } catch (e) {
   332→      toast.error(e instanceof Error ? e.message : "Update failed");
   333→    } finally {
   334→      setToggling(null);
   335→    }
   336→  };
   337→
   338→  const remove = async () => {
   339→    if (!deleteTarget) return;
   340→    try {
   341→      await del(`/api/beneficiaries/${deleteTarget.id}`);
   342→      toast.success(`${deleteTarget.name} removed`);
   343→      setDeleteTarget(null);
   344→      setSheet(null);
   345→      load();
   346→    } catch (e) {
   347→      toast.error(e instanceof Error ? e.message : "Delete failed");
   348→    }
   349→  };
   350→
   351→  const rowActions = (b: Beneficiary) => (
   352→    <>
   353→      <Button variant="outline" size="sm" onClick={() => openEdit(b)} aria-label={`Edit ${b.name}`}>
   354→        <Pencil className="size-3.5" aria-hidden="true" />
   355→      </Button>
   356→      <Button
   357→        variant="outline"
   358→        size="sm"
   359→        className="text-red-600 hover:bg-red-500/10 dark:text-red-400"
   360→        onClick={() => setDeleteTarget(b)}
   361→        aria-label={`Delete ${b.name}`}
   362→      >
   363→        <Trash2 className="size-3.5" aria-hidden="true" />
   364→      </Button>
   365→    </>
   366→  );
   367→
   368→  /* --------------------------------- render -------------------------------- */
   369→
   370→  return (
   371→    <div className="space-y-6">
   372→      <PageHeader
   373→        title="Beneficiaries"
   374→        description="Families and individuals onboarded for UAR support programmes."
   375→        icon={<HeartHandshake className="size-5" aria-hidden="true" />}
   376→        actions={
   377→          <>
   378→            <Button variant="outline" onClick={load} aria-label="Refresh beneficiaries">
   379→              <RefreshCw className="size-4" aria-hidden="true" /> Refresh
   380→            </Button>
   381→            <Button onClick={openCreate}>
   382→              <Plus className="size-4" aria-hidden="true" /> Onboard
   383→            </Button>
   384→          </>
   385→        }
   386→      />
   387→
   388→      {/* stats */}
   389→      <div className="grid gap-4 sm:grid-cols-2 xl:grid-cols-4">
   390→        <StatCard label="On register" value={stats.total} accent="emerald" icon={<HeartHandshake className="size-5" aria-hidden="true" />} loading={loading} />
   391→        <StatCard label="Active" value={stats.active} accent="teal" icon={<UserRound className="size-5" aria-hidden="true" />} loading={loading} />
   392→        <StatCard label="Critical cases" value={stats.critical} accent="red" icon={<HeartHandshake className="size-5" aria-hidden="true" />} loading={loading} hint="Need urgent support" />
   393→        <StatCard label="Monthly need" value={formatINR(stats.monthlyNeed)} accent="amber" icon={<HeartHandshake className="size-5" aria-hidden="true" />} loading={loading} hint="Across all beneficiaries" />
   394→      </div>
   395→
   396→      {/* filters */}
   397→      <div className="flex flex-col gap-2 lg:flex-row lg:items-center">
   398→        <Select value={categoryFilter} onValueChange={setCategoryFilter}>
   399→          <SelectTrigger className="w-full lg:w-44" aria-label="Filter by category">
   400→            <SelectValue />
   401→          </SelectTrigger>
   402→          <SelectContent>
   403→            <SelectItem value="ALL">All categories</SelectItem>
   404→            {CATEGORIES.map((c) => (
   405→              <SelectItem key={c} value={c}>
   406→                {titleCase(c)}
   407→              </SelectItem>
   408→            ))}
   409→          </SelectContent>
   410→        </Select>
   411→        <Select value={urgencyFilter} onValueChange={setUrgencyFilter}>
   412→          <SelectTrigger className="w-full lg:w-40" aria-label="Filter by urgency">
   413→            <SelectValue />
   414→          </SelectTrigger>
   415→          <SelectContent>
   416→            <SelectItem value="ALL">All urgency</SelectItem>
   417→            {URGENCIES.map((u) => (
   418→              <SelectItem key={u} value={u}>
   419→                {titleCase(u)}
   420→              </SelectItem>
   421→            ))}
   422→          </SelectContent>
   423→        </Select>
   424→        <Input
   425→          value={search}
   426→          onChange={(e) => setSearch(e.target.value)}
   427→          placeholder="Search name, city or phone…"
   428→          aria-label="Search beneficiaries"
   429→          className="w-full lg:max-w-xs"
   430→        />
   431→        <p className="text-xs text-muted-foreground lg:ml-auto">
   432→          {filtered.length} of {rows.length} beneficiaries
   433→        </p>
   434→      </div>
   435→
   436→      {/* list */}
   437→      {loading ? (
   438→        <div className="space-y-2" aria-busy="true">
   439→          {[0, 1, 2, 3, 4].map((i) => (
   440→            <Skeleton key={i} className="h-14 rounded-lg" />
   441→          ))}
   442→        </div>
   443→      ) : filtered.length === 0 ? (
   444→        <EmptyState
   445→          icon={<HeartHandshake className="size-8" aria-hidden="true" />}
   446→          title={rows.length === 0 ? "No beneficiaries yet" : "No matches"}
   447→          hint={
   448→            rows.length === 0
   449→              ? "Onboard your first beneficiary to start tracking distributions."
   450→              : "Try clearing filters or searching a different name."
   451→          }
   452→        >
   453→          {rows.length === 0 && (
   454→            <Button onClick={openCreate}>
   455→              <Plus className="size-4" aria-hidden="true" /> Onboard
   456→            </Button>
   457→          )}
   458→        </EmptyState>
   459→      ) : (
   460→        <>
   461→          {/* mobile cards */}
   462→          <ul className="space-y-2 sm:hidden">
   463→            {filtered.map((b) => (
   464→              <li key={b.id} className="rounded-lg border p-3">
   465→                <button type="button" className="w-full text-left" onClick={() => openSheet(b)} aria-label={`Open ${b.name}`}>
   466→                  <div className="flex items-center gap-2">
   467→                    <span className="truncate font-semibold">{b.name}</span>
   468→                    <Badge variant="outline" className={cn("ml-auto shrink-0", URGENCY[b.urgencyLevel] ?? URGENCY.MEDIUM)}>
   469→                      {titleCase(b.urgencyLevel)}
   470→                    </Badge>
   471→                  </div>
   472→                  <p className="mt-0.5 text-xs text-muted-foreground">
   473→                    {titleCase(b.category)} · {b.city || "—"} · {b.familyMembers} in family
   474→                  </p>
   475→                </button>
   476→                <div className="mt-2 flex items-center gap-1.5">
   477→                  {rowActions(b)}
   478→                  <div className="ml-auto flex items-center gap-2">
   479→                    <span className="text-xs text-muted-foreground">{b.isActive ? "Active" : "Inactive"}</span>
   480→                    <Switch
   481→                      checked={b.isActive}
   482→                      disabled={toggling === b.id}
   483→                      onCheckedChange={() => toggleActive(b)}
   484→                      aria-label={`${b.isActive ? "Deactivate" : "Activate"} ${b.name}`}
   485→                    />
   486→                  </div>
   487→                </div>
   488→              </li>
   489→            ))}
   490→          </ul>
   491→          {/* desktop table */}
   492→          <div className="hidden overflow-x-auto rounded-lg border sm:block">
   493→            <Table>
   494→              <TableHeader>
   495→                <TableRow>
   496→                  <TableHead>Name</TableHead>
   497→                  <TableHead>Category</TableHead>
   498→                  <TableHead>Urgency</TableHead>
   499→                  <TableHead>City</TableHead>
   500→                  <TableHead className="text-right">Family</TableHead>
   501→                  <TableHead>Active</TableHead>
   502→                  <TableHead className="text-right">Actions</TableHead>
   503→                </TableRow>
   504→              </TableHeader>
   505→              <TableBody>
   506→                {filtered.map((b) => (
   507→                  <TableRow key={b.id}>
   508→                    <TableCell>
   509→                      <button type="button" className="text-left font-medium hover:text-primary" onClick={() => openSheet(b)} aria-label={`Open ${b.name}`}>
   510→                        {b.name}
   511→                      </button>
   512→                      {b.phone ? <span className="block text-xs text-muted-foreground">{b.phone}</span> : null}
   513→                    </TableCell>
   514→                    <TableCell>{titleCase(b.category)}</TableCell>
   515→                    <TableCell>
   516→                      <Badge variant="outline" className={cn(URGENCY[b.urgencyLevel] ?? URGENCY.MEDIUM)}>
   517→                        {titleCase(b.urgencyLevel)}
   518→                      </Badge>
   519→                    </TableCell>
   520→                    <TableCell>{b.city || "—"}</TableCell>
   521→                    <TableCell className="text-right">{b.familyMembers}</TableCell>
   522→                    <TableCell>
   523→                      <Switch
   524→                        checked={b.isActive}
   525→                        disabled={toggling === b.id}
   526→                        onCheckedChange={() => toggleActive(b)}
   527→                        aria-label={`${b.isActive ? "Deactivate" : "Activate"} ${b.name}`}
   528→                      />
   529→                    </TableCell>
   530→                    <TableCell className="text-right">
   531→                      <div className="flex justify-end gap-1.5">{rowActions(b)}</div>
   532→                    </TableCell>
   533→                  </TableRow>
   534→                ))}
   535→              </TableBody>
   536→            </Table>
   537→          </div>
   538→        </>
   539→      )}
   540→
   541→      {/* ============================== DETAIL SHEET ============================= */}
   542→      <Sheet open={!!sheet} onOpenChange={(o) => !o && setSheet(null)}>
   543→        <SheetContent side="right" className="w-full overflow-y-auto p-0 sm:max-w-md">
   544→          {sheet && (
   545→            <div className="space-y-5 p-5">
   546→              <SheetHeader className="p-0 text-left">
   547→                <SheetTitle className="flex items-center gap-2">
   548→                  {sheet.name}
   549→                  <Badge variant="outline" className={cn("ml-auto", URGENCY[sheet.urgencyLevel] ?? URGENCY.MEDIUM)}>
   550→                    {titleCase(sheet.urgencyLevel)}
   551→                  </Badge>
   552→                </SheetTitle>
   553→                <SheetDescription>
   554→                  {titleCase(sheet.category)} · onboarded by {sheet.onboardedByName || "—"}
   555→                </SheetDescription>
   556→              </SheetHeader>
   557→
   558→              <div className="grid grid-cols-2 gap-3">
   559→                <DataField label="Phone" value={sheet.phone || "—"} icon={<Phone className="size-3.5" aria-hidden="true" />} mono />
   560→                <DataField label="Email" value={sheet.email || "—"} />

   571→              {sheet.address ? (

   577→              {sheet.needs ? (

   579→                  <p className="text-sm leading-relaxed text-muted-foreground">{sheet.needs}</p>

   582→
   583→              {sheet.notes ? (
   584→                <SectionCard title="Notes">
   585→                  <p className="text-sm leading-relaxed text-muted-foreground">{sheet.notes}</p>

   608→                            {d.location ? ` · ${d.location}` : ""}
   609→                          </p>
   610→                        </div>
   611→                        <span className="shrink-0 text-xs font-medium">
   612→                          {d.value != null ? formatINR(d.value) : ""}
   613→                        </span>
   614→                      </li>
   615→                    ))}
   616→                  </ul>
   617→                )}
   618→              </SectionCard>
   619→
   620→              <div className="flex flex-wrap items-center gap-2 border-t pt-4">
   621→                <Button variant="outline" size="sm" onClick={() => openEdit(sheet)}>
   622→                  <Pencil className="size-3.5" aria-hidden="true" /> Edit
   623→                </Button>
   624→                <div className="ml-auto flex items-center gap-2">
   625→                  <Label htmlFor="sheet-active" className="text-xs text-muted-foreground">
   626→                    {sheet.isActive ? "Active" : "Inactive"}
   627→                  </Label>
   628→                  <Switch
   629→                    id="sheet-active"
   630→                    checked={sheet.isActive}
   631→                    disabled={toggling === sheet.id}
   632→                    onCheckedChange={() => toggleActive(sheet)}
   633→                    aria-label={`${sheet.isActive ? "Deactivate" : "Activate"} ${sheet.name}`}
   634→                  />
   635→                </div>
   636→              </div>
   637→            </div>
   638→          )}
   639→        </SheetContent>
   640→      </Sheet>
   641→
   642→      {/* ============================ CREATE / EDIT ============================= */}
   643→      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
   644→        <DialogContent className="max-h-[90vh] overflow-y-auto sm:max-w-lg">
   645→          <DialogHeader>
   646→            <DialogTitle>{editing ? `Edit ${editing.name}` : "Onboard beneficiary"}</DialogTitle>
   647→            <DialogDescription>
   648→              All fields help volunteers plan distribution visits. Phone and pincode are validated.
   649→            </DialogDescription>
   650→          </DialogHeader>
   651→          <div className="grid gap-4">
   652→            <div className="grid gap-4 sm:grid-cols-2">
   653→              <div className="grid gap-2">
   654→                <Label htmlFor="ben-name">Name *</Label>
   655→                <Input
   656→                  id="ben-name"
   657→                  value={form.name}
   658→                  onChange={(e) => setForm({ ...form, name: e.target.value })}
   659→                  placeholder="Lakshmi Devi"
   660→                />
   661→              </div>

   666→                  inputMode="numeric"

   669→                  placeholder="9876543210"

   673→            <div className="grid gap-2">
   674→              <Label htmlFor="ben-email">Email</Label>
   675→              <Input
   676→                id="ben-email"
   677→                type="email"
   678→                value={form.email}
   679→                onChange={(e) => setForm({ ...form, email: e.target.value })}
   680→                placeholder="optional"
   681→              />
   682→            </div>
   683→            <div className="grid gap-2">
   684→              <Label htmlFor="ben-address">Address</Label>
   685→              <Textarea
   686→                id="ben-address"
   687→                rows={2}
   688→                value={form.address}
   689→                onChange={(e) => setForm({ ...form, address: e.target.value })}
   690→              />
   691→            </div>
   692→            <div className="grid gap-4 sm:grid-cols-3">
   693→              <div className="grid gap-2">
   694→                <Label htmlFor="ben-city">City</Label>
   695→                <Input id="ben-city" value={form.city} onChange={(e) => setForm({ ...form, city: e.target.value })} />
   696→              </div>
   697→              <div className="grid gap-2">
   698→                <Label htmlFor="ben-state">State</Label>

   700→              </div>
   701→              <div className="grid gap-2">
   702→                <Label htmlFor="ben-pin">Pincode (6 digits)</Label>
   703→                <Input
   704→                  id="ben-pin"
   705→                  inputMode="numeric"
   706→                  value={form.pincode}
   707→                  onChange={(e) => setForm({ ...form, pincode: e.target.value.replace(/[^\d]/g, "").slice(0, 6) })}
   708→                />
   709→              </div>
   710→            </div>
   711→            <div className="grid gap-4 sm:grid-cols-2">
   712→              <div className="grid gap-2">

   714→                <Select value={form.category} onValueChange={(v) => setForm({ ...form, category: v })}>

   719→                    {CATEGORIES.map((c) => (

   721→                        {titleCase(c)}
   722→                      </SelectItem>
   723→                    ))}
   724→                  </SelectContent>
   725→                </Select>
   726→              </div>
   727→              <div className="grid gap-2">
   728→                <Label>Urgency</Label>
   729→                <Select value={form.urgencyLevel} onValueChange={(v) => setForm({ ...form, urgencyLevel: v })}>
   730→                  <SelectTrigger aria-label="Urgency level">
   731→                    <SelectValue />
   732→                  </SelectTrigger>
   733→                  <SelectContent>
   734→                    {URGENCIES.map((u) => (
   735→                      <SelectItem key={u} value={u}>
   736→                        {titleCase(u)}
   737→                      </SelectItem>
   738→                    ))}
   739→                  </SelectContent>
   740→                </Select>

   751→                  onChange={(e) => setForm({ ...form, familyMembers: e.target.value })}
   752→                />
   753→              </div>
   754→              <div className="grid gap-2">
   755→                <Label htmlFor="ben-income">Monthly income (₹)</Label>
   756→                <Input
   757→                  id="ben-income"
   758→                  type="number"
   759→                  min={0}
   760→                  value={form.monthlyIncome}
   761→                  onChange={(e) => setForm({ ...form, monthlyIncome: e.target.value })}

   768→                  type="number"
   769→                  min={0}

   771→                  onChange={(e) => setForm({ ...form, monthlyNeed: e.target.value })}

   779→                rows={2}
