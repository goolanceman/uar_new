     1→"use client";
     2→
     3→import { useCallback, useEffect, useMemo, useState } from "react";
     4→import {
     5→  AlertTriangle,
     6→  Boxes,
     7→  CalendarClock,
     8→  Package,
     9→  PackagePlus,
    10→  Pencil,
    11→  Plus,
    12→  RefreshCw,
    13→  Trash2,
    14→} from "lucide-react";
    15→import { toast } from "sonner";
    16→import { del, get, patch, post } from "@/lib/api-client";
    17→import { labelOf, useMasters } from "@/hooks/use-masters";
    18→import { formatDate, formatINR, titleCase } from "@/lib/format";
    19→import { cn } from "@/lib/utils";
    20→import { PageHeader } from "@/components/shared/page-header";
    21→import { SectionCard } from "@/components/shared/section-card";
    22→import { StatCard } from "@/components/shared/stat-card";
    23→import { DataField } from "@/components/shared/data-field";
    24→import { EmptyState } from "@/components/shared/empty-state";
    25→import { Badge } from "@/components/ui/badge";
    26→import { Button } from "@/components/ui/button";
    27→import { Input } from "@/components/ui/input";
    28→import { Label } from "@/components/ui/label";
    29→import { Textarea } from "@/components/ui/textarea";
    30→import { Skeleton } from "@/components/ui/skeleton";
    31→import {
    32→  Select,
    33→  SelectContent,
    34→  SelectItem,
    35→  SelectTrigger,
    36→  SelectValue,
    37→} from "@/components/ui/select";
    38→import {
    39→  Sheet,
    40→  SheetContent,
    41→  SheetDescription,
    42→  SheetHeader,
    43→  SheetTitle,
    44→} from "@/components/ui/sheet";
    45→import {
    46→  Dialog,
    47→  DialogContent,
    48→  DialogDescription,
    49→  DialogFooter,
    50→  DialogHeader,
    51→  DialogTitle,
    52→} from "@/components/ui/dialog";
    53→import {
    54→  AlertDialog,
    55→  AlertDialogAction,
    56→  AlertDialogCancel,
    57→  AlertDialogContent,
    58→  AlertDialogDescription,
    59→  AlertDialogFooter,
    60→  AlertDialogHeader,
    61→  AlertDialogTitle,
    62→} from "@/components/ui/alert-dialog";
    63→import {
    64→  Table,
    65→  TableBody,
    66→  TableCell,
    67→  TableHead,
    68→  TableHeader,
    69→  TableRow,
    70→} from "@/components/ui/table";
    71→
    72→/* ---------------------------------- types ---------------------------------- */
    73→
    74→interface InventoryItem {
    75→  id: string;
    76→  name: string;
    77→  description?: string | null;
    78→  category: string;
    79→  unit: string;
    80→  quantity: number;
    81→  minStock: number;
    82→  pricePerUnit?: number | null;
    83→  location?: string | null;
    84→  expiryDate?: string | null;
    85→}
    86→
    87→interface ItemForm {
    88→  name: string;
    89→  description: string;
    90→  category: string;
    91→  unit: string;
    92→  quantity: string;
    93→  minStock: string;
    94→  pricePerUnit: string;
    95→  location: string;
    96→  expiryDate: string;
    97→}
    98→
    99→interface StockForm {
   100→  quantity: string;
   101→  pricePerUnit: string;
   102→  supplier: string;
   103→  invoiceNumber: string;
   104→}
   105→
   106→const EMPTY_ITEM_FORM: ItemForm = {
   107→  name: "",
   108→  description: "",
   109→  category: "FOOD_KIT",
   110→  unit: "pack",
   111→  quantity: "0",
   112→  minStock: "10",
   113→  pricePerUnit: "",
   114→  location: "",
   115→  expiryDate: "",
   116→};
   117→
   118→const EMPTY_STOCK_FORM: StockForm = {
   119→  quantity: "",
   120→  pricePerUnit: "",
   121→  supplier: "",
   122→  invoiceNumber: "",
   123→};
   124→
   125→function isLowStock(i: InventoryItem): boolean {
   126→  return i.quantity <= i.minStock;
   127→}
   128→
   129→function isExpiring(i: InventoryItem): boolean {
   130→  if (!i.expiryDate) return false;
   131→  const d = new Date(i.expiryDate);
   132→  if (Number.isNaN(d.getTime())) return false;
   133→  const in30 = Date.now() + 30 * 24 * 60 * 60 * 1000;
   134→  return d.getTime() <= in30;
   135→}
   136→
   137→/* ---------------------------------- view ----------------------------------- */
   138→
   139→export default function InventoryView() {
   140→  const [items, setItems] = useState<InventoryItem[]>([]);
   141→  const [loading, setLoading] = useState(true);
   142→
   143→  const [sheet, setSheet] = useState<InventoryItem | null>(null);
   144→  const [itemDialog, setItemDialog] = useState(false);
   145→  const [editing, setEditing] = useState<InventoryItem | null>(null);
   146→  const [form, setForm] = useState<ItemForm>(EMPTY_ITEM_FORM);
   147→  const [saving, setSaving] = useState(false);
   148→  const [deleteTarget, setDeleteTarget] = useState<InventoryItem | null>(null);
   149→
   150→  const [stockTarget, setStockTarget] = useState<InventoryItem | null>(null);
   151→  const [stockForm, setStockForm] = useState<StockForm>(EMPTY_STOCK_FORM);
   152→  const [addingStock, setAddingStock] = useState(false);
   153→
   154→  const categoryOptions = useMasters("INVENTORY_CATEGORY");
   155→  const unitOptions = useMasters("UNIT");
   156→
   157→  const load = useCallback(async () => {
   158→    try {
   159→      const data = await get<InventoryItem[]>("/api/inventory");
   160→      setItems(Array.isArray(data) ? data : []);
   161→    } catch (e) {
   162→      toast.error(e instanceof Error ? e.message : "Failed to load inventory");
   163→    } finally {
   164→      setLoading(false);
   165→    }
   166→  }, []);
   167→
   168→  useEffect(() => {
   169→    load();
   170→  }, [load]);
   171→
   172→  const stats = useMemo(
   173→    () => ({
   174→      total: items.length,
   175→      low: items.filter(isLowStock).length,
   176→      value: items.reduce((s, i) => s + (i.pricePerUnit ?? 0) * i.quantity, 0),
   177→    }),
   178→    [items]
   179→  );
   180→
   181→  /* ------------------------------ form actions ----------------------------- */
   182→
   183→  const openCreate = () => {
   184→    setEditing(null);
   185→    setForm({ ...EMPTY_ITEM_FORM, unit: unitOptions.options[0]?.value ?? "pack" });
   186→    setItemDialog(true);
   187→  };
   188→
   189→  const openEdit = (i: InventoryItem) => {
   190→    setEditing(i);
   191→    setForm({
   192→      name: i.name ?? "",
   193→      description: i.description ?? "",
   194→      category: i.category ?? "OTHER",
   195→      unit: i.unit ?? "pack",
   196→      quantity: String(i.quantity ?? 0),
   197→      minStock: String(i.minStock ?? 0),
   198→      pricePerUnit: i.pricePerUnit != null ? String(i.pricePerUnit) : "",
   199→      location: i.location ?? "",
   200→      expiryDate: i.expiryDate ? i.expiryDate.slice(0, 10) : "",
   201→    });
   202→    setItemDialog(true);
   203→  };
   204→
   205→  const validate = (): string | null => {
   206→    if (!form.name.trim()) return "Item name is required";
   207→    const qty = Number(form.quantity);
   208→    if (!Number.isFinite(qty) || qty < 0) return "Quantity must be 0 or more";
   209→    const min = Number(form.minStock);
   210→    if (!Number.isFinite(min) || min < 0) return "Minimum stock must be 0 or more";
   211→    if (form.pricePerUnit) {
   212→      const price = Number(form.pricePerUnit);
   213→      if (!Number.isFinite(price) || price < 0) return "Price per unit must be 0 or more";
   214→    }
   215→    return null;
   216→  };
   217→
   218→  const save = async () => {
   219→    const problem = validate();
   220→    if (problem) {
   221→      toast.error(problem);
   222→      return;
   223→    }
   224→    setSaving(true);
   225→    try {
   226→      const body = {
   227→        name: form.name.trim(),
   228→        description: form.description.trim() || undefined,
   229→        category: form.category,
   230→        unit: form.unit,
   231→        quantity: Math.trunc(Number(form.quantity)),
   232→        minStock: Math.trunc(Number(form.minStock)),
   233→        pricePerUnit: form.pricePerUnit === "" ? undefined : Number(form.pricePerUnit),
   234→        location: form.location.trim() || undefined,
   235→        expiryDate: form.expiryDate || undefined,
   236→      };
   237→      if (editing) {
   238→        await patch(`/api/inventory/${editing.id}`, body);
   239→        toast.success(`${body.name} updated`);
   240→      } else {
   241→        await post("/api/inventory", body);
   242→        toast.success(`${body.name} added to inventory`);
   243→      }
   244→      setItemDialog(false);
   245→      setSheet(null);
   246→      load();
   247→    } catch (e) {
   248→      toast.error(e instanceof Error ? e.message : "Save failed");
   249→    } finally {
   250→      setSaving(false);
   251→    }
   252→  };
   253→
   254→  const addStock = async () => {
   255→    if (!stockTarget) return;
   256→    const qty = Math.trunc(Number(stockForm.quantity));
   257→    if (!Number.isFinite(qty) || qty <= 0) {
   258→      toast.error("Quantity to add must be at least 1");
   259→      return;
   260→    }
   261→    setAddingStock(true);
   262→    try {
   263→      const body: Record<string, unknown> = { addQuantity: qty };
   264→      if (stockForm.pricePerUnit !== "") body.pricePerUnit = Number(stockForm.pricePerUnit);
   265→      await patch(`/api/inventory/${stockTarget.id}`, body);
   266→      toast.success(
   267→        `+${qty} ${stockTarget.unit} added to ${stockTarget.name}${
   268→          stockForm.supplier ? ` (supplier: ${stockForm.supplier.trim()})` : ""
   269→        }`
   270→      );
   271→      setStockTarget(null);
   272→      setStockForm(EMPTY_STOCK_FORM);
   273→      setSheet(null);
   274→      load();
   275→    } catch (e) {
   276→      toast.error(e instanceof Error ? e.message : "Add stock failed");
   277→    } finally {
   278→      setAddingStock(false);
   279→    }
   280→  };
   281→
   282→  const remove = async () => {
   283→    if (!deleteTarget) return;
   284→    try {
   285→      await del(`/api/inventory/${deleteTarget.id}`);
   286→      toast.success(`${deleteTarget.name} removed from inventory`);
   287→      setDeleteTarget(null);
   288→      setSheet(null);
   289→      load();
   290→    } catch (e) {
   291→      toast.error(e instanceof Error ? e.message : "Delete failed");
   292→    }
   293→  };
   294→
   295→  const stockCell = (i: InventoryItem) => (
   296→    <span className={cn("inline-flex items-center gap-1 font-medium", isLowStock(i) && "text-amber-700 dark:text-amber-400")}>
   297→      {isLowStock(i) && <AlertTriangle className="size-3.5" aria-hidden="true" />}
   298→      {i.quantity} {i.unit}
   299→    </span>
   300→  );
   301→
   302→  const expiryChip = (i: InventoryItem) =>
   303→    i.expiryDate ? (
   304→      <Badge
   305→        variant="outline"
   306→        className={cn(
   307→          isExpiring(i)
   308→            ? "border-red-500/30 bg-red-500/10 text-red-700 dark:text-red-400"
   309→            : "text-muted-foreground"
   310→        )}
   311→      >
   312→        <CalendarClock className="size-3" aria-hidden="true" />
   313→        {isExpiring(i) ? "Expires " : "Exp. "}
   314→        {formatDate(i.expiryDate)}
   315→      </Badge>
   316→    ) : (
   317→      <span className="text-xs text-muted-foreground">—</span>
   318→    );
   319→
   320→  const rowActions = (i: InventoryItem) => (
   321→    <>
   322→      <Button variant="outline" size="sm" onClick={() => setStockTarget(i)} aria-label={`Add stock to ${i.name}`}>
   323→        <PackagePlus className="size-3.5" aria-hidden="true" />
   324→      </Button>
   325→      <Button variant="outline" size="sm" onClick={() => openEdit(i)} aria-label={`Edit ${i.name}`}>
   326→        <Pencil className="size-3.5" aria-hidden="true" />
   327→      </Button>
   328→      <Button
   329→        variant="outline"
   330→        size="sm"
   331→        className="text-red-600 hover:bg-red-500/10 dark:text-red-400"
   332→        onClick={() => setDeleteTarget(i)}
   333→        aria-label={`Delete ${i.name}`}
   334→      >
   335→        <Trash2 className="size-3.5" aria-hidden="true" />
   336→      </Button>
   337→    </>
   338→  );
   339→
   340→  /* --------------------------------- render -------------------------------- */
   341→
   342→  return (
   343→    <div className="space-y-6">
   344→      <PageHeader
   345→        title="Inventory"
   346→        description="Stock of ration supplies, sanitary pads, medicine and equipment."
   347→        icon={<Boxes className="size-5" aria-hidden="true" />}
   348→        actions={
   349→          <>
   350→            <Button variant="outline" onClick={load} aria-label="Refresh inventory">
   351→              <RefreshCw className="size-4" aria-hidden="true" /> Refresh
   352→            </Button>
   353→            <Button onClick={openCreate}>
   354→              <Plus className="size-4" aria-hidden="true" /> New item
   355→            </Button>
   356→          </>
   357→        }
   358→      />
   359→
   360→      {/* stats */}
   361→      <div className="grid gap-4 sm:grid-cols-2 xl:grid-cols-4">
   362→        <StatCard label="Total items" value={stats.total} accent="emerald" icon={<Boxes className="size-5" aria-hidden="true" />} loading={loading} />
   363→        <StatCard
   364→          label="Low stock"
   365→          value={stats.low}
   366→          accent="amber"
   367→          icon={<AlertTriangle className="size-5" aria-hidden="true" />}
   368→          loading={loading}
   369→          hint="At or below minimum"
   370→        />
   371→        <StatCard label="Stock value" value={formatINR(stats.value)} accent="teal" icon={<Package className="size-5" aria-hidden="true" />} loading={loading} hint="Qty × unit price" />
   372→        <StatCard
   373→          label="Expiring ≤30 days"
   374→          value={items.filter(isExpiring).length}
   375→          accent="red"
   376→          icon={<CalendarClock className="size-5" aria-hidden="true" />}
   377→          loading={loading}
   378→        />
   379→      </div>
   380→
   381→      {/* table */}
   382→      {loading ? (
   383→        <div className="space-y-2" aria-busy="true">
   384→          {[0, 1, 2, 3, 4].map((i) => (
   385→            <Skeleton key={i} className="h-12 rounded-lg" />
   386→          ))}
   387→        </div>
   388→      ) : items.length === 0 ? (
   389→        <EmptyState
   390→          icon={<Boxes className="size-8" aria-hidden="true" />}
   391→          title="Inventory is empty"
   392→          hint="Add the first stock item — ration kits, pads, medicine or equipment."
   393→        >
   394→          <Button onClick={openCreate}>
   395→            <Plus className="size-4" aria-hidden="true" /> New item
   396→          </Button>
   397→        </EmptyState>
   398→      ) : (
   399→        <>
   400→          {/* mobile cards */}
   401→          <ul className="space-y-2 sm:hidden">
   402→            {items.map((i) => (
   403→              <li
   404→                key={i.id}
   405→                className={cn("rounded-lg border p-3", isLowStock(i) && "border-amber-500/40 bg-amber-500/5")}
   406→              >
   407→                <button type="button" className="w-full text-left" onClick={() => setSheet(i)} aria-label={`Open ${i.name}`}>
   408→                  <div className="flex items-center gap-2">
   409→                    <span className="truncate font-semibold">{i.name}</span>
   410→                    {isLowStock(i) && <AlertTriangle className="size-4 shrink-0 text-amber-600 dark:text-amber-400" aria-hidden="true" />}
   411→                  </div>
   412→                  <p className="mt-0.5 text-xs text-muted-foreground">
   413→                    {labelOf(categoryOptions.options, i.category)} · {i.location || "—"}
   414→                  </p>
   415→                  <div className="mt-1.5 flex flex-wrap items-center gap-1.5">
   416→                    {stockCell(i)}
   417→                    <span className="text-xs text-muted-foreground">min {i.minStock}</span>
   418→                    {expiryChip(i)}
   419→                  </div>
   420→                </button>
   421→                <div className="mt-2 flex gap-1.5">{rowActions(i)}</div>
   422→              </li>
   423→            ))}
   424→          </ul>
   425→          {/* desktop table */}
   426→          <div className="hidden overflow-x-auto rounded-lg border sm:block">
   427→            <Table>
   428→              <TableHeader>
   429→                <TableRow>
   430→                  <TableHead>Item</TableHead>
   431→                  <TableHead>Category</TableHead>
   432→                  <TableHead className="text-right">Stock</TableHead>
   433→                  <TableHead className="text-right">Min</TableHead>
   434→                  <TableHead className="text-right">Unit price</TableHead>
   435→                  <TableHead>Location</TableHead>
   436→                  <TableHead>Expiry</TableHead>
   437→                  <TableHead className="text-right">Actions</TableHead>
   438→                </TableRow>
   439→              </TableHeader>
   440→              <TableBody>
   441→                {items.map((i) => (
   442→                  <TableRow key={i.id} className={cn(isLowStock(i) && "bg-amber-500/5")}>
   443→                    <TableCell>
   444→                      <button type="button" className="text-left font-medium hover:text-primary" onClick={() => setSheet(i)} aria-label={`Open ${i.name}`}>
   445→                        {i.name}
   446→                      </button>
   447→                      {isLowStock(i) ? (
   448→                        <span className="block text-xs text-amber-700 dark:text-amber-400">Low stock — reorder</span>
   449→                      ) : null}
   450→                    </TableCell>
   451→                    <TableCell>{labelOf(categoryOptions.options, i.category)}</TableCell>
   452→                    <TableCell className="text-right">{stockCell(i)}</TableCell>
   453→                    <TableCell className="text-right text-muted-foreground">{i.minStock}</TableCell>
   454→                    <TableCell className="text-right">{i.pricePerUnit != null ? formatINR(i.pricePerUnit) : "—"}</TableCell>
   455→                    <TableCell className="max-w-36 truncate">{i.location || "—"}</TableCell>
   456→                    <TableCell>{expiryChip(i)}</TableCell>
   457→                    <TableCell className="text-right">
   458→                      <div className="flex justify-end gap-1.5">{rowActions(i)}</div>
   459→                    </TableCell>
   460→                  </TableRow>
   461→                ))}
   462→              </TableBody>
   463→            </Table>
   464→          </div>
   465→        </>
   466→      )}
   467→
   468→      {/* ============================== DETAIL SHEET ============================= */}
   469→      <Sheet open={!!sheet} onOpenChange={(o) => !o && setSheet(null)}>
   470→        <SheetContent side="right" className="w-full overflow-y-auto p-0 sm:max-w-md">
   471→          {sheet && (
   472→            <div className="space-y-5 p-5">
   473→              <SheetHeader className="p-0 text-left">
   474→                <SheetTitle>{sheet.name}</SheetTitle>
   475→                <SheetDescription>{labelOf(categoryOptions.options, sheet.category)}</SheetDescription>
   476→              </SheetHeader>
   477→
   478→              {isLowStock(sheet) && (
   479→                <div className="flex items-center gap-2 rounded-xl border border-amber-500/40 bg-amber-500/10 px-3 py-2.5 text-sm font-medium text-amber-700 dark:text-amber-400">
   480→                  <AlertTriangle className="size-4" aria-hidden="true" />
   481→                  At or below minimum stock ({sheet.minStock} {sheet.unit}) — reorder soon.
   482→                </div>
   483→              )}
   484→
   485→              <div className="grid grid-cols-2 gap-3">
   486→                <DataField label="In stock" value={`${sheet.quantity} ${sheet.unit}`} />
   487→                <DataField label="Minimum stock" value={`${sheet.minStock} ${sheet.unit}`} />
   488→                <DataField label="Price per unit" value={sheet.pricePerUnit != null ? formatINR(sheet.pricePerUnit) : "—"} />
   489→                <DataField label="Stock value" value={formatINR((sheet.pricePerUnit ?? 0) * sheet.quantity)} />
   490→                <DataField label="Location" value={sheet.location || "—"} />
   491→                <DataField label="Expiry" value={sheet.expiryDate ? formatDate(sheet.expiryDate) : "No expiry"} />
   492→              </div>
   493→
   494→              {sheet.description ? (
   495→                <SectionCard title="Description">
   496→                  <p className="text-sm leading-relaxed text-muted-foreground">{sheet.description}</p>
   497→                </SectionCard>
   498→              ) : null}
   499→
   500→              <div className="rounded-lg border border-dashed bg-muted/40 p-3 text-xs text-muted-foreground">
   501→                Purchase history is not exposed by the inventory API yet — stock movements are
   502→                applied directly with “Add stock”.
   503→              </div>
   504→
   505→              <div className="flex flex-wrap gap-2 border-t pt-4">
   506→                <Button size="sm" onClick={() => setStockTarget(sheet)}>
   507→                  <PackagePlus className="size-3.5" aria-hidden="true" /> Add stock
   508→                </Button>
   509→                <Button variant="outline" size="sm" onClick={() => openEdit(sheet)}>
   510→                  <Pencil className="size-3.5" aria-hidden="true" /> Edit
   511→                </Button>
   512→                <Button
   513→                  variant="outline"
   514→                  size="sm"
   515→                  className="ml-auto text-red-600 hover:bg-red-500/10 dark:text-red-400"
   516→                  onClick={() => setDeleteTarget(sheet)}
   517→                >
   518→                  <Trash2 className="size-3.5" aria-hidden="true" /> Delete
   519→                </Button>
   520→              </div>
   521→            </div>
   522→          )}
   523→        </SheetContent>
   524→      </Sheet>
   525→
   526→      {/* ============================== ADD STOCK =============================== */}
   527→      <Dialog open={!!stockTarget} onOpenChange={(o) => !o && setStockTarget(null)}>
   528→        <DialogContent className="sm:max-w-md">
   529→          <DialogHeader>
   530→            <DialogTitle>Add stock — {stockTarget?.name}</DialogTitle>
   531→            <DialogDescription>
   532→              Currently {stockTarget?.quantity} {stockTarget?.unit} in stock. The quantity is added
   533→              immediately; supplier and invoice are noted in the activity log.
   534→            </DialogDescription>
   535→          </DialogHeader>
   536→          <div className="grid gap-4">
   537→            <div className="grid gap-2">
   538→              <Label htmlFor="stock-qty">Quantity to add *</Label>
   539→              <Input
   540→                id="stock-qty"
   541→                type="number"
   542→                min={1}
   543→                value={stockForm.quantity}
   544→                onChange={(e) => setStockForm({ ...stockForm, quantity: e.target.value })}
   545→                placeholder="50"
   546→              />
   547→            </div>
   548→            <div className="grid gap-2">
   549→              <Label htmlFor="stock-price">Price per unit (₹)</Label>
   550→              <Input
   551→                id="stock-price"
   552→                type="number"
   553→                min={0}
   554→                value={stockForm.pricePerUnit}
   555→                onChange={(e) => setStockForm({ ...stockForm, pricePerUnit: e.target.value })}
   556→                placeholder="Leave blank to keep current"
   557→              />
   558→            </div>
   559→            <div className="grid gap-4 sm:grid-cols-2">
