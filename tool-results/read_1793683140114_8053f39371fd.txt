   220→   220→function parseRoute(json: string | null | undefined): { name: string; congestion: string; lengthKm: number }[] {
   221→   221→  if (!json) return [];
   222→   222→  try {
   223→   223→    const parsed = JSON.parse(json) as { name?: string; congestion?: string; lengthKm?: number }[];
   224→   224→    return Array.isArray(parsed)
   225→   225→      ? parsed.map((s) => ({
   226→   226→          name: String(s?.name ?? "Segment"),
   227→   227→          congestion: String(s?.congestion ?? "MODERATE"),
   228→   228→          lengthKm: Number(s?.lengthKm ?? 0),
   229→   229→        }))
   230→   230→      : [];
   231→   231→  } catch {
   232→   232→    return [];
   233→   233→  }
   234→   234→}
